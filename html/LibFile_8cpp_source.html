205bool LibFile::checkTimingArcMonotonicity(const json &cell, const json &pin, const json &arc,
206 const std::string &timing_arc_name, const bool is_slew) {
207 if (arc.contains("when") && !arc["when"].get<std::string>().empty()) {
208 logger_->debug(
209 "Checking cell: '{}', pin: '{}', related_pin: '{}', timing_arc: '{}', when: '{}'",
210 cell["cell_name"].get<std::string>(), pin["pin_name"].get<std::string>(),
211 arc["related_pin"].get<std::string>(), timing_arc_name, arc["when"].get<std::string>());
212 } else {
213 logger_->debug("Checking cell: '{}', pin: '{}', related_pin: '{}', timing_arc: '{}'",
214 cell["cell_name"].get<std::string>(), pin["pin_name"].get<std::string>(),
215 arc["related_pin"].get<std::string>(), timing_arc_name);
216 }
217 bool is_monotonic = true; // Assume the values are monotonic initially
218 if (arc.contains(timing_arc_name) && arc[timing_arc_name].contains("values")) {
219 std::vector<std::vector<double>> value_matrix;
220 // Generate a matrix of values from the JSON array
221 for (const auto &row_val : arc[timing_arc_name]["values"]) {
222 std::vector<double> row_data;
223 // Check if the value is an array
224 if (!row_val.is_array()) {
225 logger_->error("Invalid format: '{}' values should be an array of arrays in cell '{}', pin "
226 "'{}', related_pin '{}'",
227 timing_arc_name, cell["cell_name"].get<std::string>(),
228 pin["pin_name"].get<std::string>(), arc["related_pin"].get<std::string>());
229 return false;
230 }
231 // Check for non-numeric values
232 for (const auto &val : row_val) {
233 if (val.is_number()) {
234 row_data.push_back(val.get<double>());
235 } else {
236 logger_->warn("Non-numeric value found in '{}'.values, skipping value: {} in cell '{}', "
237 "pin '{}', related_pin '{}'",
238 timing_arc_name, val.dump(), cell["cell_name"].get<std::string>(),
239 pin["pin_name"].get<std::string>(), arc["related_pin"].get<std::string>());
240 }
241 }
242 value_matrix.push_back(row_data);
243 }
244 // Check the matrix for monotonicity
245 if (!value_matrix.empty() && !value_matrix[0].empty()) {
246 // Check the monotonic incrementality of rows (by output load capacitance, index 2)
247 for (size_t i = 0; i < value_matrix.size(); ++i) {
248 for (size_t j = 1; j < value_matrix[i].size(); ++j) {
249 if ((value_matrix[i][j] < value_matrix[i][j - 1] &&
250 pin["pin_name"] != arc["related_pin"]) ||
251 (value_matrix[i][j] == 0 && value_matrix[i][j - 1] == 0)) {
252 // If contains "when" key, log the when value
253 if (arc.contains("when") && !arc["when"].get<std::string>().empty()) {
254 logger_->warn("Non-monotonic (by load) '{}' values: ({}, {}) {} < ({}, {}) {} "
255 "for Cell: {} Pin: {}->{} when: \"{}\"",
256 timing_arc_name, i, j, value_matrix[i][j], i, j - 1,
257 value_matrix[i][j - 1], cell["cell_name"].get<std::string>(),
258 arc["related_pin"].get<std::string>(),
259 pin["pin_name"].get<std::string>(), arc["when"].get<std::string>());
260 } else {
261 logger_->warn("Non-monotonic (by load) '{}' values: ({}, {}) {} < ({}, {}) {} "
262 "for Cell: {} Pin: {}->{}",
263 timing_arc_name, i, j, value_matrix[i][j], i, j - 1,
264 value_matrix[i][j - 1], cell["cell_name"].get<std::string>(),
265 arc["related_pin"].get<std::string>(),
266 pin["pin_name"].get<std::string>());
267 }
268 is_monotonic = false;
269 }
270 }
271 }
272 if (is_slew) {
273 // Check the monotonic incrementality of columns (by input slew, index 1)
274 for (size_t j = 0; j < value_matrix[0].size(); ++j) {
275 for (size_t i = 1; i < value_matrix.size(); ++i) {
276 if ((value_matrix[i][j] < value_matrix[i - 1][j] &&
277 pin["pin_name"] != arc["related_pin"]) ||
278 (value_matrix[i][j] == 0 && value_matrix[i - 1][j] == 0)) {
279 // If contains "when" key, log the when value
280 if (arc.contains("when") && !arc["when"].get<std::string>().empty()) {
281 logger_->warn("Non-monotonic (by slew) '{}' values: ({}, {}) {} < ({}, {}) {} "
282 "for Cell: {} Pin: {}->{} when: \"{}\"",
283 timing_arc_name, i, j, value_matrix[i][j], i - 1, j,
284 value_matrix[i - 1][j], cell["cell_name"].get<std::string>(),
285 arc["related_pin"].get<std::string>(),
286 pin["pin_name"].get<std::string>(), arc["when"].get<std::string>());
287 } else {
288 logger_->warn("Non-monotonic (by slew) '{}' values: ({}, {}) {} < ({}, {}) {} "
289 "for Cell: {} Pin: {}->{}",
290 timing_arc_name, i, j, value_matrix[i][j], i - 1, j,
291 value_matrix[i - 1][j], cell["cell_name"].get<std::string>(),
292 arc["related_pin"].get<std::string>(),
293 pin["pin_name"].get<std::string>());
294 }
295 is_monotonic = false;
296 }
297 }
298 }
299 }
300 } else {
301 logger_->warn(
302 "Empty or invalid 'values' array found for cell: '{}', pin: '{}', related_pin: '{}', "
303 "timing_arc: '{}'",
304 cell["cell_name"].get<std::string>(), pin["pin_name"].get<std::string>(),
305 arc["related_pin"].get<std::string>(), timing_arc_name);
306 }
307 }
308 return is_monotonic;
309}

331void LibFile::mono(const bool is_slew) {
332 /*
333 * Use this command to check the following data in the current library to ensure
334 * the tables are monotonically increasing with respect to output load:
335 * cell_rise retaining_rise
336 * cell_fall retaining_fall
337 * rise_transition retain_rise_slew
338 * fall_transition retain_fall_slew
339 * mpw
340 */
341 logger_->info("Monotonicity check of '{}' starting ...", filename_);
342
343 if (!std::filesystem::exists(jsonname_)) {
344 logger_->info("JSON file not found. Parsing Liberty file first.");
345 this->parse();
346 this->writeJsonToFile();
347 } else {
348 // Read the JSON file into json object
349 std::ifstream in(jsonname_);
350 if (!in.is_open()) {
351 logger_->error("Could not open file '{}' for reading", jsonname_);
352 return;
353 }
354 try {
355 lib_json_ = json::parse(in);
356 } catch (const json::parse_error &e) {
357 logger_->error("JSON parsing error in file '{}': {}", jsonname_, e.what());
358 in.close();
359 return;
360 }
361 in.close();
362 }
363
364 std::map<std::string, bool> cell_monotonicity_status; // Track pass/fail status for each cell
365 std::vector<std::string> failed_cells; // List of failed cell names
366 int total_cells = 0;
367 int passed_cells = 0;
368
369 // Check the monotonicity of delay values
370 // The index 1 (time values) must be monotonically increasing and >= 0
371 for (const auto &cell : lib_json_["cells"]) {
372 total_cells++;
373 bool cell_is_monotonic = true; // Assume cell is monotonic initially
374 std::string cell_name = cell["cell_name"].get<std::string>();
375 cell_monotonicity_status[cell_name] =
376 true; // Initialize to pass, will be set to false if any check fails
377
378 if (cell.contains("output_pins")) {
379 for (const auto &pin : cell["output_pins"]) {
380 if (pin.contains("timing_arcs")) {
381 for (const auto &arc : pin["timing_arcs"]) {
382 // Check 4 timing arc names and accumulate monotonicity status
383 for (const auto &name :
384 {"cell_rise", "cell_fall", "rise_transition", "fall_transition"}) {
385 if (!checkTimingArcMonotonicity(cell, pin, arc, name, is_slew)) {
386 cell_is_monotonic = false;
387 }
388 }
389 }
390 }
391 }
392 }
393 if (cell.contains("input_pins")) {
394 for (const auto &pin : cell["input_pins"]) {
395 if (!pin.contains("clock")) {
396 continue; // Skip non-clock pins
397 }
398 // logger_->debug("Clock pin: {}", pin["pin_name"].get<std::string>());
399 if (pin.contains("timing_arcs")) {
400 for (const auto &arc : pin["timing_arcs"]) {
401 if (arc.contains("timing_type")) {
402 // logger_->debug("Timing type: {}", arc["timing_type"].get<std::string>());
403 if (arc["timing_type"].get<std::string>() == "min_pulse_width") {
404 for (const auto &name : {"rise_constraint", "fall_constraint"}) {
405 if (!checkTimingArcMonotonicity(cell, pin, arc, name, is_slew)) {
406 cell_is_monotonic = false;
407 }
408 }
409 }
410 }
411 }
412 }
413 }
414 }
415 // Update cell status based on all checks
416 cell_monotonicity_status[cell_name] = cell_is_monotonic;
417 if (cell_is_monotonic) {
418 passed_cells++;
419 } else {
420 failed_cells.push_back(cell_name);
421 }
422 }
423 logger_->info("Monotonicity check of '{}' completed.", filename_);
424
425 // Output summary statistics
426 logger_->info("validate_monotonicity : {} out of {} cells passed", passed_cells, total_cells);
427 logger_->info("validate_monotonicity : {} out of {} cells failed", total_cells - passed_cells,
428 total_cells);
429 if (!failed_cells.empty()) {
430 std::stringstream failed_cells_ss;
431 for (size_t i = 0; i < failed_cells.size(); ++i) {
432 failed_cells_ss << failed_cells[i];
433 if (i < failed_cells.size() - 1) {
434 failed_cells_ss << ", "; // Add comma if not the last element
435 }
436 }
437 logger_->info("Failed cell list : {}", failed_cells_ss.str());
438 }

460void LibFile::supercell(const int chain_length, const std::vector<std::string> &cell_names) {
461 /*
462 * Supercells are named as follows:
463 * <cellname>__X<chain_length>__<input_pin>__<output_pin>
464 */
465 logger_->info("Creating supercells for '{}'", filename_);
466
467 if (!std::filesystem::exists(jsonname_)) {
468 logger_->info("JSON file not found. Parsing Liberty file first.");
469 this->parse();
470 this->writeJsonToFile();
471 } else {
472 // Read the JSON file into json object
473 std::ifstream in(jsonname_);
474 if (!in.is_open()) {
475 logger_->error("Could not open file '{}' for reading", jsonname_);
476 return;
477 }
478 try {
479 lib_json_ = json::parse(in);
480 } catch (const json::parse_error &e) {
481 logger_->error("JSON parsing error in file '{}': {}", jsonname_, e.what());
482 in.close();
483 return;
484 }
485 in.close();
486 }
487
488 // write to .map file
489 std::ofstream out(basename_ + ".map");
490 if (!out.is_open()) {
491 logger_->error("Could not open file '{}' for writing", basename_ + ".map");
492 return;
493 }
494 // if celll_names is empty, create supercells for all cells
495 // else create supercells for only the specified cells
496
497 // Check if we're processing specific cells or all cells
498 bool process_all_cells = cell_names.empty();
499
500 if (process_all_cells) {
501 logger_->info("Creating supercells for ALL cells in '{}'", filename_);
502 } else {
503 logger_->info("Creating supercells for {} specified cells in '{}'", cell_names.size(),
504 filename_);
505 }
506
507 // Create a set for faster lookups if we have specific cell names
508 std::unordered_set<std::string> cell_set;
509 std::unordered_set<std::string> found_cells;
510 if (!process_all_cells) {
511 cell_set.insert(cell_names.begin(), cell_names.end());
512 }
513
514 for (const auto &cell : lib_json_["cells"]) {
515 bool is_sequential = false;
516 std::string cell_name = cell["cell_name"].get<std::string>();
517
518 // Skip cells not in the specified list
519 if (!process_all_cells && cell_set.find(cell_name) == cell_set.end()) {
520 continue;
521 }
522
523 // Mark this cell as found
524 if (!process_all_cells) {
525 found_cells.insert(cell_name);
526 }
527
528 logger_->debug("Creating supercells for cell: '{}'", cell_name);
529
530 std::vector<std::string> output_pins;
531 std::vector<std::string> input_pins;
532 // Extract input and output pins
533 if (cell.contains("output_pins")) {
534 for (const auto &pin : cell["output_pins"]) {
535 output_pins.push_back(pin["pin_name"].get<std::string>());
536 }
537 }
538 if (cell.contains("input_pins")) {
539 for (const auto &pin : cell["input_pins"]) {
540 if (pin.contains("clock")) {
541 is_sequential = true;
542 // clock pin not use for supercell
543 continue;
544 }
545 input_pins.push_back(pin["pin_name"].get<std::string>());
546 }
547 }
548
549 // create supercells for all combinations of input and output pins
550 for (const auto &output_pin : output_pins) {
551 for (const auto &input_pin : input_pins) {
552 if (is_sequential) {
553 const int chain_len = 1;
554 std::string supercell_name =
555 cell_name + "__X" + std::to_string(chain_len) + "__" + input_pin + "__" + output_pin;
556 out << cell_name << " " << supercell_name << std::endl;
557 } else {
558 std::string supercell_name = cell_name + "__X" + std::to_string(chain_length) + "__" +
559 input_pin + "__" + output_pin;
560 out << cell_name << " " << supercell_name << std::endl;
561 }
562 }
563 }
564 }
565
566 // Check for cells that were specified but not found
567 if (!process_all_cells) {
568 std::vector<std::string> not_found_cells;
569 for (const auto &requested_cell : cell_names) {
570 if (found_cells.find(requested_cell) == found_cells.end()) {
571 not_found_cells.push_back(requested_cell);
572 }
573 }
574
575 // Output warning for cells that weren't found
576 if (!not_found_cells.empty()) {
577 std::string missing_cells = not_found_cells[0];
578 for (size_t i = 1; i < not_found_cells.size(); ++i) {
579 missing_cells += ", " + not_found_cells[i];
580 }
581 logger_->warn("{} specified cell{} not found in the library: {}", not_found_cells.size(),
582 not_found_cells.size() > 1 ? "s were" : " was", missing_cells);
583 }
584 }
585
586 out.close();
587 logger_->info("Supercell creation complete in '{}'", basename_ + ".map");
588}

756 } else {

773 std::vector<std::string> all_output_ports;

778 // Collect input port names

812 top_instances << " " << module_name << " " << instance_name << " (" << instance_ports_str
813 << ");\n";
814 }
815
816 // Now generate the port list with all inputs first, then all outputs
817 std::stringstream top_ports;
818
819 // Add all input ports first
820 for (size_t i = 0; i < all_input_ports.size(); ++i) {
821 top_ports << all_input_ports[i];
822 if (i < all_input_ports.size() - 1 || !all_output_ports.empty()) {
823 top_ports << ", ";
824 }
825 }
826
827 // Then add all output ports
828 for (size_t i = 0; i < all_output_ports.size(); ++i) {

831 top_ports << ", ";
832 }
833 }
834
835 // Generate input and output declarations
836 std::stringstream top_inputs;
837 std::stringstream top_outputs;
838
839 // Generate input declarations

841 top_inputs << " input " << port << ";\n";
842 }
843
844 // Generate output declarations
845 for (const auto &port : all_output_ports) {
846 top_outputs << " output " << port << ";\n";
847 }
848
849 // Write the top module
850 final_out << "`timescale 1ns/10ps" << std::endl;
851 final_out << "module validate_top ( " << top_ports.str() << " );" << std::endl;
852 final_out << std::endl; // Add newline before port declarations
853 final_out << top_inputs.str();
854 final_out << top_outputs.str();
855 final_out << std::endl;

1013bool LibFile::modifySpiceNetlist(const std::string &v2lvsSpiceFile, // Input is the v2lvs output
1014 const std::string &finalSpiceFile, // Output is the final file
1015 const std::string &targetGlobalLine) {

1017
1018 std::ifstream inFile(v2lvsSpiceFile);
1019 // Write directly to the final output file, overwriting if it exists
1020 std::ofstream outFile(finalSpiceFile, std::ios::trunc);
1021
1022 if (!inFile) {
1023 logger_->error("Could not open input v2lvs SPICE file for modification: {}", v2lvsSpiceFile);
1024 return false;
1025 }
1026 if (!outFile) {
1027 logger_->error("Could not open final output SPICE file for writing: {}", finalSpiceFile);
1028 inFile.close(); // Close input file if output fails
1029 return false;
1030 }
1031
1032 // --- Add Metadata Comments ---
1033 outFile << "** Generated by " << APP_NAME << " v" << APP_VERSION << " from " << APP_AUTHOR;
1034 auto now = std::chrono::system_clock::now();
1035 std::time_t now_time_t = std::chrono::system_clock::to_time_t(now);
1036 // Use localtime_s on Windows, localtime_r on POSIX, or std::localtime (less safe)
1037 // std::tm now_tm;
1038 // localtime_s(&now_tm, &now_time_t); // Example for Windows
1039 std::tm *now_tm_ptr = std::localtime(&now_time_t); // Standard C++, potentially less thread-safe
1040 if (now_tm_ptr) {
1041 outFile << ". On: " << std::put_time(now_tm_ptr, "%c %Z") << " **\n" << std::endl;

1047 std::string line;
1048 bool includeFound = false;
1049 bool globalInserted = false;

1053

1055 // Trim leading/trailing whitespace

1074 bool isFinal = (outputNet == "CO" || outputNet == "S" || outputNet == "ZN");
1075 tokens[tokens.size() - 2] = outputNet + ":1"; // Modify output pin
1076
1077 // Reconstruct line with correct VDD/VSS order

1093 }

1100 }
1101
1102 // Insert the target global line after .INCLUDE

1129 }
1130 previousInstanceLine = "";
1131 }
1132
1133 inSubckt = true;
1134 instanceIndex = 0; // Reset R/C counter for new subcircuit
1135 outFile << line << " VDD VSS" << std::endl; // Add VDD VSS to ports

1139 inSubckt = false;

1177

1180 outFile << tokens[i] << " ";
1181 }

1183
1184 this->generateRCLines(outFile, outputNet, instanceIndex - 1,
1185 isFinal); // Use previous index
1186 } else {
1187 logger_->warn("Previous line stored but not recognized as instance: {}",
1188 previousInstanceLine);
1189 outFile << previousInstanceLine << std::endl;
1190 }

1234 outFile << "VDD VSS " << moduleName << std::endl;

1300

1343
1344 // Do NOT delete temp_verilog_file
1345
1346 if (ret != 0) {
1347 logger_->error("V2LVS command failed with exit code: {}. Command: {}", ret, v2lvs_cmd_full);
1348 // Attempt to remove potentially incomplete v2lvs output SPICE file
1349 std::remove(v2lvs_output_spice_file.c_str());

1351 }
1352 logger_->info("V2LVS completed successfully, output at '{}'.", v2lvs_output_spice_file);

1365 // 5. Final success message

1367 logger_->info("Temporary Verilog input kept in '{}'", temp_verilog_file);
1368 logger_->info("SPICE generation and post-processing complete. Final output in '{}'",
1369 final_output_spice_file);
1370}
