184void LibFile::writeToDB(const std::string &db_path, const std::string &pvt_corner,
185 double aged_year) {
186 logger_->info("Writing LUT entries to DB: '{}'", db_path);
187
188 if (!lib_json_.contains("cells") || !lib_json_["cells"].is_array()) {
189 logger_->warn("No cells found in parsed JSON, skipping DB write");
190 return;
191 }
192
193 LibDatabase database(db_path);
194 database.initialize();
195
196 int total_entries = 0;
197
198 // --- Extract header-level PVT ---
199 int process = lib_json_.value("process", 0);
200 double temperature = lib_json_.value("temperature", 0.0);
201 double voltage = lib_json_.value("voltage", 0.0);
202
203 // library_name = .lib basename (scenario_id form)
204 std::string library_name = basename_;
205
206 try {
207 // Walk: cells → output_pins → timing_arcs → arc_types
208 database.beginTransaction();
209
210 for (const auto &cell : lib_json_["cells"]) {
211 if (!cell.contains("cell_name") || !cell["cell_name"].is_string()) continue;
212 std::string cell_name = cell["cell_name"].get<std::string>();
213
214 if (!cell.contains("output_pins") || !cell["output_pins"].is_array()) continue;
215
216 for (const auto &pin : cell["output_pins"]) {
217 if (!pin.contains("pin_name") || !pin["pin_name"].is_string()) continue;
218 std::string output_pin = pin["pin_name"].get<std::string>();
219
220 if (!pin.contains("timing_arcs") || !pin["timing_arcs"].is_array()) continue;
221
222 for (const auto &arc : pin["timing_arcs"]) {
223 std::string related_pin = arc.value("related_pin", "");
224 std::string timing_sense = arc.value("timing_sense", "");
225 std::string timing_type = arc.value("timing_type", "");
226 std::string when = arc.value("when", "");
227
228 // Four arc types to check
229 static const char *ARC_NAMES[] = {"cell_rise", "cell_fall", "rise_transition",
230 "fall_transition"};
231 for (const char *arc_type : ARC_NAMES) {
232 if (!arc.contains(arc_type)) continue;
233 const auto &lut = arc[arc_type];
234 if (!lut.is_object()) continue;
235 if (!lut.contains("index_1") || !lut["index_1"].is_array() ||
236 !lut.contains("values") || !lut["values"].is_array()) continue;
237
238 // index_1 → rows_n
239 std::vector<double> index_1 = lut["index_1"].get<std::vector<double>>();
240 int rows_n = static_cast<int>(index_1.size());
241
242 // index_2 (optional, derive cols_n from values shape)
243 std::vector<double> index_2;
244 if (lut.contains("index_2") && lut["index_2"].is_array()) {
245 index_2 = lut["index_2"].get<std::vector<double>>();
246 }
247
248 // values → flatten
249 const auto &vals_json = lut["values"];
250 std::vector<double> values;
251 int cols_n = 0;
252 if (vals_json.is_array() && !vals_json.empty()) {
253 cols_n = static_cast<int>(vals_json[0].size());
254 for (const auto &row : vals_json) {
255 for (const auto &v : row) {
256 values.push_back(v.get<double>());
257 }
258 }
259 }
260
261 // If index_2 is missing, default to empty vector of size cols_n
262 if (index_2.empty()) {
263 index_2.resize(cols_n, 0.0);
264 }
265
266 database.writeLutEntry(
267 filename_, // file_path (original path for provenance)
268 library_name, pvt_corner, aged_year, cell_name, output_pin,
269 related_pin, timing_sense, timing_type, when, arc_type, rows_n,
270 cols_n, index_1, index_2, values, process, temperature, voltage);
271
272 total_entries++;
273 }
274 }
275 }
276 }
277
278 database.commitTransaction();
279
280 logger_->info("Wrote {} LUT entries to DB: '{}'", total_entries, db_path);
281} catch (const std::exception &e) {
282 // Rollback on any error, then rethrow
283 database.commitTransaction();
284 logger_->error("DB write failed, rolled back: {}", e.what());
285 throw;
286}

312bool LibFile::checkTimingArcMonotonicity(const json &cell, const json &pin, const json &arc,
313 const std::string &timing_arc_name, const bool is_slew) {
314 if (arc.contains("when") && !arc["when"].get<std::string>().empty()) {
315 logger_->debug(
316 "Checking cell: '{}', pin: '{}', related_pin: '{}', timing_arc: '{}', when: '{}'",
317 cell["cell_name"].get<std::string>(), pin["pin_name"].get<std::string>(),
318 arc["related_pin"].get<std::string>(), timing_arc_name, arc["when"].get<std::string>());
319 } else {
320 logger_->debug("Checking cell: '{}', pin: '{}', related_pin: '{}', timing_arc: '{}'",
321 cell["cell_name"].get<std::string>(), pin["pin_name"].get<std::string>(),
322 arc["related_pin"].get<std::string>(), timing_arc_name);
323 }
324 bool is_monotonic = true; // Assume the values are monotonic initially
325 if (arc.contains(timing_arc_name) && arc[timing_arc_name].contains("values")) {
326 std::vector<std::vector<double>> value_matrix;
327 // Generate a matrix of values from the JSON array
328 for (const auto &row_val : arc[timing_arc_name]["values"]) {
329 std::vector<double> row_data;
330 // Check if the value is an array
331 if (!row_val.is_array()) {
332 logger_->error("Invalid format: '{}' values should be an array of arrays in cell '{}', pin "
333 "'{}', related_pin '{}'",
334 timing_arc_name, cell["cell_name"].get<std::string>(),
335 pin["pin_name"].get<std::string>(), arc["related_pin"].get<std::string>());
336 return false;
337 }
338 // Check for non-numeric values
339 for (const auto &val : row_val) {
340 if (val.is_number()) {
341 row_data.push_back(val.get<double>());
342 } else {
343 logger_->warn("Non-numeric value found in '{}'.values, skipping value: {} in cell '{}', "
344 "pin '{}', related_pin '{}'",
345 timing_arc_name, val.dump(), cell["cell_name"].get<std::string>(),
346 pin["pin_name"].get<std::string>(), arc["related_pin"].get<std::string>());
347 }
348 }
349 value_matrix.push_back(row_data);
350 }
351 // Check the matrix for monotonicity
352 if (!value_matrix.empty() && !value_matrix[0].empty()) {
353 // Check the monotonic incrementality of rows (by output load capacitance, index 2)
354 for (size_t i = 0; i < value_matrix.size(); ++i) {
355 for (size_t j = 1; j < value_matrix[i].size(); ++j) {
356 if ((value_matrix[i][j] < value_matrix[i][j - 1] &&
357 pin["pin_name"] != arc["related_pin"]) ||
358 (value_matrix[i][j] == 0 && value_matrix[i][j - 1] == 0)) {
359 // If contains "when" key, log the when value
360 if (arc.contains("when") && !arc["when"].get<std::string>().empty()) {
361 logger_->warn("Non-monotonic (by load) '{}' values: ({}, {}) {} < ({}, {}) {} "
362 "for Cell: {} Pin: {}->{} when: \"{}\"",
363 timing_arc_name, i, j, value_matrix[i][j], i, j - 1,
364 value_matrix[i][j - 1], cell["cell_name"].get<std::string>(),
365 arc["related_pin"].get<std::string>(),
366 pin["pin_name"].get<std::string>(), arc["when"].get<std::string>());
367 } else {
368 logger_->warn("Non-monotonic (by load) '{}' values: ({}, {}) {} < ({}, {}) {} "
369 "for Cell: {} Pin: {}->{}",
370 timing_arc_name, i, j, value_matrix[i][j], i, j - 1,
371 value_matrix[i][j - 1], cell["cell_name"].get<std::string>(),
372 arc["related_pin"].get<std::string>(),
373 pin["pin_name"].get<std::string>());
374 }
375 is_monotonic = false;
376 }
377 }
378 }
379 if (is_slew) {
380 // Check the monotonic incrementality of columns (by input slew, index 1)
381 for (size_t j = 0; j < value_matrix[0].size(); ++j) {
382 for (size_t i = 1; i < value_matrix.size(); ++i) {
383 if ((value_matrix[i][j] < value_matrix[i - 1][j] &&
384 pin["pin_name"] != arc["related_pin"]) ||
385 (value_matrix[i][j] == 0 && value_matrix[i - 1][j] == 0)) {
386 // If contains "when" key, log the when value
387 if (arc.contains("when") && !arc["when"].get<std::string>().empty()) {
388 logger_->warn("Non-monotonic (by slew) '{}' values: ({}, {}) {} < ({}, {}) {} "
389 "for Cell: {} Pin: {}->{} when: \"{}\"",
390 timing_arc_name, i, j, value_matrix[i][j], i - 1, j,
391 value_matrix[i - 1][j], cell["cell_name"].get<std::string>(),
392 arc["related_pin"].get<std::string>(),
393 pin["pin_name"].get<std::string>(), arc["when"].get<std::string>());
394 } else {
395 logger_->warn("Non-monotonic (by slew) '{}' values: ({}, {}) {} < ({}, {}) {} "
396 "for Cell: {} Pin: {}->{}",
397 timing_arc_name, i, j, value_matrix[i][j], i - 1, j,
398 value_matrix[i - 1][j], cell["cell_name"].get<std::string>(),
399 arc["related_pin"].get<std::string>(),
400 pin["pin_name"].get<std::string>());
401 }
402 is_monotonic = false;
403 }
404 }
405 }
406 }
407 } else {
408 logger_->warn(
409 "Empty or invalid 'values' array found for cell: '{}', pin: '{}', related_pin: '{}', "
410 "timing_arc: '{}'",
411 cell["cell_name"].get<std::string>(), pin["pin_name"].get<std::string>(),
412 arc["related_pin"].get<std::string>(), timing_arc_name);
413 }
414 }
415 return is_monotonic;
416}

438void LibFile::mono(const bool is_slew) {
439 /*
440 * Use this command to check the following data in the current library to ensure
441 * the tables are monotonically increasing with respect to output load:
442 * cell_rise retaining_rise
443 * cell_fall retaining_fall
444 * rise_transition retain_rise_slew
445 * fall_transition retain_fall_slew
446 * mpw
447 */
448 logger_->info("Monotonicity check of '{}' starting ...", filename_);
449
450 if (!std::filesystem::exists(jsonname_)) {
451 logger_->info("JSON file not found. Parsing Liberty file first.");
452 this->parse();
453 this->writeJsonToFile();
454 } else {
455 // Read the JSON file into json object
456 std::ifstream in(jsonname_);
457 if (!in.is_open()) {
458 logger_->error("Could not open file '{}' for reading", jsonname_);
459 return;
460 }
461 try {
462 lib_json_ = json::parse(in);
463 } catch (const json::parse_error &e) {
464 logger_->error("JSON parsing error in file '{}': {}", jsonname_, e.what());
465 in.close();
466 return;
467 }
468 in.close();
469 }
470
471 std::map<std::string, bool> cell_monotonicity_status; // Track pass/fail status for each cell
472 std::vector<std::string> failed_cells; // List of failed cell names
473 int total_cells = 0;
474 int passed_cells = 0;
475
476 // Check the monotonicity of delay values
477 // The index 1 (time values) must be monotonically increasing and >= 0
478 for (const auto &cell : lib_json_["cells"]) {
479 total_cells++;
480 bool cell_is_monotonic = true; // Assume cell is monotonic initially
481 std::string cell_name = cell["cell_name"].get<std::string>();
482 cell_monotonicity_status[cell_name] =
483 true; // Initialize to pass, will be set to false if any check fails
484
485 if (cell.contains("output_pins")) {
486 for (const auto &pin : cell["output_pins"]) {
487 if (pin.contains("timing_arcs")) {
488 for (const auto &arc : pin["timing_arcs"]) {
489 // Check 4 timing arc names and accumulate monotonicity status
490 for (const auto &name :
491 {"cell_rise", "cell_fall", "rise_transition", "fall_transition"}) {
492 if (!checkTimingArcMonotonicity(cell, pin, arc, name, is_slew)) {
493 cell_is_monotonic = false;
494 }
495 }
496 }
497 }
498 }
499 }
500 if (cell.contains("input_pins")) {
501 for (const auto &pin : cell["input_pins"]) {
502 if (!pin.contains("clock")) {
503 continue; // Skip non-clock pins
504 }
505 // logger_->debug("Clock pin: {}", pin["pin_name"].get<std::string>());
506 if (pin.contains("timing_arcs")) {
507 for (const auto &arc : pin["timing_arcs"]) {
508 if (arc.contains("timing_type")) {
509 // logger_->debug("Timing type: {}", arc["timing_type"].get<std::string>());
510 if (arc["timing_type"].get<std::string>() == "min_pulse_width") {
511 for (const auto &name : {"rise_constraint", "fall_constraint"}) {
512 if (!checkTimingArcMonotonicity(cell, pin, arc, name, is_slew)) {
513 cell_is_monotonic = false;
514 }
515 }
516 }
517 }
518 }
519 }
520 }
521 }
522 // Update cell status based on all checks
523 cell_monotonicity_status[cell_name] = cell_is_monotonic;
524 if (cell_is_monotonic) {
525 passed_cells++;
526 } else {
527 failed_cells.push_back(cell_name);
528 }
529 }
530 logger_->info("Monotonicity check of '{}' completed.", filename_);
531
532 // Output summary statistics
533 logger_->info("validate_monotonicity : {} out of {} cells passed", passed_cells, total_cells);
534 logger_->info("validate_monotonicity : {} out of {} cells failed", total_cells - passed_cells,
535 total_cells);
536 if (!failed_cells.empty()) {
537 std::stringstream failed_cells_ss;
538 for (size_t i = 0; i < failed_cells.size(); ++i) {
539 failed_cells_ss << failed_cells[i];
540 if (i < failed_cells.size() - 1) {
541 failed_cells_ss << ", "; // Add comma if not the last element
542 }
543 }
544 logger_->info("Failed cell list : {}", failed_cells_ss.str());
545 }

567void LibFile::supercell(const int chain_length, const std::vector<std::string> &cell_names) {
568 /*
569 * Supercells are named as follows:
570 * <cellname>__X<chain_length>__<input_pin>__<output_pin>
571 */
572 logger_->info("Creating supercells for '{}'", filename_);
573
574 if (!std::filesystem::exists(jsonname_)) {
575 logger_->info("JSON file not found. Parsing Liberty file first.");
576 this->parse();
577 this->writeJsonToFile();
578 } else {
579 // Read the JSON file into json object
580 std::ifstream in(jsonname_);
581 if (!in.is_open()) {
582 logger_->error("Could not open file '{}' for reading", jsonname_);
583 return;
584 }
585 try {
586 lib_json_ = json::parse(in);
587 } catch (const json::parse_error &e) {
588 logger_->error("JSON parsing error in file '{}': {}", jsonname_, e.what());
589 in.close();
590 return;
591 }
592 in.close();
593 }
594
595 // write to .map file
596 std::ofstream out(basename_ + ".map");
597 if (!out.is_open()) {
598 logger_->error("Could not open file '{}' for writing", basename_ + ".map");
599 return;
600 }
601 // if celll_names is empty, create supercells for all cells
602 // else create supercells for only the specified cells
603
604 // Check if we're processing specific cells or all cells
605 bool process_all_cells = cell_names.empty();
606
607 if (process_all_cells) {
608 logger_->info("Creating supercells for ALL cells in '{}'", filename_);
609 } else {
610 logger_->info("Creating supercells for {} specified cells in '{}'", cell_names.size(),
611 filename_);
612 }
613
614 // Create a set for faster lookups if we have specific cell names
615 std::unordered_set<std::string> cell_set;
616 std::unordered_set<std::string> found_cells;
617 if (!process_all_cells) {
618 cell_set.insert(cell_names.begin(), cell_names.end());
619 }
620
621 for (const auto &cell : lib_json_["cells"]) {
622 bool is_sequential = false;
623 std::string cell_name = cell["cell_name"].get<std::string>();
624
625 // Skip cells not in the specified list
626 if (!process_all_cells && cell_set.find(cell_name) == cell_set.end()) {
627 continue;
628 }
629
630 // Mark this cell as found
631 if (!process_all_cells) {
632 found_cells.insert(cell_name);
633 }
634
635 logger_->debug("Creating supercells for cell: '{}'", cell_name);
636
637 std::vector<std::string> output_pins;
638 std::vector<std::string> input_pins;
639 // Extract input and output pins
640 if (cell.contains("output_pins")) {
641 for (const auto &pin : cell["output_pins"]) {
642 output_pins.push_back(pin["pin_name"].get<std::string>());
643 }
644 }
645 if (cell.contains("input_pins")) {
646 for (const auto &pin : cell["input_pins"]) {
647 if (pin.contains("clock")) {
648 is_sequential = true;
649 // clock pin not use for supercell
650 continue;
651 }
652 input_pins.push_back(pin["pin_name"].get<std::string>());
653 }
654 }
655
656 // create supercells for all combinations of input and output pins
657 for (const auto &output_pin : output_pins) {
658 for (const auto &input_pin : input_pins) {
659 if (is_sequential) {
660 const int chain_len = 1;
661 std::string supercell_name =
662 cell_name + "__X" + std::to_string(chain_len) + "__" + input_pin + "__" + output_pin;
663 out << cell_name << " " << supercell_name << std::endl;
664 } else {
665 std::string supercell_name = cell_name + "__X" + std::to_string(chain_length) + "__" +
666 input_pin + "__" + output_pin;
667 out << cell_name << " " << supercell_name << std::endl;
668 }
669 }
670 }
671 }
672
673 // Check for cells that were specified but not found
674 if (!process_all_cells) {
675 std::vector<std::string> not_found_cells;
676 for (const auto &requested_cell : cell_names) {
677 if (found_cells.find(requested_cell) == found_cells.end()) {
678 not_found_cells.push_back(requested_cell);
679 }
680 }
681
682 // Output warning for cells that weren't found
683 if (!not_found_cells.empty()) {
684 std::string missing_cells = not_found_cells[0];
685 for (size_t i = 1; i < not_found_cells.size(); ++i) {
686 missing_cells += ", " + not_found_cells[i];
687 }
688 logger_->warn("{} specified cell{} not found in the library: {}", not_found_cells.size(),
689 not_found_cells.size() > 1 ? "s were" : " was", missing_cells);
690 }
691 }
692
693 out.close();
694 logger_->info("Supercell creation complete in '{}'", basename_ + ".map");
695}

721void LibFile::verilog(const int chain_length, const std::vector<std::string> &cell_names) {
722 logger_->info("Creating Verilog for '{}'", filename_);
723
724 this->supercell(chain_length, cell_names);
725
726 // Create a temporary file for individual modules
727 std::string temp_file = basename_ + "_temp.v";
728 std::ofstream out(temp_file);
729 if (!out.is_open()) {
730 logger_->error("Could not open temp file '{}' for writing", temp_file);
731 return;
732 }
733
734 // Read the .map file into a vector to preserve all entries and their order
735 std::vector<std::pair<std::string, std::string>> supercell_entries;
736 std::ifstream in(basename_ + ".map");
737 if (!in.is_open()) {
738 logger_->error("Could not open file '{}' for reading", basename_ + ".map");
739 return;
740 }
741
742 std::string line;
743 while (std::getline(in, line)) {
744 std::istringstream iss(line);
745 std::string cell_name, supercell_name;
746 if (!(iss >> cell_name >> supercell_name)) {
747 logger_->warn("Invalid line format in map file: '{}'", line);
748 continue;
749 }
750 supercell_entries.push_back({cell_name, supercell_name});
751 }
752 in.close();
753 logger_->info("Read {} supercells from '{}'", supercell_entries.size(), basename_ + ".map");
754
755 // Store module information for top-level creation
756 std::vector<std::string> module_names;
757 std::map<std::string, std::vector<std::string>> module_inputs;
758 std::map<std::string, std::vector<std::string>> module_outputs;
759
760 // Generate verilog module for each supercell
761 for (const auto &supercell_entry : supercell_entries) {
762 // Check if the cell is sequential
763 bool is_sequential = false;
764 // Count the number of instances will be created in verilog
765 int instance_count = 0;
766 // Get original cell name from pair
767 const std::string &cell_name = supercell_entry.first;
768 // Get supercell name(as well as module name) from pair
769 const std::string &module_name = supercell_entry.second;
770
771 // Store module name for top-level creation
772 module_names.push_back(module_name);
773
774 logger_->info("Creating Module: '{}' from Cell '{}", module_name, cell_name);
775
776 // Get the cell JSON object
777 json cell_json;
778 for (const auto &cell : lib_json_["cells"]) {
779 if (cell["cell_name"].get<std::string>() == cell_name) {
780 cell_json = cell;
781 break;
782 }
783 }
784
785 // Get input/output pins from the cell JSON object
786 std::vector<std::string> input_pins;
787 std::vector<std::string> output_pins;
788 std::stringstream input_pins_ss;
789 std::stringstream output_pins_ss;
790 if (cell_json.contains("input_pins")) {
791 for (const auto &pin : cell_json["input_pins"]) {
792 if (pin.contains("clock")) {
793 is_sequential = true;
794 }
795 std::string pin_name = pin["pin_name"].get<std::string>();
796 input_pins.push_back(pin_name);
797 input_pins_ss << pin_name << ", ";
798
799 // Store input pin name for top-level creation
800 module_inputs[module_name].push_back(pin_name);
801 }
802 }
803 if (cell_json.contains("output_pins")) {
804 for (const auto &pin : cell_json["output_pins"]) {
805 std::string pin_name = pin["pin_name"].get<std::string>();
806 output_pins.push_back(pin_name);
807 output_pins_ss << pin_name << ", ";
808
809 // Store output pin name for top-level creation
810 module_outputs[module_name].push_back(pin_name);
811 }
812 }
813 logger_->debug("Input pins set: {}", input_pins_ss.str());
814 logger_->debug("Output pins set: {}", output_pins_ss.str());
815
816 // Sequential cells will have only one instance
817 // Combinational cells will have instances equal to chain length
818 if (is_sequential) {
819 instance_count = 1;
820 } else {
821 instance_count = chain_length;
822 }
823
824 // Create ANSI port list
825 std::string port_list_text = "(";
826 // Add input ports
827 bool isFirstPort = true;
828 for (const auto &input_pin : input_pins) {
829 if (!isFirstPort) {
830 port_list_text += ", ";
831 }
832 port_list_text += "input " + input_pin;
833 isFirstPort = false;
834 }
835 // Add output ports
836 for (const auto &output_pin : output_pins) {
837 if (!isFirstPort) {
838 port_list_text += ", ";
839 }
840 port_list_text += "output " + output_pin;
841 isFirstPort = false;
842 }
843 port_list_text += ")";
844 logger_->debug("ANSI Port list: {}", port_list_text);
845
846 // Creat full module header text
847 std::string fullModuleText = "module " + module_name + port_list_text + ";\nendmodule";
848 logger_->debug("Full module text: {}", fullModuleText);
849
850 // Generate the syntax tree for the module using slang library
851 auto tree = slang::syntax::SyntaxTree::fromText(fullModuleText);
852 if (tree) {
853 // Add ports to the syntax tree
854 ModuleRewriter rewriter(input_pins, output_pins, supercell_entry, instance_count, logger_);
855
856 tree = rewriter.transform(tree);
857 // Revisit the syntax tree to check architecture
858 // rewriter.visit(tree->root());
859
860 // Output the transformed syntax tree
861 out << "`timescale 1ns/10ps" << std::endl;
862 out << slang::syntax::SyntaxPrinter::printFile(*tree) << std::endl << std::endl;
863 } else {
864 logger_->error("Failed to create syntax tree for module '{}'", module_name);
865 continue;
866 }
867 }
868
869 out.close();
870
871 // Create the top-level module
872 std::ofstream final_out(basename_ + ".v");
873 if (!final_out.is_open()) {
874 logger_->error("Could not open file '{}' for writing", basename_ + ".v");
875 return;
876 }
877
878 // Collect all port names separately for inputs and outputs
879 std::vector<std::string> all_input_ports;
880 std::vector<std::string> all_output_ports;
881 std::stringstream top_instances;
882
883 // First pass: collect all port names
884 for (const auto &module_name : module_names) {
885 // Collect input port names
886 for (const auto &pin : module_inputs[module_name]) {
887 all_input_ports.push_back(module_name + "__" + pin);
888 }
889
890 // Collect output port names
891 for (const auto &pin : module_outputs[module_name]) {
892 all_output_ports.push_back(module_name + "__" + pin);
893 }
894
895 // Create instance
896 std::string instance_name = "I_" + module_name.substr(0, module_name.find("__X"));
897 for (size_t i = module_name.find("__X") + 3; i < module_name.length(); i++) {
898 if (module_name[i] == '_' && module_name[i + 1] == '_') {
899 instance_name += "__" + module_name.substr(i + 2);
900 break;
901 }
902 }
903
904 // Generate port connections for this instance
905 std::stringstream instance_ports;
906 for (const auto &pin : module_inputs[module_name]) {
907 instance_ports << "." << pin << "(" << module_name << "__" << pin << "), ";
908 }
909 for (const auto &pin : module_outputs[module_name]) {
910 instance_ports << "." << pin << "(" << module_name << "__" << pin << "), ";
911 }
912
913 // Remove last comma and space
914 std::string instance_ports_str = instance_ports.str();
915 if (!instance_ports_str.empty()) {
916 instance_ports_str = instance_ports_str.substr(0, instance_ports_str.length() - 2);
917 }
918
919 top_instances << " " << module_name << " " << instance_name << " (" << instance_ports_str
920 << ");\n";
921 }
922
923 // Now generate the port list with all inputs first, then all outputs
924 std::stringstream top_ports;
925
926 // Add all input ports first
927 for (size_t i = 0; i < all_input_ports.size(); ++i) {
928 top_ports << all_input_ports[i];
929 if (i < all_input_ports.size() - 1 || !all_output_ports.empty()) {
930 top_ports << ", ";
931 }
932 }
933
934 // Then add all output ports
935 for (size_t i = 0; i < all_output_ports.size(); ++i) {
936 top_ports << all_output_ports[i];
937 if (i < all_output_ports.size() - 1) {
938 top_ports << ", ";
939 }
940 }
941
942 // Generate input and output declarations
943 std::stringstream top_inputs;
944 std::stringstream top_outputs;
945
946 // Generate input declarations
947 for (const auto &port : all_input_ports) {
948 top_inputs << " input " << port << ";\n";
949 }
950
951 // Generate output declarations
952 for (const auto &port : all_output_ports) {
953 top_outputs << " output " << port << ";\n";
954 }
955
956 // Write the top module
957 final_out << "`timescale 1ns/10ps" << std::endl;
958 final_out << "module validate_top ( " << top_ports.str() << " );" << std::endl;
959 final_out << std::endl; // Add newline before port declarations
960 final_out << top_inputs.str();
961 final_out << top_outputs.str();
962 final_out << std::endl;
963 final_out << top_instances.str();
964 final_out << "endmodule" << std::endl << std::endl;
965
966 // Append the module definitions from the temporary file
967 std::ifstream temp_in(temp_file);
968 if (temp_in.is_open()) {
969 final_out << temp_in.rdbuf();
970 temp_in.close();
971 } else {
972 logger_->error("Could not open temp file '{}' for reading", temp_file);
973 }
974
975 final_out.close();
976
977 // Remove temporary file
978 // if (std::filesystem::exists(temp_file)) {
979 // std::filesystem::remove(temp_file);
980 // }
981
982 logger_->info("Verilog creation complete in '{}'", basename_ + ".v");
983}

1038void LibFile::generateRCLines(std::ofstream &outFile, const std::string &netName, int instanceIndex,
1039 bool isFinalStage) {
1040 // --- Default RC Values (as observed in the target SPICE) ---
1041 const double R1_INTERMEDIATE = 0.01;
1042 const double C1_INTERMEDIATE = 5.3e-16;
1043 const double R2_INTERMEDIATE = 0.01;
1044 const double R1_FINAL = 1e-2;
1045 const double C1_FINAL = 1e-18;
1046 const std::string CAP_GROUND = "COREGND1"; // Assumed ground for capacitors
1047
1048 std::string r1Name = "R1_" + std::to_string(instanceIndex);
1049 std::string c1Name = "C1_" + std::to_string(instanceIndex);
1050 std::string r2Name = "R2_" + std::to_string(instanceIndex);
1051 std::string node1 = netName + ":1"; // Intermediate node 1
1052 std::string node2 = netName + ":2"; // Intermediate node 2
1053
1054 // Set precision for scientific notation output
1055 outFile << std::scientific << std::setprecision(1);
1056
1057 if (!isFinalStage) {
1058 // Intermediate RCR structure (R1-C1-R2)
1059 outFile << r1Name << " " << node1 << " " << node2 << " " << R1_INTERMEDIATE << std::endl;
1060 outFile << c1Name << " " << node2 << " " << CAP_GROUND << " " << C1_INTERMEDIATE << std::endl;
1061 outFile << r2Name << " " << node2 << " " << netName << " " << R2_INTERMEDIATE << std::endl;
1062 } else {
1063 // Final RC structure (R1-C1 connected to the instance output node :1)
1064 // Note: The target SPICE connects R1 between node1 and the final port (netName).
1065 outFile << r1Name << " " << node1 << " " << netName << " " << R1_FINAL << std::endl;
1066 outFile << c1Name << " " << node1 << " " << CAP_GROUND << " " << C1_FINAL << std::endl;
1067 }
1068 // Reset precision/format to default
1069 outFile << std::defaultfloat << std::setprecision(6);
1070}

1120bool LibFile::modifySpiceNetlist(const std::string &v2lvsSpiceFile, // Input is the v2lvs output
1121 const std::string &finalSpiceFile, // Output is the final file
1122 const std::string &targetGlobalLine) {
1123 logger_->info("Post-processing SPICE netlist: {} -> {}", v2lvsSpiceFile, finalSpiceFile);
1124
1125 std::ifstream inFile(v2lvsSpiceFile);
1126 // Write directly to the final output file, overwriting if it exists
1127 std::ofstream outFile(finalSpiceFile, std::ios::trunc);
1128
1129 if (!inFile) {
1130 logger_->error("Could not open input v2lvs SPICE file for modification: {}", v2lvsSpiceFile);
1131 return false;
1132 }
1133 if (!outFile) {
1134 logger_->error("Could not open final output SPICE file for writing: {}", finalSpiceFile);
1135 inFile.close(); // Close input file if output fails
1136 return false;
1137 }
1138
1139 // --- Add Metadata Comments ---
1140 outFile << "** Generated by " << APP_NAME << " v" << APP_VERSION << " from " << APP_AUTHOR;
1141 auto now = std::chrono::system_clock::now();
1142 std::time_t now_time_t = std::chrono::system_clock::to_time_t(now);
1143 // Use localtime_s on Windows, localtime_r on POSIX, or std::localtime (less safe)
1144 // std::tm now_tm;
1145 // localtime_s(&now_tm, &now_time_t); // Example for Windows
1146 std::tm *now_tm_ptr = std::localtime(&now_time_t); // Standard C++, potentially less thread-safe
1147 if (now_tm_ptr) {
1148 outFile << ". On: " << std::put_time(now_tm_ptr, "%c %Z") << " **\n" << std::endl;
1149 } else {
1150 outFile << ". Timestamp unavailable **\n" << std::endl;
1151 }
1152 // --- End Metadata ---
1153
1154 std::string line;
1155 bool includeFound = false;
1156 bool globalInserted = false;
1157 bool inSubckt = false;
1158 std::string previousInstanceLine = "";
1159 int instanceIndex = 0; // Counter for R/C naming within a subckt
1160
1161 while (std::getline(inFile, line)) {
1162 // Trim leading/trailing whitespace
1163 line.erase(0, line.find_first_not_of(" \t\n\r\f\v"));
1164 line.erase(line.find_last_not_of(" \t\n\r\f\v") + 1);
1165
1166 // Skip empty lines and v2lvs header comments
1167 if (line.empty() || line.find("$ Spice netlist generated by v2lvs") == 0 ||
1168 line.find("$ v2") == 0 || // Catch version line too
1169 line.find("*.BUSDELIMITER") == 0) {
1170 continue; // Skip these lines entirely
1171 }
1172
1173 // Write other comments directly after processing any pending instance
1174 if (line[0] == '*') {
1175 // Process previous instance before writing comment if needed
1176 if (!previousInstanceLine.empty()) {
1177 std::vector<std::string> tokens = this->splitString(previousInstanceLine);
1178 if (tokens.size() > 2 && (tokens[0][0] == 'X' || tokens[0][0] == 'x')) {
1179 std::string moduleName = tokens.back();
1180 std::string outputNet = tokens[tokens.size() - 2]; // Assumption
1181 bool isFinal = (outputNet == "CO" || outputNet == "S" || outputNet == "ZN");
1182 tokens[tokens.size() - 2] = outputNet + ":1"; // Modify output pin
1183
1184 // Reconstruct line with correct VDD/VSS order
1185 for (size_t i = 0; i < tokens.size() - 1; ++i) { // Up to module name
1186 outFile << tokens[i] << " ";
1187 }
1188 outFile << "VDD VSS " << moduleName << std::endl; // Insert VDD VSS before module name
1189
1190 this->generateRCLines(outFile, outputNet, instanceIndex - 1, isFinal);
1191 } else {
1192 logger_->warn("Previous line stored but not recognized as instance: {}",
1193 previousInstanceLine);
1194 outFile << previousInstanceLine << std::endl;
1195 }
1196 previousInstanceLine = ""; // Clear after processing
1197 }
1198 outFile << line << std::endl; // Write the comment line
1199 continue;
1200 }
1201
1202 // Handle .INCLUDE
1203 if (line.find(".INCLUDE") == 0) {
1204 outFile << line << std::endl;
1205 includeFound = true;
1206 continue; // Process next line to insert global
1207 }
1208
1209 // Insert the target global line after .INCLUDE
1210 if (includeFound && !globalInserted) {
1211 outFile << targetGlobalLine << std::endl << std::endl; // Add extra newline for spacing
1212 globalInserted = true;
1213 // Fall through to process the current line
1214 }
1215
1216 // Handle .SUBCKT
1217 if (line.find(".SUBCKT") == 0 || line.find(".subckt") == 0) { // Handle case-insensitivity
1218 if (!previousInstanceLine.empty()) {
1219 std::vector<std::string> tokens = this->splitString(previousInstanceLine);
1220 if (tokens.size() > 2 && (tokens[0][0] == 'X' || tokens[0][0] == 'x')) {
1221 std::string moduleName = tokens.back();
1222 std::string outputNet = tokens[tokens.size() - 2];
1223 bool isFinal = (outputNet == "CO" || outputNet == "S" || outputNet == "ZN");
1224 tokens[tokens.size() - 2] = outputNet + ":1";
1225
1226 for (size_t i = 0; i < tokens.size() - 1; ++i) {
1227 outFile << tokens[i] << " ";
1228 }
1229 outFile << "VDD VSS " << moduleName << std::endl;
1230
1231 this->generateRCLines(outFile, outputNet, instanceIndex - 1, isFinal);
1232 } else {
1233 logger_->warn("Previous line stored but not recognized as instance: {}",
1234 previousInstanceLine);
1235 outFile << previousInstanceLine << std::endl;
1236 }
1237 previousInstanceLine = "";
1238 }
1239
1240 inSubckt = true;
1241 instanceIndex = 0; // Reset R/C counter for new subcircuit
1242 outFile << line << " VDD VSS" << std::endl; // Add VDD VSS to ports
1243 }
1244 // Handle .ENDS
1245 else if (line.find(".ENDS") == 0 || line.find(".ends") == 0) { // Handle case-insensitivity
1246 inSubckt = false;
1247 // Process the last instance line *before* writing .ENDS
1248 if (!previousInstanceLine.empty()) {
1249 std::vector<std::string> tokens = this->splitString(previousInstanceLine);
1250 if (tokens.size() > 2 && (tokens[0][0] == 'X' || tokens[0][0] == 'x')) {
1251 std::string moduleName = tokens.back();
1252 std::string outputNet = tokens[tokens.size() - 2];
1253 bool isFinal = (outputNet == "CO" || outputNet == "S" || outputNet == "ZN");
1254 tokens[tokens.size() - 2] = outputNet + ":1";
1255
1256 for (size_t i = 0; i < tokens.size() - 1; ++i) {
1257 outFile << tokens[i] << " ";
1258 }
1259 outFile << "VDD VSS " << moduleName << std::endl;
1260
1261 this->generateRCLines(outFile, outputNet, instanceIndex - 1, isFinal);
1262 } else {
1263 logger_->warn("Previous line stored but not recognized as instance: {}",
1264 previousInstanceLine);
1265 outFile << previousInstanceLine << std::endl;
1266 }
1267 previousInstanceLine = ""; // Clear after processing
1268 }
1269 outFile << line << std::endl << std::endl; // Add extra newline after .ENDS
1270 }
1271 // Skip original .GLOBAL lines
1272 else if (line.find(".GLOBAL") == 0 || line.find(".global") == 0) {
1273 continue;
1274 }
1275 // Handle Instance lines
1276 else if (inSubckt && (line[0] == 'X' || line[0] == 'x')) {
1277 if (!previousInstanceLine.empty()) {
1278 std::vector<std::string> tokens = this->splitString(previousInstanceLine);
1279 if (tokens.size() > 2 && (tokens[0][0] == 'X' || tokens[0][0] == 'x')) {
1280 std::string moduleName = tokens.back();
1281 std::string outputNet = tokens[tokens.size() - 2]; // Assumption
1282 bool isFinal = (outputNet == "CO" || outputNet == "S" || outputNet == "ZN");
1283 tokens[tokens.size() - 2] = outputNet + ":1"; // Modify output pin
1284
1285 // Reconstruct line with correct VDD/VSS order
1286 for (size_t i = 0; i < tokens.size() - 1; ++i) { // Up to module name
1287 outFile << tokens[i] << " ";
1288 }
1289 outFile << "VDD VSS " << moduleName << std::endl; // Insert VDD VSS before module name
1290
1291 this->generateRCLines(outFile, outputNet, instanceIndex - 1,
1292 isFinal); // Use previous index
1293 } else {
1294 logger_->warn("Previous line stored but not recognized as instance: {}",
1295 previousInstanceLine);
1296 outFile << previousInstanceLine << std::endl;
1297 }
1298 }
1299 previousInstanceLine = line;
1300 instanceIndex++;
1301 }
1302 // Handle other lines
1303 else {
1304 if (!previousInstanceLine.empty()) {
1305 std::vector<std::string> tokens = this->splitString(previousInstanceLine);
1306 if (tokens.size() > 2 && (tokens[0][0] == 'X' || tokens[0][0] == 'x')) {
1307 std::string moduleName = tokens.back();
1308 std::string outputNet = tokens[tokens.size() - 2];
1309 bool isFinal = (outputNet == "CO" || outputNet == "S" || outputNet == "ZN");
1310 tokens[tokens.size() - 2] = outputNet + ":1";
1311
1312 for (size_t i = 0; i < tokens.size() - 1; ++i) {
1313 outFile << tokens[i] << " ";
1314 }
1315 outFile << "VDD VSS " << moduleName << std::endl;
1316
1317 this->generateRCLines(outFile, outputNet, instanceIndex - 1, isFinal);
1318 } else {
1319 logger_->warn("Previous line stored but not recognized as instance: {}",
1320 previousInstanceLine);
1321 outFile << previousInstanceLine << std::endl;
1322 }
1323 previousInstanceLine = ""; // Clear after processing
1324 }
1325 outFile << line << std::endl;
1326 }
1327 }
1328
1329 // Process the very last instance line
1330 if (!previousInstanceLine.empty()) {
1331 std::vector<std::string> tokens = this->splitString(previousInstanceLine);
1332 if (tokens.size() > 2 && (tokens[0][0] == 'X' || tokens[0][0] == 'x')) {
1333 std::string moduleName = tokens.back();
1334 std::string outputNet = tokens[tokens.size() - 2];
1335 bool isFinal = (outputNet == "CO" || outputNet == "S" || outputNet == "ZN");
1336 tokens[tokens.size() - 2] = outputNet + ":1";
1337
1338 for (size_t i = 0; i < tokens.size() - 1; ++i) {
1339 outFile << tokens[i] << " ";
1340 }
1341 outFile << "VDD VSS " << moduleName << std::endl;
1342
1343 this->generateRCLines(outFile, outputNet, instanceIndex - 1, isFinal);
1344 } else {
1345 logger_->warn("Last stored line not recognized as instance: {}", previousInstanceLine);
1346 outFile << previousInstanceLine << std::endl;
1347 }
1348 }
1349
1350 inFile.close();
1351 outFile.close();
1352
1353 // No renaming needed, we wrote directly to the final file
1354 logger_->info("SPICE netlist post-processing successful for: {}", finalSpiceFile);
1355 return true;
1356}

1386void LibFile::spice(const int chain_length, const std::vector<std::string> &cell_names,
1387 const std::string &verilog_lib_file, const std::string &spice_lib_file) {
1388 logger_->info("Creating SPICE for '{}'", filename_);
1389
1390 // 1. Generate the temporary Verilog file first
1391 this->verilog(chain_length, cell_names);
1392
1393 // Define input and output filenames
1394 std::string temp_verilog_file = basename_ + "_temp.v";
1395 std::string v2lvs_output_spice_file = basename_ + ".v2lvs.spi"; // v2lvs output
1396 std::string final_output_spice_file = basename_ + ".spi"; // Final processed output
1397
1398 // Check if temporary Verilog file exists
1399 std::ifstream check_v_in(temp_verilog_file);
1400 if (!check_v_in.is_open()) {
1401 logger_->error("Temporary Verilog file {} not found or could not be opened. Verilog generation "
1402 "might have failed.",
1403 temp_verilog_file);
1404 return;
1405 }
1406 check_v_in.close(); // Close after checking
1407
1408 // 2. Check if V2LVS tool is available
1409 logger_->debug("Checking for v2lvs tool...");
1410 std::string v2lvs_path_cmd = "which v2lvs";
1411 std::string v2lvs_path_result;
1412 std::array<char, 128> buffer;
1413 std::unique_ptr<FILE, decltype(&pclose)> pipe(popen(v2lvs_path_cmd.c_str(), "r"), pclose);
1414 if (!pipe) {
1415 logger_->error("Failed to run command to find v2lvs: {}", v2lvs_path_cmd);
1416 // Do NOT delete temp_verilog_file here if popen fails
1417 return;
1418 }
1419 while (fgets(buffer.data(), buffer.size(), pipe.get()) != nullptr) {
1420 v2lvs_path_result += buffer.data();
1421 }
1422
1423 // Trim potential newline from the result
1424 if (!v2lvs_path_result.empty() && v2lvs_path_result.back() == '\n') {
1425 v2lvs_path_result.pop_back();
1426 }
1427
1428 if (v2lvs_path_result.empty()) {
1429 logger_->error(
1430 "V2LVS tool not found in PATH. Please install Calibre or ensure v2lvs is accessible.");
1431 // Do NOT delete temp_verilog_file
1432 return;
1433 } else {
1434 logger_->info("V2LVS tool found at: {}", v2lvs_path_result);
1435 }
1436
1437 // 3. Construct and Run V2LVS command
1438 logger_->info("Running V2LVS to generate initial SPICE netlist...");
1439 std::string v2lvs_cmd_full = "v2lvs"; // Start with the command name
1440 v2lvs_cmd_full += " -v " + temp_verilog_file; // Input Verilog
