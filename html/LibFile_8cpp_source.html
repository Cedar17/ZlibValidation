123 // Create a scope for the top-level groups iteration

184void LibFile::writeToDB(const std::string &db_path, const std::string &pvt_corner,
185 double aged_year) {
186 logger_->info("Writing LUT entries to DB: '{}'", db_path);
187
188 if (!lib_json_.contains("cells") || !lib_json_["cells"].is_array()) {
189 logger_->warn("No cells found in parsed JSON, skipping DB write");
190 return;
191 }
192
193 LibDatabase database(db_path);
194 database.initialize();
195
196 int total_entries = 0;
197
198 // --- Extract header-level PVT ---
199 int process = lib_json_.value("process", 0);
200 double temperature = lib_json_.value("temperature", 0.0);
201 double voltage = lib_json_.value("voltage", 0.0);
202
203 // library_name = .lib basename (scenario_id form)
204 std::string library_name = basename_;
205
206 // Walk: cells → output_pins → timing_arcs → arc_types
207 for (const auto &cell : lib_json_["cells"]) {
208 std::string cell_name = cell["cell_name"].get<std::string>();
209
210 if (!cell.contains("output_pins") || !cell["output_pins"].is_array()) continue;
211
212 for (const auto &pin : cell["output_pins"]) {
213 std::string output_pin = pin["pin_name"].get<std::string>();
214
215 if (!pin.contains("timing_arcs") || !pin["timing_arcs"].is_array()) continue;
216
217 for (const auto &arc : pin["timing_arcs"]) {
218 std::string related_pin = arc.value("related_pin", "");
219 std::string timing_sense = arc.value("timing_sense", "");
220 std::string timing_type = arc.value("timing_type", "");
221
222 // Four arc types to check
223 static const char *ARC_NAMES[] = {"cell_rise", "cell_fall", "rise_transition",
224 "fall_transition"};
225 for (const char *arc_type : ARC_NAMES) {
226 if (!arc.contains(arc_type)) continue;
227 const auto &lut = arc[arc_type];
228 if (!lut.contains("index_1") || !lut.contains("values")) continue;
229
230 // index_1 → rows_n
231 const auto &idx1_json = lut["index_1"];
232 std::vector<double> index_1 = idx1_json.get<std::vector<double>>();
233 int rows_n = static_cast<int>(index_1.size());
234
235 // index_2 (optional, derive cols_n from values shape)
236 std::vector<double> index_2;
237 if (lut.contains("index_2")) {
238 index_2 = lut["index_2"].get<std::vector<double>>();
239 }
240
241 // values → flatten
242 const auto &vals_json = lut["values"];
243 std::vector<double> values;
244 int cols_n = 0;
245 if (vals_json.is_array() && !vals_json.empty()) {
246 cols_n = static_cast<int>(vals_json[0].size());
247 for (const auto &row : vals_json) {
248 for (const auto &v : row) {
249 values.push_back(v.get<double>());
250 }
251 }
252 }
253
254 // If index_2 is missing, default to empty vector of size cols_n
255 if (index_2.empty()) {
256 index_2.resize(cols_n, 0.0);
257 }
258
259 database.writeLutEntry(
260 filename_, // file_path (original path for provenance)
261 library_name, pvt_corner, aged_year, cell_name, output_pin,
262 related_pin, timing_sense, timing_type, arc_type, rows_n, cols_n,
263 index_1, index_2, values, process, temperature, voltage);
264
265 total_entries++;
266 }
267 }
268 }
269 }
270
271 logger_->info("Wrote {} LUT entries to DB: '{}'", total_entries, db_path);
272}

297bool LibFile::checkTimingArcMonotonicity(const json &cell, const json &pin, const json &arc,
298 const std::string &timing_arc_name, const bool is_slew) {
299 if (arc.contains("when") && !arc["when"].get<std::string>().empty()) {
300 logger_->debug(
301 "Checking cell: '{}', pin: '{}', related_pin: '{}', timing_arc: '{}', when: '{}'",
302 cell["cell_name"].get<std::string>(), pin["pin_name"].get<std::string>(),
303 arc["related_pin"].get<std::string>(), timing_arc_name, arc["when"].get<std::string>());
304 } else {
305 logger_->debug("Checking cell: '{}', pin: '{}', related_pin: '{}', timing_arc: '{}'",
306 cell["cell_name"].get<std::string>(), pin["pin_name"].get<std::string>(),
307 arc["related_pin"].get<std::string>(), timing_arc_name);
308 }
309 bool is_monotonic = true; // Assume the values are monotonic initially
310 if (arc.contains(timing_arc_name) && arc[timing_arc_name].contains("values")) {
311 std::vector<std::vector<double>> value_matrix;
312 // Generate a matrix of values from the JSON array
313 for (const auto &row_val : arc[timing_arc_name]["values"]) {
314 std::vector<double> row_data;
315 // Check if the value is an array
316 if (!row_val.is_array()) {
317 logger_->error("Invalid format: '{}' values should be an array of arrays in cell '{}', pin "
318 "'{}', related_pin '{}'",
319 timing_arc_name, cell["cell_name"].get<std::string>(),
320 pin["pin_name"].get<std::string>(), arc["related_pin"].get<std::string>());
321 return false;
322 }
323 // Check for non-numeric values
324 for (const auto &val : row_val) {
325 if (val.is_number()) {
326 row_data.push_back(val.get<double>());
327 } else {
328 logger_->warn("Non-numeric value found in '{}'.values, skipping value: {} in cell '{}', "
329 "pin '{}', related_pin '{}'",
330 timing_arc_name, val.dump(), cell["cell_name"].get<std::string>(),
331 pin["pin_name"].get<std::string>(), arc["related_pin"].get<std::string>());
332 }
333 }
334 value_matrix.push_back(row_data);
335 }
336 // Check the matrix for monotonicity
337 if (!value_matrix.empty() && !value_matrix[0].empty()) {
338 // Check the monotonic incrementality of rows (by output load capacitance, index 2)
339 for (size_t i = 0; i < value_matrix.size(); ++i) {
340 for (size_t j = 1; j < value_matrix[i].size(); ++j) {
341 if ((value_matrix[i][j] < value_matrix[i][j - 1] &&
342 pin["pin_name"] != arc["related_pin"]) ||
343 (value_matrix[i][j] == 0 && value_matrix[i][j - 1] == 0)) {
344 // If contains "when" key, log the when value
345 if (arc.contains("when") && !arc["when"].get<std::string>().empty()) {
346 logger_->warn("Non-monotonic (by load) '{}' values: ({}, {}) {} < ({}, {}) {} "
347 "for Cell: {} Pin: {}->{} when: \"{}\"",
348 timing_arc_name, i, j, value_matrix[i][j], i, j - 1,
349 value_matrix[i][j - 1], cell["cell_name"].get<std::string>(),
350 arc["related_pin"].get<std::string>(),
351 pin["pin_name"].get<std::string>(), arc["when"].get<std::string>());
352 } else {
353 logger_->warn("Non-monotonic (by load) '{}' values: ({}, {}) {} < ({}, {}) {} "
354 "for Cell: {} Pin: {}->{}",
355 timing_arc_name, i, j, value_matrix[i][j], i, j - 1,
356 value_matrix[i][j - 1], cell["cell_name"].get<std::string>(),
357 arc["related_pin"].get<std::string>(),
358 pin["pin_name"].get<std::string>());
359 }
360 is_monotonic = false;
361 }
362 }
363 }
364 if (is_slew) {
365 // Check the monotonic incrementality of columns (by input slew, index 1)
366 for (size_t j = 0; j < value_matrix[0].size(); ++j) {
367 for (size_t i = 1; i < value_matrix.size(); ++i) {
368 if ((value_matrix[i][j] < value_matrix[i - 1][j] &&
369 pin["pin_name"] != arc["related_pin"]) ||
370 (value_matrix[i][j] == 0 && value_matrix[i - 1][j] == 0)) {
371 // If contains "when" key, log the when value
372 if (arc.contains("when") && !arc["when"].get<std::string>().empty()) {
373 logger_->warn("Non-monotonic (by slew) '{}' values: ({}, {}) {} < ({}, {}) {} "
374 "for Cell: {} Pin: {}->{} when: \"{}\"",
375 timing_arc_name, i, j, value_matrix[i][j], i - 1, j,
376 value_matrix[i - 1][j], cell["cell_name"].get<std::string>(),
377 arc["related_pin"].get<std::string>(),
378 pin["pin_name"].get<std::string>(), arc["when"].get<std::string>());
379 } else {
380 logger_->warn("Non-monotonic (by slew) '{}' values: ({}, {}) {} < ({}, {}) {} "
381 "for Cell: {} Pin: {}->{}",
382 timing_arc_name, i, j, value_matrix[i][j], i - 1, j,
383 value_matrix[i - 1][j], cell["cell_name"].get<std::string>(),
384 arc["related_pin"].get<std::string>(),
385 pin["pin_name"].get<std::string>());
386 }
387 is_monotonic = false;
388 }
389 }
390 }
391 }
392 } else {
393 logger_->warn(
394 "Empty or invalid 'values' array found for cell: '{}', pin: '{}', related_pin: '{}', "
395 "timing_arc: '{}'",
396 cell["cell_name"].get<std::string>(), pin["pin_name"].get<std::string>(),
397 arc["related_pin"].get<std::string>(), timing_arc_name);
398 }
399 }
400 return is_monotonic;

423void LibFile::mono(const bool is_slew) {
424 /*
425 * Use this command to check the following data in the current library to ensure
426 * the tables are monotonically increasing with respect to output load:
427 * cell_rise retaining_rise
428 * cell_fall retaining_fall
429 * rise_transition retain_rise_slew
430 * fall_transition retain_fall_slew
431 * mpw
432 */
433 logger_->info("Monotonicity check of '{}' starting ...", filename_);
434
435 if (!std::filesystem::exists(jsonname_)) {
436 logger_->info("JSON file not found. Parsing Liberty file first.");
437 this->parse();
438 this->writeJsonToFile();
439 } else {
440 // Read the JSON file into json object
441 std::ifstream in(jsonname_);
442 if (!in.is_open()) {
443 logger_->error("Could not open file '{}' for reading", jsonname_);
444 return;
445 }
446 try {
447 lib_json_ = json::parse(in);
448 } catch (const json::parse_error &e) {
449 logger_->error("JSON parsing error in file '{}': {}", jsonname_, e.what());
450 in.close();
451 return;
452 }
453 in.close();
454 }
455
456 std::map<std::string, bool> cell_monotonicity_status; // Track pass/fail status for each cell
457 std::vector<std::string> failed_cells; // List of failed cell names
458 int total_cells = 0;
459 int passed_cells = 0;
460
461 // Check the monotonicity of delay values
462 // The index 1 (time values) must be monotonically increasing and >= 0
463 for (const auto &cell : lib_json_["cells"]) {
464 total_cells++;
465 bool cell_is_monotonic = true; // Assume cell is monotonic initially
466 std::string cell_name = cell["cell_name"].get<std::string>();
467 cell_monotonicity_status[cell_name] =
468 true; // Initialize to pass, will be set to false if any check fails
469
470 if (cell.contains("output_pins")) {
471 for (const auto &pin : cell["output_pins"]) {
472 if (pin.contains("timing_arcs")) {
473 for (const auto &arc : pin["timing_arcs"]) {
474 // Check 4 timing arc names and accumulate monotonicity status
475 for (const auto &name :
476 {"cell_rise", "cell_fall", "rise_transition", "fall_transition"}) {
477 if (!checkTimingArcMonotonicity(cell, pin, arc, name, is_slew)) {
478 cell_is_monotonic = false;
479 }
480 }
481 }
482 }
483 }
484 }
485 if (cell.contains("input_pins")) {
486 for (const auto &pin : cell["input_pins"]) {
487 if (!pin.contains("clock")) {
488 continue; // Skip non-clock pins
489 }
490 // logger_->debug("Clock pin: {}", pin["pin_name"].get<std::string>());
491 if (pin.contains("timing_arcs")) {
492 for (const auto &arc : pin["timing_arcs"]) {
493 if (arc.contains("timing_type")) {
494 // logger_->debug("Timing type: {}", arc["timing_type"].get<std::string>());
495 if (arc["timing_type"].get<std::string>() == "min_pulse_width") {
496 for (const auto &name : {"rise_constraint", "fall_constraint"}) {
497 if (!checkTimingArcMonotonicity(cell, pin, arc, name, is_slew)) {
498 cell_is_monotonic = false;
499 }
500 }
501 }
502 }
503 }
504 }
505 }
506 }
507 // Update cell status based on all checks
508 cell_monotonicity_status[cell_name] = cell_is_monotonic;
509 if (cell_is_monotonic) {
510 passed_cells++;
511 } else {
512 failed_cells.push_back(cell_name);
513 }
514 }
515 logger_->info("Monotonicity check of '{}' completed.", filename_);
516
517 // Output summary statistics
518 logger_->info("validate_monotonicity : {} out of {} cells passed", passed_cells, total_cells);
519 logger_->info("validate_monotonicity : {} out of {} cells failed", total_cells - passed_cells,
520 total_cells);
521 if (!failed_cells.empty()) {
522 std::stringstream failed_cells_ss;
523 for (size_t i = 0; i < failed_cells.size(); ++i) {
524 failed_cells_ss << failed_cells[i];
525 if (i < failed_cells.size() - 1) {
526 failed_cells_ss << ", "; // Add comma if not the last element
527 }
528 }
529 logger_->info("Failed cell list : {}", failed_cells_ss.str());
530 }

552void LibFile::supercell(const int chain_length, const std::vector<std::string> &cell_names) {
553 /*
554 * Supercells are named as follows:
555 * <cellname>__X<chain_length>__<input_pin>__<output_pin>
556 */
557 logger_->info("Creating supercells for '{}'", filename_);
558
559 if (!std::filesystem::exists(jsonname_)) {
560 logger_->info("JSON file not found. Parsing Liberty file first.");
561 this->parse();
562 this->writeJsonToFile();
563 } else {
564 // Read the JSON file into json object
565 std::ifstream in(jsonname_);
566 if (!in.is_open()) {
567 logger_->error("Could not open file '{}' for reading", jsonname_);
568 return;
569 }
570 try {
571 lib_json_ = json::parse(in);
572 } catch (const json::parse_error &e) {
573 logger_->error("JSON parsing error in file '{}': {}", jsonname_, e.what());
574 in.close();
575 return;
576 }
577 in.close();
578 }
579
580 // write to .map file
581 std::ofstream out(basename_ + ".map");
582 if (!out.is_open()) {
583 logger_->error("Could not open file '{}' for writing", basename_ + ".map");
584 return;
585 }
586 // if celll_names is empty, create supercells for all cells
587 // else create supercells for only the specified cells
588
589 // Check if we're processing specific cells or all cells
590 bool process_all_cells = cell_names.empty();
591
592 if (process_all_cells) {
593 logger_->info("Creating supercells for ALL cells in '{}'", filename_);
594 } else {
595 logger_->info("Creating supercells for {} specified cells in '{}'", cell_names.size(),
596 filename_);
597 }
598
599 // Create a set for faster lookups if we have specific cell names
600 std::unordered_set<std::string> cell_set;
601 std::unordered_set<std::string> found_cells;
602 if (!process_all_cells) {
603 cell_set.insert(cell_names.begin(), cell_names.end());
604 }
605
606 for (const auto &cell : lib_json_["cells"]) {
607 bool is_sequential = false;
608 std::string cell_name = cell["cell_name"].get<std::string>();
609
610 // Skip cells not in the specified list
611 if (!process_all_cells && cell_set.find(cell_name) == cell_set.end()) {
612 continue;
613 }
614
615 // Mark this cell as found
616 if (!process_all_cells) {
617 found_cells.insert(cell_name);
618 }
619
620 logger_->debug("Creating supercells for cell: '{}'", cell_name);
621
622 std::vector<std::string> output_pins;
623 std::vector<std::string> input_pins;
624 // Extract input and output pins
625 if (cell.contains("output_pins")) {
626 for (const auto &pin : cell["output_pins"]) {
627 output_pins.push_back(pin["pin_name"].get<std::string>());
628 }
629 }
630 if (cell.contains("input_pins")) {
631 for (const auto &pin : cell["input_pins"]) {
632 if (pin.contains("clock")) {
633 is_sequential = true;
634 // clock pin not use for supercell
635 continue;
636 }
637 input_pins.push_back(pin["pin_name"].get<std::string>());
638 }
639 }
640
641 // create supercells for all combinations of input and output pins
642 for (const auto &output_pin : output_pins) {
643 for (const auto &input_pin : input_pins) {
644 if (is_sequential) {
645 const int chain_len = 1;
646 std::string supercell_name =
647 cell_name + "__X" + std::to_string(chain_len) + "__" + input_pin + "__" + output_pin;
648 out << cell_name << " " << supercell_name << std::endl;
649 } else {
650 std::string supercell_name = cell_name + "__X" + std::to_string(chain_length) + "__" +
651 input_pin + "__" + output_pin;
652 out << cell_name << " " << supercell_name << std::endl;
653 }
654 }
655 }
656 }
657
658 // Check for cells that were specified but not found
659 if (!process_all_cells) {
660 std::vector<std::string> not_found_cells;
661 for (const auto &requested_cell : cell_names) {
662 if (found_cells.find(requested_cell) == found_cells.end()) {
663 not_found_cells.push_back(requested_cell);
664 }
665 }
666
667 // Output warning for cells that weren't found
668 if (!not_found_cells.empty()) {
669 std::string missing_cells = not_found_cells[0];
670 for (size_t i = 1; i < not_found_cells.size(); ++i) {
671 missing_cells += ", " + not_found_cells[i];
672 }
673 logger_->warn("{} specified cell{} not found in the library: {}", not_found_cells.size(),
674 not_found_cells.size() > 1 ? "s were" : " was", missing_cells);
675 }
676 }
677
678 out.close();
679 logger_->info("Supercell creation complete in '{}'", basename_ + ".map");
680}

706void LibFile::verilog(const int chain_length, const std::vector<std::string> &cell_names) {
707 logger_->info("Creating Verilog for '{}'", filename_);
708
709 this->supercell(chain_length, cell_names);
710
711 // Create a temporary file for individual modules
712 std::string temp_file = basename_ + "_temp.v";
713 std::ofstream out(temp_file);
714 if (!out.is_open()) {
715 logger_->error("Could not open temp file '{}' for writing", temp_file);
716 return;
717 }
718
719 // Read the .map file into a vector to preserve all entries and their order
720 std::vector<std::pair<std::string, std::string>> supercell_entries;
721 std::ifstream in(basename_ + ".map");
722 if (!in.is_open()) {
723 logger_->error("Could not open file '{}' for reading", basename_ + ".map");
724 return;
725 }
726
727 std::string line;
728 while (std::getline(in, line)) {
729 std::istringstream iss(line);
730 std::string cell_name, supercell_name;
731 if (!(iss >> cell_name >> supercell_name)) {
732 logger_->warn("Invalid line format in map file: '{}'", line);
733 continue;
734 }
735 supercell_entries.push_back({cell_name, supercell_name});
736 }
737 in.close();
738 logger_->info("Read {} supercells from '{}'", supercell_entries.size(), basename_ + ".map");
739
740 // Store module information for top-level creation
741 std::vector<std::string> module_names;
742 std::map<std::string, std::vector<std::string>> module_inputs;
743 std::map<std::string, std::vector<std::string>> module_outputs;
744
745 // Generate verilog module for each supercell
746 for (const auto &supercell_entry : supercell_entries) {
747 // Check if the cell is sequential
748 bool is_sequential = false;
749 // Count the number of instances will be created in verilog
750 int instance_count = 0;
751 // Get original cell name from pair
752 const std::string &cell_name = supercell_entry.first;
753 // Get supercell name(as well as module name) from pair
754 const std::string &module_name = supercell_entry.second;
755
756 // Store module name for top-level creation
757 module_names.push_back(module_name);
758
759 logger_->info("Creating Module: '{}' from Cell '{}", module_name, cell_name);
760
761 // Get the cell JSON object
762 json cell_json;
763 for (const auto &cell : lib_json_["cells"]) {
764 if (cell["cell_name"].get<std::string>() == cell_name) {
765 cell_json = cell;
766 break;
767 }
768 }
769
770 // Get input/output pins from the cell JSON object
771 std::vector<std::string> input_pins;
772 std::vector<std::string> output_pins;
773 std::stringstream input_pins_ss;
774 std::stringstream output_pins_ss;
775 if (cell_json.contains("input_pins")) {
776 for (const auto &pin : cell_json["input_pins"]) {
777 if (pin.contains("clock")) {
778 is_sequential = true;
779 }
780 std::string pin_name = pin["pin_name"].get<std::string>();
781 input_pins.push_back(pin_name);
782 input_pins_ss << pin_name << ", ";
783
784 // Store input pin name for top-level creation
785 module_inputs[module_name].push_back(pin_name);
786 }
787 }
788 if (cell_json.contains("output_pins")) {
789 for (const auto &pin : cell_json["output_pins"]) {
790 std::string pin_name = pin["pin_name"].get<std::string>();
791 output_pins.push_back(pin_name);
792 output_pins_ss << pin_name << ", ";
793
794 // Store output pin name for top-level creation
795 module_outputs[module_name].push_back(pin_name);
796 }
797 }
798 logger_->debug("Input pins set: {}", input_pins_ss.str());
799 logger_->debug("Output pins set: {}", output_pins_ss.str());
800
801 // Sequential cells will have only one instance
802 // Combinational cells will have instances equal to chain length
803 if (is_sequential) {
804 instance_count = 1;
805 } else {
806 instance_count = chain_length;
807 }
808
809 // Create ANSI port list
810 std::string port_list_text = "(";
811 // Add input ports
812 bool isFirstPort = true;
813 for (const auto &input_pin : input_pins) {
814 if (!isFirstPort) {
815 port_list_text += ", ";
816 }
817 port_list_text += "input " + input_pin;
818 isFirstPort = false;
819 }
820 // Add output ports
821 for (const auto &output_pin : output_pins) {
822 if (!isFirstPort) {
823 port_list_text += ", ";
824 }
825 port_list_text += "output " + output_pin;
826 isFirstPort = false;
827 }
828 port_list_text += ")";
829 logger_->debug("ANSI Port list: {}", port_list_text);
830
831 // Creat full module header text
832 std::string fullModuleText = "module " + module_name + port_list_text + ";\nendmodule";
833 logger_->debug("Full module text: {}", fullModuleText);
834
835 // Generate the syntax tree for the module using slang library
836 auto tree = slang::syntax::SyntaxTree::fromText(fullModuleText);
837 if (tree) {
838 // Add ports to the syntax tree
839 ModuleRewriter rewriter(input_pins, output_pins, supercell_entry, instance_count, logger_);
840
841 tree = rewriter.transform(tree);
842 // Revisit the syntax tree to check architecture
843 // rewriter.visit(tree->root());
844
845 // Output the transformed syntax tree
846 out << "`timescale 1ns/10ps" << std::endl;
847 out << slang::syntax::SyntaxPrinter::printFile(*tree) << std::endl << std::endl;
848 } else {
849 logger_->error("Failed to create syntax tree for module '{}'", module_name);
850 continue;
851 }
852 }
853
854 out.close();
855
856 // Create the top-level module
857 std::ofstream final_out(basename_ + ".v");
858 if (!final_out.is_open()) {
859 logger_->error("Could not open file '{}' for writing", basename_ + ".v");
860 return;
861 }
862
863 // Collect all port names separately for inputs and outputs
864 std::vector<std::string> all_input_ports;
865 std::vector<std::string> all_output_ports;
866 std::stringstream top_instances;
867
868 // First pass: collect all port names
869 for (const auto &module_name : module_names) {
870 // Collect input port names
871 for (const auto &pin : module_inputs[module_name]) {
872 all_input_ports.push_back(module_name + "__" + pin);
873 }
874
875 // Collect output port names
876 for (const auto &pin : module_outputs[module_name]) {
877 all_output_ports.push_back(module_name + "__" + pin);
878 }
879
880 // Create instance
881 std::string instance_name = "I_" + module_name.substr(0, module_name.find("__X"));
882 for (size_t i = module_name.find("__X") + 3; i < module_name.length(); i++) {
883 if (module_name[i] == '_' && module_name[i + 1] == '_') {
884 instance_name += "__" + module_name.substr(i + 2);
885 break;
886 }
887 }
888
889 // Generate port connections for this instance
890 std::stringstream instance_ports;
891 for (const auto &pin : module_inputs[module_name]) {
892 instance_ports << "." << pin << "(" << module_name << "__" << pin << "), ";
893 }
894 for (const auto &pin : module_outputs[module_name]) {
895 instance_ports << "." << pin << "(" << module_name << "__" << pin << "), ";
896 }
897
898 // Remove last comma and space
899 std::string instance_ports_str = instance_ports.str();
900 if (!instance_ports_str.empty()) {
901 instance_ports_str = instance_ports_str.substr(0, instance_ports_str.length() - 2);
902 }
903
904 top_instances << " " << module_name << " " << instance_name << " (" << instance_ports_str
905 << ");\n";
906 }
907
908 // Now generate the port list with all inputs first, then all outputs
909 std::stringstream top_ports;
910
911 // Add all input ports first
912 for (size_t i = 0; i < all_input_ports.size(); ++i) {
913 top_ports << all_input_ports[i];
914 if (i < all_input_ports.size() - 1 || !all_output_ports.empty()) {
915 top_ports << ", ";
916 }
917 }
918
919 // Then add all output ports
920 for (size_t i = 0; i < all_output_ports.size(); ++i) {
921 top_ports << all_output_ports[i];
922 if (i < all_output_ports.size() - 1) {
923 top_ports << ", ";
924 }
925 }
926
927 // Generate input and output declarations
928 std::stringstream top_inputs;
929 std::stringstream top_outputs;
930
931 // Generate input declarations
932 for (const auto &port : all_input_ports) {
933 top_inputs << " input " << port << ";\n";
934 }
935
936 // Generate output declarations
937 for (const auto &port : all_output_ports) {
938 top_outputs << " output " << port << ";\n";
939 }
940
941 // Write the top module
942 final_out << "`timescale 1ns/10ps" << std::endl;
943 final_out << "module validate_top ( " << top_ports.str() << " );" << std::endl;
944 final_out << std::endl; // Add newline before port declarations
945 final_out << top_inputs.str();
946 final_out << top_outputs.str();
947 final_out << std::endl;
948 final_out << top_instances.str();
949 final_out << "endmodule" << std::endl << std::endl;
950
951 // Append the module definitions from the temporary file
952 std::ifstream temp_in(temp_file);
953 if (temp_in.is_open()) {
954 final_out << temp_in.rdbuf();
955 temp_in.close();
956 } else {
957 logger_->error("Could not open temp file '{}' for reading", temp_file);
958 }
959
960 final_out.close();
961
962 // Remove temporary file
963 // if (std::filesystem::exists(temp_file)) {
964 // std::filesystem::remove(temp_file);
965 // }
966
967 logger_->info("Verilog creation complete in '{}'", basename_ + ".v");
968}

1023void LibFile::generateRCLines(std::ofstream &outFile, const std::string &netName, int instanceIndex,
1024 bool isFinalStage) {
1025 // --- Default RC Values (as observed in the target SPICE) ---
1026 const double R1_INTERMEDIATE = 0.01;
1027 const double C1_INTERMEDIATE = 5.3e-16;
1028 const double R2_INTERMEDIATE = 0.01;
1029 const double R1_FINAL = 1e-2;
1030 const double C1_FINAL = 1e-18;
1031 const std::string CAP_GROUND = "COREGND1"; // Assumed ground for capacitors
1032
1033 std::string r1Name = "R1_" + std::to_string(instanceIndex);
1034 std::string c1Name = "C1_" + std::to_string(instanceIndex);
1035 std::string r2Name = "R2_" + std::to_string(instanceIndex);
1036 std::string node1 = netName + ":1"; // Intermediate node 1
1037 std::string node2 = netName + ":2"; // Intermediate node 2
1038
1039 // Set precision for scientific notation output
1040 outFile << std::scientific << std::setprecision(1);
1041
1042 if (!isFinalStage) {
1043 // Intermediate RCR structure (R1-C1-R2)
1044 outFile << r1Name << " " << node1 << " " << node2 << " " << R1_INTERMEDIATE << std::endl;
1045 outFile << c1Name << " " << node2 << " " << CAP_GROUND << " " << C1_INTERMEDIATE << std::endl;
1046 outFile << r2Name << " " << node2 << " " << netName << " " << R2_INTERMEDIATE << std::endl;
1047 } else {
1048 // Final RC structure (R1-C1 connected to the instance output node :1)
1049 // Note: The target SPICE connects R1 between node1 and the final port (netName).
1050 outFile << r1Name << " " << node1 << " " << netName << " " << R1_FINAL << std::endl;
1051 outFile << c1Name << " " << node1 << " " << CAP_GROUND << " " << C1_FINAL << std::endl;
1052 }
1053 // Reset precision/format to default
1054 outFile << std::defaultfloat << std::setprecision(6);
1055}

1105bool LibFile::modifySpiceNetlist(const std::string &v2lvsSpiceFile, // Input is the v2lvs output
1106 const std::string &finalSpiceFile, // Output is the final file
1107 const std::string &targetGlobalLine) {
1108 logger_->info("Post-processing SPICE netlist: {} -> {}", v2lvsSpiceFile, finalSpiceFile);
1109
1110 std::ifstream inFile(v2lvsSpiceFile);
1111 // Write directly to the final output file, overwriting if it exists
1112 std::ofstream outFile(finalSpiceFile, std::ios::trunc);
1113
1114 if (!inFile) {
1115 logger_->error("Could not open input v2lvs SPICE file for modification: {}", v2lvsSpiceFile);
1116 return false;
1117 }
1118 if (!outFile) {
1119 logger_->error("Could not open final output SPICE file for writing: {}", finalSpiceFile);
1120 inFile.close(); // Close input file if output fails
1121 return false;
1122 }
1123
1124 // --- Add Metadata Comments ---
1125 outFile << "** Generated by " << APP_NAME << " v" << APP_VERSION << " from " << APP_AUTHOR;
1126 auto now = std::chrono::system_clock::now();
1127 std::time_t now_time_t = std::chrono::system_clock::to_time_t(now);
1128 // Use localtime_s on Windows, localtime_r on POSIX, or std::localtime (less safe)
1129 // std::tm now_tm;
1130 // localtime_s(&now_tm, &now_time_t); // Example for Windows
1131 std::tm *now_tm_ptr = std::localtime(&now_time_t); // Standard C++, potentially less thread-safe
1132 if (now_tm_ptr) {
1133 outFile << ". On: " << std::put_time(now_tm_ptr, "%c %Z") << " **\n" << std::endl;
1134 } else {
1135 outFile << ". Timestamp unavailable **\n" << std::endl;
1136 }
1137 // --- End Metadata ---
1138
1139 std::string line;
1140 bool includeFound = false;
1141 bool globalInserted = false;
1142 bool inSubckt = false;
1143 std::string previousInstanceLine = "";
1144 int instanceIndex = 0; // Counter for R/C naming within a subckt
1145
1146 while (std::getline(inFile, line)) {
1147 // Trim leading/trailing whitespace
1148 line.erase(0, line.find_first_not_of(" \t\n\r\f\v"));
1149 line.erase(line.find_last_not_of(" \t\n\r\f\v") + 1);
1150
1151 // Skip empty lines and v2lvs header comments
1152 if (line.empty() || line.find("$ Spice netlist generated by v2lvs") == 0 ||
1153 line.find("$ v2") == 0 || // Catch version line too
1154 line.find("*.BUSDELIMITER") == 0) {
1155 continue; // Skip these lines entirely
1156 }
1157
1158 // Write other comments directly after processing any pending instance
1159 if (line[0] == '*') {
1160 // Process previous instance before writing comment if needed
1161 if (!previousInstanceLine.empty()) {
1162 std::vector<std::string> tokens = this->splitString(previousInstanceLine);
1163 if (tokens.size() > 2 && (tokens[0][0] == 'X' || tokens[0][0] == 'x')) {
1164 std::string moduleName = tokens.back();
1165 std::string outputNet = tokens[tokens.size() - 2]; // Assumption
1166 bool isFinal = (outputNet == "CO" || outputNet == "S" || outputNet == "ZN");
1167 tokens[tokens.size() - 2] = outputNet + ":1"; // Modify output pin
1168
1169 // Reconstruct line with correct VDD/VSS order
1170 for (size_t i = 0; i < tokens.size() - 1; ++i) { // Up to module name
1171 outFile << tokens[i] << " ";
1172 }
1173 outFile << "VDD VSS " << moduleName << std::endl; // Insert VDD VSS before module name
1174
1175 this->generateRCLines(outFile, outputNet, instanceIndex - 1, isFinal);
1176 } else {
1177 logger_->warn("Previous line stored but not recognized as instance: {}",
1178 previousInstanceLine);
1179 outFile << previousInstanceLine << std::endl;
1180 }
1181 previousInstanceLine = ""; // Clear after processing
1182 }
1183 outFile << line << std::endl; // Write the comment line
1184 continue;
1185 }
1186
1187 // Handle .INCLUDE
1188 if (line.find(".INCLUDE") == 0) {
1189 outFile << line << std::endl;
1190 includeFound = true;
1191 continue; // Process next line to insert global
1192 }
1193
1194 // Insert the target global line after .INCLUDE
1195 if (includeFound && !globalInserted) {
1196 outFile << targetGlobalLine << std::endl << std::endl; // Add extra newline for spacing
1197 globalInserted = true;
1198 // Fall through to process the current line
1199 }
1200
1201 // Handle .SUBCKT
1202 if (line.find(".SUBCKT") == 0 || line.find(".subckt") == 0) { // Handle case-insensitivity
1203 if (!previousInstanceLine.empty()) {
1204 std::vector<std::string> tokens = this->splitString(previousInstanceLine);
1205 if (tokens.size() > 2 && (tokens[0][0] == 'X' || tokens[0][0] == 'x')) {
1206 std::string moduleName = tokens.back();
1207 std::string outputNet = tokens[tokens.size() - 2];
1208 bool isFinal = (outputNet == "CO" || outputNet == "S" || outputNet == "ZN");
1209 tokens[tokens.size() - 2] = outputNet + ":1";
1210
1211 for (size_t i = 0; i < tokens.size() - 1; ++i) {
1212 outFile << tokens[i] << " ";
1213 }
1214 outFile << "VDD VSS " << moduleName << std::endl;
1215
1216 this->generateRCLines(outFile, outputNet, instanceIndex - 1, isFinal);
1217 } else {
1218 logger_->warn("Previous line stored but not recognized as instance: {}",
1219 previousInstanceLine);
1220 outFile << previousInstanceLine << std::endl;
1221 }
1222 previousInstanceLine = "";
1223 }
1224
1225 inSubckt = true;
1226 instanceIndex = 0; // Reset R/C counter for new subcircuit
1227 outFile << line << " VDD VSS" << std::endl; // Add VDD VSS to ports
1228 }
1229 // Handle .ENDS
1230 else if (line.find(".ENDS") == 0 || line.find(".ends") == 0) { // Handle case-insensitivity
1231 inSubckt = false;
1232 // Process the last instance line *before* writing .ENDS
1233 if (!previousInstanceLine.empty()) {
1234 std::vector<std::string> tokens = this->splitString(previousInstanceLine);
1235 if (tokens.size() > 2 && (tokens[0][0] == 'X' || tokens[0][0] == 'x')) {
1236 std::string moduleName = tokens.back();
1237 std::string outputNet = tokens[tokens.size() - 2];
1238 bool isFinal = (outputNet == "CO" || outputNet == "S" || outputNet == "ZN");
1239 tokens[tokens.size() - 2] = outputNet + ":1";
1240
1241 for (size_t i = 0; i < tokens.size() - 1; ++i) {
1242 outFile << tokens[i] << " ";
1243 }
1244 outFile << "VDD VSS " << moduleName << std::endl;
1245
1246 this->generateRCLines(outFile, outputNet, instanceIndex - 1, isFinal);
1247 } else {
1248 logger_->warn("Previous line stored but not recognized as instance: {}",
1249 previousInstanceLine);
1250 outFile << previousInstanceLine << std::endl;
1251 }
1252 previousInstanceLine = ""; // Clear after processing
1253 }
1254 outFile << line << std::endl << std::endl; // Add extra newline after .ENDS
1255 }
1256 // Skip original .GLOBAL lines
1257 else if (line.find(".GLOBAL") == 0 || line.find(".global") == 0) {
1258 continue;
1259 }
1260 // Handle Instance lines
1261 else if (inSubckt && (line[0] == 'X' || line[0] == 'x')) {
1262 if (!previousInstanceLine.empty()) {
1263 std::vector<std::string> tokens = this->splitString(previousInstanceLine);
1264 if (tokens.size() > 2 && (tokens[0][0] == 'X' || tokens[0][0] == 'x')) {
1265 std::string moduleName = tokens.back();
1266 std::string outputNet = tokens[tokens.size() - 2]; // Assumption
1267 bool isFinal = (outputNet == "CO" || outputNet == "S" || outputNet == "ZN");
1268 tokens[tokens.size() - 2] = outputNet + ":1"; // Modify output pin
1269
1270 // Reconstruct line with correct VDD/VSS order
1271 for (size_t i = 0; i < tokens.size() - 1; ++i) { // Up to module name
1272 outFile << tokens[i] << " ";
1273 }
1274 outFile << "VDD VSS " << moduleName << std::endl; // Insert VDD VSS before module name
1275
1276 this->generateRCLines(outFile, outputNet, instanceIndex - 1,
1277 isFinal); // Use previous index
1278 } else {
1279 logger_->warn("Previous line stored but not recognized as instance: {}",
1280 previousInstanceLine);
1281 outFile << previousInstanceLine << std::endl;
1282 }
1283 }
1284 previousInstanceLine = line;
1285 instanceIndex++;
1286 }
1287 // Handle other lines
1288 else {
1289 if (!previousInstanceLine.empty()) {
1290 std::vector<std::string> tokens = this->splitString(previousInstanceLine);
1291 if (tokens.size() > 2 && (tokens[0][0] == 'X' || tokens[0][0] == 'x')) {
1292 std::string moduleName = tokens.back();
1293 std::string outputNet = tokens[tokens.size() - 2];
1294 bool isFinal = (outputNet == "CO" || outputNet == "S" || outputNet == "ZN");
1295 tokens[tokens.size() - 2] = outputNet + ":1";
1296
1297 for (size_t i = 0; i < tokens.size() - 1; ++i) {
1298 outFile << tokens[i] << " ";
1299 }
1300 outFile << "VDD VSS " << moduleName << std::endl;
1301
1302 this->generateRCLines(outFile, outputNet, instanceIndex - 1, isFinal);
1303 } else {
1304 logger_->warn("Previous line stored but not recognized as instance: {}",
1305 previousInstanceLine);
1306 outFile << previousInstanceLine << std::endl;
1307 }
1308 previousInstanceLine = ""; // Clear after processing
1309 }
1310 outFile << line << std::endl;
1311 }
1312 }
1313
1314 // Process the very last instance line
1315 if (!previousInstanceLine.empty()) {
1316 std::vector<std::string> tokens = this->splitString(previousInstanceLine);
1317 if (tokens.size() > 2 && (tokens[0][0] == 'X' || tokens[0][0] == 'x')) {
1318 std::string moduleName = tokens.back();
1319 std::string outputNet = tokens[tokens.size() - 2];
1320 bool isFinal = (outputNet == "CO" || outputNet == "S" || outputNet == "ZN");
1321 tokens[tokens.size() - 2] = outputNet + ":1";
1322
1323 for (size_t i = 0; i < tokens.size() - 1; ++i) {
1324 outFile << tokens[i] << " ";
1325 }
1326 outFile << "VDD VSS " << moduleName << std::endl;
1327
1328 this->generateRCLines(outFile, outputNet, instanceIndex - 1, isFinal);
1329 } else {
1330 logger_->warn("Last stored line not recognized as instance: {}", previousInstanceLine);
1331 outFile << previousInstanceLine << std::endl;
1332 }
1333 }
1334
1335 inFile.close();
1336 outFile.close();
1337
1338 // No renaming needed, we wrote directly to the final file
1339 logger_->info("SPICE netlist post-processing successful for: {}", finalSpiceFile);
1340 return true;
1341}

1371void LibFile::spice(const int chain_length, const std::vector<std::string> &cell_names,
1372 const std::string &verilog_lib_file, const std::string &spice_lib_file) {
1373 logger_->info("Creating SPICE for '{}'", filename_);
1374
1375 // 1. Generate the temporary Verilog file first
1376 this->verilog(chain_length, cell_names);
1377
1378 // Define input and output filenames
1379 std::string temp_verilog_file = basename_ + "_temp.v";
1380 std::string v2lvs_output_spice_file = basename_ + ".v2lvs.spi"; // v2lvs output
1381 std::string final_output_spice_file = basename_ + ".spi"; // Final processed output
1382
1383 // Check if temporary Verilog file exists
1384 std::ifstream check_v_in(temp_verilog_file);
1385 if (!check_v_in.is_open()) {
1386 logger_->error("Temporary Verilog file {} not found or could not be opened. Verilog generation "
1387 "might have failed.",
1388 temp_verilog_file);
1389 return;
1390 }
1391 check_v_in.close(); // Close after checking
1392
1393 // 2. Check if V2LVS tool is available
1394 logger_->debug("Checking for v2lvs tool...");
1395 std::string v2lvs_path_cmd = "which v2lvs";
1396 std::string v2lvs_path_result;
1397 std::array<char, 128> buffer;
1398 std::unique_ptr<FILE, decltype(&pclose)> pipe(popen(v2lvs_path_cmd.c_str(), "r"), pclose);
1399 if (!pipe) {
1400 logger_->error("Failed to run command to find v2lvs: {}", v2lvs_path_cmd);
1401 // Do NOT delete temp_verilog_file here if popen fails
1402 return;
1403 }
1404 while (fgets(buffer.data(), buffer.size(), pipe.get()) != nullptr) {
1405 v2lvs_path_result += buffer.data();
1406 }
1407
1408 // Trim potential newline from the result
1409 if (!v2lvs_path_result.empty() && v2lvs_path_result.back() == '\n') {
1410 v2lvs_path_result.pop_back();
1411 }
1412
1413 if (v2lvs_path_result.empty()) {
1414 logger_->error(
1415 "V2LVS tool not found in PATH. Please install Calibre or ensure v2lvs is accessible.");
1416 // Do NOT delete temp_verilog_file
1417 return;
1418 } else {
1419 logger_->info("V2LVS tool found at: {}", v2lvs_path_result);
1420 }
1421
1422 // 3. Construct and Run V2LVS command
1423 logger_->info("Running V2LVS to generate initial SPICE netlist...");
1424 std::string v2lvs_cmd_full = "v2lvs"; // Start with the command name
1425 v2lvs_cmd_full += " -v " + temp_verilog_file; // Input Verilog
1426 v2lvs_cmd_full += " -o " + v2lvs_output_spice_file; // Output SPICE
1427 v2lvs_cmd_full += " -s0 VSS"; // Default ground net, generates .GLOBAL VSS
1428 v2lvs_cmd_full += " -s1 VDD"; // Default power net, generates .GLOBAL VDD
1429 v2lvs_cmd_full += " -i"; // Use positional pins (traditional SPICE)
1430 v2lvs_cmd_full += " -l " + verilog_lib_file; // Verilog library for pin order
1431 v2lvs_cmd_full += " -s " + spice_lib_file; // SPICE library to include
1432
1433 logger_->debug("Executing V2LVS command: {}", v2lvs_cmd_full);
1434 int ret = system(v2lvs_cmd_full.c_str());
1435
1436 // Do NOT delete temp_verilog_file
1437
1438 if (ret != 0) {
