25std::vector<double> parseStringToVector(const std::string &str) {
26 std::vector<double> result;
27 std::string cleaned_str;
28
29 // Remove backslashes and newline characters
30 for (char c : str) {
31 if (c != '\\' && c != '\n') {
32 cleaned_str += c;
33 }
34 }
35
36 std::stringstream ss(cleaned_str);
37 std::string item;
38 while (std::getline(ss, item, ',')) {
39 result.push_back(std::stod(item));
40 }
41 return result;
42}

60json generateLutJson(LibGroup &lib_lut_group, si2drErrorT &err) {

63 AttributesIterator attr_iter(lib_lut_group.getAttrs(), err);
64 for (; !attr_iter.end(); attr_iter.next()) {

76 lut_json[lut_attr_name] = parseStringToVector(std::string(values_iter.str_));
77 }
78 } else if (lut_attr_name == "values") {
79 ValuesIterator values_iter(lib_attr.getValues(), err);
80 for (; !values_iter.end(); values_iter.next()) {
81 // spdlog::debug("{}", values_iter.str_);
82 lut_json["values"].push_back(parseStringToVector(std::string(values_iter.str_)));
83 }
84 } else {
85 spdlog::warn("Unknown LUT attribute name: {}", lut_attr_name);
86 }
87 }
