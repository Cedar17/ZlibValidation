476template<typename T>
477SeparatedSyntaxList<T>* deepClone(const SeparatedSyntaxList<T>& node, BumpAllocator& alloc) {
478 SmallVector<TokenOrSyntax> buffer(node.size(), UninitializedTag());
479 for (const auto& ele : node.elems()) {
480 if (ele.isToken())
481 buffer.push_back(ele.token().deepClone(alloc));
482 else
483 buffer.push_back(static_cast<T*>(deepClone(*ele.node(), alloc)));
484 }
485 return alloc.emplace<SeparatedSyntaxList<T>>(buffer.copy(alloc));
486}
