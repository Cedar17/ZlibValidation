612struct liberty_value_data
613{
614 int dimensions;
615 int *dim_sizes; /* a malloc'd array of <dimensions> numbers,
616 each number is the size of the array in that dim.*/
617 LONG_DOUBLE **index_info; /* a Ptr to a malloc'd array of size <dimensions>, ptrs to
618 malloc'd arrays of long double index values.
619 Each array of long double is of length set by
620 the corresponding dim_sizes array values */
621 LONG_DOUBLE *values; /* a ptr to a malloc'd array of long doubles,
622 starting with [0,0,...,0,0], and ending with
623 [z-1,y-1,....b-1,a-1], where a-z are the max
624 number of elements in each dimension. */
625};
