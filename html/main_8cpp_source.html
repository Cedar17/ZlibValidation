216void compareLibFiles(const std::string &ref_lib, const std::string &comp_lib, const double reltol,
217 const double abstol, std::string &report_file_name) {
218 std::string ref_logname = std::filesystem::path(ref_lib).stem().string() + ".cmp.log";
219 std::string comp_logname = std::filesystem::path(comp_lib).stem().string() + ".cmp.log";
220 LibFile ref_libfile(ref_lib, ref_logname), comp_libfile(comp_lib, comp_logname);
221
222 // Check relative tolerance validity
223 if (reltol < 0.0) {
224 spdlog::error("Invalid relative tolerance: {}. Relative tolerance must be >= 0.0.", reltol);
225 return;
226 }
227 if (report_file_name.empty()) {
228 report_file_name = comp_libfile.basename_ + ".cmp.md";
229 } else if (std::filesystem::path(report_file_name).extension() != ".txt" &&
230 std::filesystem::path(report_file_name).extension() != ".md") {
231 // report file name not end in .txt or .md, warn user and append .md
232 report_file_name = report_file_name + ".md";
233 spdlog::warn("Report file name does not end in .txt or .md. Report will be written to '{}'",
234 report_file_name);
235 }
236
237 LibraryComparator comparator(ref_libfile, comp_libfile, reltol, abstol);
238 comparator.generateReport(report_file_name);
239 spdlog::info("Comparison completed. Report written to: '{}'", report_file_name);
240}
