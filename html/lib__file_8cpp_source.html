218bool LibFile::checkTimingArcMonotonicity(const json &cell, const json &pin, const json &arc,
219 const std::string &timing_arc_name, const bool is_slew) {
220 if (arc.contains("when") && !arc["when"].get<std::string>().empty()) {
221 logger_->debug(
222 "Checking cell: '{}', pin: '{}', related_pin: '{}', timing_arc: '{}', when: '{}'",
223 cell["cell_name"].get<std::string>(), pin["pin_name"].get<std::string>(),
224 arc["related_pin"].get<std::string>(), timing_arc_name, arc["when"].get<std::string>());
225 } else {
226 logger_->debug("Checking cell: '{}', pin: '{}', related_pin: '{}', timing_arc: '{}'",
227 cell["cell_name"].get<std::string>(), pin["pin_name"].get<std::string>(),
228 arc["related_pin"].get<std::string>(), timing_arc_name);
229 }
230 bool is_monotonic = true; // Assume the values are monotonic initially
231 if (arc.contains(timing_arc_name) && arc[timing_arc_name].contains("values")) {
232 std::vector<std::vector<double>> value_matrix;
233 // Generate a matrix of values from the JSON array
234 for (const auto &row_val : arc[timing_arc_name]["values"]) {
235 std::vector<double> row_data;
236 // Check if the value is an array
237 if (!row_val.is_array()) {
238 logger_->error("Invalid format: '{}' values should be an array of arrays in cell '{}', pin "
239 "'{}', related_pin '{}'",
240 timing_arc_name, cell["cell_name"].get<std::string>(),
241 pin["pin_name"].get<std::string>(), arc["related_pin"].get<std::string>());
242 return false;
243 }
244 // Check for non-numeric values
245 for (const auto &val : row_val) {
246 if (val.is_number()) {
247 row_data.push_back(val.get<double>());
248 } else {
249 logger_->warn("Non-numeric value found in '{}'.values, skipping value: {} in cell '{}', "
250 "pin '{}', related_pin '{}'",
251 timing_arc_name, val.dump(), cell["cell_name"].get<std::string>(),
252 pin["pin_name"].get<std::string>(), arc["related_pin"].get<std::string>());
253 }
254 }
255 value_matrix.push_back(row_data);
256 }
257 // Check the matrix for monotonicity
258 if (!value_matrix.empty() && !value_matrix[0].empty()) {
259 // Check the monotonic incrementality of rows (by output load capacitance, index 2)
260 for (size_t i = 0; i < value_matrix.size(); ++i) {
261 for (size_t j = 1; j < value_matrix[i].size(); ++j) {
262 if ((value_matrix[i][j] < value_matrix[i][j - 1] &&
263 pin["pin_name"] != arc["related_pin"]) ||
264 (value_matrix[i][j] == 0 && value_matrix[i][j - 1] == 0)) {
265 // If contains "when" key, log the when value
266 if (arc.contains("when") && !arc["when"].get<std::string>().empty()) {
267 logger_->warn("Non-monotonic (by load) '{}' values: ({}, {}) {} < ({}, {}) {} "
268 "for Cell: {} Pin: {}->{} when: \"{}\"",
269 timing_arc_name, i, j, value_matrix[i][j], i, j - 1,
270 value_matrix[i][j - 1], cell["cell_name"].get<std::string>(),
271 arc["related_pin"].get<std::string>(),
272 pin["pin_name"].get<std::string>(), arc["when"].get<std::string>());
273 } else {
274 logger_->warn("Non-monotonic (by load) '{}' values: ({}, {}) {} < ({}, {}) {} "
275 "for Cell: {} Pin: {}->{}",
276 timing_arc_name, i, j, value_matrix[i][j], i, j - 1,
277 value_matrix[i][j - 1], cell["cell_name"].get<std::string>(),
278 arc["related_pin"].get<std::string>(),
279 pin["pin_name"].get<std::string>());
280 }
281 is_monotonic = false;
282 }
283 }
284 }
285 if (is_slew) {
286 // Check the monotonic incrementality of columns (by input slew, index 1)
287 for (size_t j = 0; j < value_matrix[0].size(); ++j) {
288 for (size_t i = 1; i < value_matrix.size(); ++i) {
289 if ((value_matrix[i][j] < value_matrix[i - 1][j] &&
290 pin["pin_name"] != arc["related_pin"]) ||
291 (value_matrix[i][j] == 0 && value_matrix[i - 1][j] == 0)) {
292 // If contains "when" key, log the when value
293 if (arc.contains("when") && !arc["when"].get<std::string>().empty()) {
294 logger_->warn("Non-monotonic (by slew) '{}' values: ({}, {}) {} < ({}, {}) {} "
295 "for Cell: {} Pin: {}->{} when: \"{}\"",
296 timing_arc_name, i, j, value_matrix[i][j], i - 1, j,
297 value_matrix[i - 1][j], cell["cell_name"].get<std::string>(),
298 arc["related_pin"].get<std::string>(),
299 pin["pin_name"].get<std::string>(), arc["when"].get<std::string>());
300 } else {
301 logger_->warn("Non-monotonic (by slew) '{}' values: ({}, {}) {} < ({}, {}) {} "
302 "for Cell: {} Pin: {}->{}",
303 timing_arc_name, i, j, value_matrix[i][j], i - 1, j,
304 value_matrix[i - 1][j], cell["cell_name"].get<std::string>(),
305 arc["related_pin"].get<std::string>(),
306 pin["pin_name"].get<std::string>());
307 }
308 is_monotonic = false;
309 }
310 }
311 }
312 }
313 } else {
314 logger_->warn(
315 "Empty or invalid 'values' array found for cell: '{}', pin: '{}', related_pin: '{}', "
316 "timing_arc: '{}'",
317 cell["cell_name"].get<std::string>(), pin["pin_name"].get<std::string>(),
318 arc["related_pin"].get<std::string>(), timing_arc_name);
319 }
320 }
321 return is_monotonic;
322}

344void LibFile::mono(const bool is_slew) {
345 /*
346 * Use this command to check the following data in the current library to ensure
347 * the tables are monotonically increasing with respect to output load:
348 * cell_rise retaining_rise
349 * cell_fall retaining_fall
350 * rise_transition retain_rise_slew
351 * fall_transition retain_fall_slew
352 * mpw
353 */
354 logger_->info("Monotonicity check of '{}' starting ...", filename_);
355
356 if (!std::filesystem::exists(jsonname_)) {
357 logger_->info("JSON file not found. Parsing Liberty file first.");
358 this->parse();
359 this->writeJsonToFile();
360 } else {
361 // Read the JSON file into json object
362 std::ifstream in(jsonname_);
363 if (!in.is_open()) {
364 logger_->error("Could not open file '{}' for reading", jsonname_);
365 return;
366 }
367 try {
368 lib_json_ = json::parse(in);
369 } catch (const json::parse_error &e) {
370 logger_->error("JSON parsing error in file '{}': {}", jsonname_, e.what());
371 in.close();
372 return;
373 }
374 in.close();
375 }
376
377 std::map<std::string, bool> cell_monotonicity_status; // Track pass/fail status for each cell
378 std::vector<std::string> failed_cells; // List of failed cell names
379 int total_cells = 0;
380 int passed_cells = 0;
381
382 // Check the monotonicity of delay values
383 // The index 1 (time values) must be monotonically increasing and >= 0
384 for (const auto &cell : lib_json_["cells"]) {
385 total_cells++;
386 bool cell_is_monotonic = true; // Assume cell is monotonic initially
387 std::string cell_name = cell["cell_name"].get<std::string>();
388 cell_monotonicity_status[cell_name] =
389 true; // Initialize to pass, will be set to false if any check fails
390
391 if (cell.contains("output_pins")) {
392 for (const auto &pin : cell["output_pins"]) {
393 if (pin.contains("timing_arcs")) {
394 for (const auto &arc : pin["timing_arcs"]) {
395 // Check 4 timing arc names and accumulate monotonicity status
396 for (const auto &name :
397 {"cell_rise", "cell_fall", "rise_transition", "fall_transition"}) {
398 if (!checkTimingArcMonotonicity(cell, pin, arc, name, is_slew)) {
399 cell_is_monotonic = false;
400 }
401 }
402 }
403 }
404 }
405 }
406 if (cell.contains("input_pins")) {
407 for (const auto &pin : cell["input_pins"]) {
408 if (!pin.contains("clock")) {
409 continue; // Skip non-clock pins
410 }
411 // logger_->debug("Clock pin: {}", pin["pin_name"].get<std::string>());
412 if (pin.contains("timing_arcs")) {
413 for (const auto &arc : pin["timing_arcs"]) {
414 if (arc.contains("timing_type")) {
415 // logger_->debug("Timing type: {}", arc["timing_type"].get<std::string>());
416 if (arc["timing_type"].get<std::string>() == "min_pulse_width") {
417 for (const auto &name : {"rise_constraint", "fall_constraint"}) {
418 if (!checkTimingArcMonotonicity(cell, pin, arc, name, is_slew)) {
419 cell_is_monotonic = false;
420 }
421 }
422 }
423 }
424 }
425 }
426 }
427 }
428 // Update cell status based on all checks
429 cell_monotonicity_status[cell_name] = cell_is_monotonic;
430 if (cell_is_monotonic) {
431 passed_cells++;
432 } else {
433 failed_cells.push_back(cell_name);
434 }
435 }
436 logger_->info("Monotonicity check of '{}' completed.", filename_);
437
438 // Output summary statistics
439 logger_->info("validate_monotonicity : {} out of {} cells passed", passed_cells, total_cells);
440 logger_->info("validate_monotonicity : {} out of {} cells failed", total_cells - passed_cells,
441 total_cells);
442 if (!failed_cells.empty()) {
443 std::stringstream failed_cells_ss;
444 for (size_t i = 0; i < failed_cells.size(); ++i) {
445 failed_cells_ss << failed_cells[i];
446 if (i < failed_cells.size() - 1) {
447 failed_cells_ss << ", "; // Add comma if not the last element
448 }
449 }
450 logger_->info("Failed cell list : {}", failed_cells_ss.str());
451 }
452}

473void LibFile::supercell(const int chain_length, const std::vector<std::string> &cell_names) {
474 /*
475 * Supercells are named as follows:
476 * <cellname>__X<chain_length>__<input_pin>__<output_pin>
477 */
478 logger_->info("Creating supercells for '{}'", filename_);
479
480 if (!std::filesystem::exists(jsonname_)) {
481 logger_->info("JSON file not found. Parsing Liberty file first.");
482 this->parse();
483 this->writeJsonToFile();
484 } else {
485 // Read the JSON file into json object
486 std::ifstream in(jsonname_);
487 if (!in.is_open()) {
488 logger_->error("Could not open file '{}' for reading", jsonname_);
489 return;
490 }
491 try {
492 lib_json_ = json::parse(in);
493 } catch (const json::parse_error &e) {
494 logger_->error("JSON parsing error in file '{}': {}", jsonname_, e.what());
495 in.close();
496 return;
497 }
498 in.close();
499 }
500
501 // write to .map file
502 std::ofstream out(basename_ + ".map");
503 if (!out.is_open()) {
504 logger_->error("Could not open file '{}' for writing", basename_ + ".map");
505 return;
506 }
507 // if celll_names is empty, create supercells for all cells
508 // else create supercells for only the specified cells
509
510 // Check if we're processing specific cells or all cells
511 bool process_all_cells = cell_names.empty();
512
513 if (process_all_cells) {
514 logger_->info("Creating supercells for ALL cells in '{}'", filename_);
515 } else {
516 logger_->info("Creating supercells for {} specified cells in '{}'", cell_names.size(),
517 filename_);
518 }
519
520 // Create a set for faster lookups if we have specific cell names
521 std::unordered_set<std::string> cell_set;
522 std::unordered_set<std::string> found_cells;
523 if (!process_all_cells) {
524 cell_set.insert(cell_names.begin(), cell_names.end());
525 }
526
527 for (const auto &cell : lib_json_["cells"]) {
528 bool is_sequential = false;
529 std::string cell_name = cell["cell_name"].get<std::string>();
530
531 // Skip cells not in the specified list
532 if (!process_all_cells && cell_set.find(cell_name) == cell_set.end()) {
533 continue;
534 }
535
536 // Mark this cell as found
537 if (!process_all_cells) {
538 found_cells.insert(cell_name);
539 }
540
541 logger_->debug("Creating supercells for cell: '{}'", cell_name);
542
543 std::vector<std::string> output_pins;
544 std::vector<std::string> input_pins;
545 // Extract input and output pins
546 if (cell.contains("output_pins")) {
547 for (const auto &pin : cell["output_pins"]) {
548 output_pins.push_back(pin["pin_name"].get<std::string>());
549 }
550 }
551 if (cell.contains("input_pins")) {
552 for (const auto &pin : cell["input_pins"]) {
553 if (pin.contains("clock")) {
554 is_sequential = true;
555 // clock pin not use for supercell
556 continue;
557 }
558 input_pins.push_back(pin["pin_name"].get<std::string>());
559 }
560 }
561
562 // create supercells for all combinations of input and output pins
563 for (const auto &output_pin : output_pins) {
564 for (const auto &input_pin : input_pins) {
565 if (is_sequential) {
566 const int chain_len = 1;
567 std::string supercell_name =
568 cell_name + "__X" + std::to_string(chain_len) + "__" + input_pin + "__" + output_pin;
569 out << cell_name << " " << supercell_name << std::endl;
570 } else {
571 std::string supercell_name = cell_name + "__X" + std::to_string(chain_length) + "__" +
572 input_pin + "__" + output_pin;
573 out << cell_name << " " << supercell_name << std::endl;
574 }
575 }
576 }
577 }
578
579 // Check for cells that were specified but not found
580 if (!process_all_cells) {
581 std::vector<std::string> not_found_cells;
582 for (const auto &requested_cell : cell_names) {
583 if (found_cells.find(requested_cell) == found_cells.end()) {
584 not_found_cells.push_back(requested_cell);
585 }
586 }
587
588 // Output warning for cells that weren't found
589 if (!not_found_cells.empty()) {
590 std::string missing_cells = not_found_cells[0];
591 for (size_t i = 1; i < not_found_cells.size(); ++i) {
592 missing_cells += ", " + not_found_cells[i];
593 }
594 logger_->warn("{} specified cell{} not found in the library: {}", not_found_cells.size(),
595 not_found_cells.size() > 1 ? "s were" : " was", missing_cells);
596 }
597 }
598
599 out.close();
600 logger_->info("Supercell creation complete in '{}'", basename_ + ".map");
601}

627void LibFile::verilog(const int chain_length, const std::vector<std::string> &cell_names) {
628 logger_->info("Creating Verilog for '{}'", filename_);
629
630 this->supercell(chain_length, cell_names);
631
632 // Create a temporary file for individual modules
633 std::string temp_file = basename_ + "_temp.v";
634 std::ofstream out(temp_file);
635 if (!out.is_open()) {
636 logger_->error("Could not open temp file '{}' for writing", temp_file);
637 return;
638 }
639
640 // Read the .map file into a vector to preserve all entries and their order
641 std::vector<std::pair<std::string, std::string>> supercell_entries;
642 std::ifstream in(basename_ + ".map");
643 if (!in.is_open()) {
644 logger_->error("Could not open file '{}' for reading", basename_ + ".map");
645 return;
646 }
647
648 std::string line;
649 while (std::getline(in, line)) {
650 std::istringstream iss(line);
651 std::string cell_name, supercell_name;
652 if (!(iss >> cell_name >> supercell_name)) {
653 logger_->warn("Invalid line format in map file: '{}'", line);
654 continue;
655 }
656 supercell_entries.push_back({cell_name, supercell_name});
657 }
658 in.close();
659 logger_->info("Read {} supercells from '{}'", supercell_entries.size(), basename_ + ".map");
660
661 // Store module information for top-level creation
662 std::vector<std::string> module_names;
663 std::map<std::string, std::vector<std::string>> module_inputs;
664 std::map<std::string, std::vector<std::string>> module_outputs;
665
666 // Generate verilog module for each supercell
667 for (const auto &supercell_entry : supercell_entries) {
668 // Check if the cell is sequential
669 bool is_sequential = false;
670 // Count the number of instances will be created in verilog
671 int instance_count = 0;
672 // Get original cell name from pair
673 const std::string &cell_name = supercell_entry.first;
674 // Get supercell name(as well as module name) from pair
675 const std::string &module_name = supercell_entry.second;
676
677 // Store module name for top-level creation
678 module_names.push_back(module_name);
679
680 logger_->info("Creating Module: '{}' from Cell '{}", module_name, cell_name);
681
682 // Get the cell JSON object
683 json cell_json;
684 for (const auto &cell : lib_json_["cells"]) {
685 if (cell["cell_name"].get<std::string>() == cell_name) {
686 cell_json = cell;
687 break;
688 }
689 }
690
691 // Get input/output pins from the cell JSON object
692 std::vector<std::string> input_pins;
693 std::vector<std::string> output_pins;
694 std::stringstream input_pins_ss;
695 std::stringstream output_pins_ss;
696 if (cell_json.contains("input_pins")) {
697 for (const auto &pin : cell_json["input_pins"]) {
698 if (pin.contains("clock")) {
699 is_sequential = true;
700 }
701 std::string pin_name = pin["pin_name"].get<std::string>();
702 input_pins.push_back(pin_name);
703 input_pins_ss << pin_name << ", ";
704
705 // Store input pin name for top-level creation
706 module_inputs[module_name].push_back(pin_name);
707 }
708 }
709 if (cell_json.contains("output_pins")) {
710 for (const auto &pin : cell_json["output_pins"]) {
711 std::string pin_name = pin["pin_name"].get<std::string>();
712 output_pins.push_back(pin_name);
713 output_pins_ss << pin_name << ", ";
714
715 // Store output pin name for top-level creation
716 module_outputs[module_name].push_back(pin_name);
717 }
718 }
719 logger_->debug("Input pins set: {}", input_pins_ss.str());
720 logger_->debug("Output pins set: {}", output_pins_ss.str());
721
722 // Sequential cells will have only one instance
723 // Combinational cells will have instances equal to chain length
724 if (is_sequential) {
725 instance_count = 1;
726 } else {
727 instance_count = chain_length;
728 }
729
730 // Create ANSI port list
731 std::string port_list_text = "(";
732 // Add input ports
733 bool isFirstPort = true;
734 for (const auto &input_pin : input_pins) {
735 if (!isFirstPort) {
736 port_list_text += ", ";
737 }
738 port_list_text += "input " + input_pin;
739 isFirstPort = false;
740 }
741 // Add output ports
742 for (const auto &output_pin : output_pins) {
743 if (!isFirstPort) {
744 port_list_text += ", ";
745 }
746 port_list_text += "output " + output_pin;
747 isFirstPort = false;
748 }
749 port_list_text += ")";
750 logger_->debug("ANSI Port list: {}", port_list_text);
751
752 // Creat full module header text
753 std::string fullModuleText = "module " + module_name + port_list_text + ";\nendmodule";
754 logger_->debug("Full module text: {}", fullModuleText);
755
756 // Generate the syntax tree for the module using slang library
