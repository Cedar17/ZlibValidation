219bool LibFile::checkTimingArcMonotonicity(const json &cell, const json &pin, const json &arc,
220 const std::string &timing_arc_name, const bool is_slew) {
221 if (arc.contains("when") && !arc["when"].get<std::string>().empty()) {
222 logger_->debug("Checking cell: '{}', pin: '{}', related_pin: '{}', timing_arc: '{}', when: '{}'",
223 cell["cell_name"].get<std::string>(), pin["pin_name"].get<std::string>(),
224 arc["related_pin"].get<std::string>(), timing_arc_name,
225 arc["when"].get<std::string>());
226 } else {
227 logger_->debug("Checking cell: '{}', pin: '{}', related_pin: '{}', timing_arc: '{}'",
228 cell["cell_name"].get<std::string>(), pin["pin_name"].get<std::string>(),
229 arc["related_pin"].get<std::string>(), timing_arc_name);
230 }
231 bool is_monotonic = true; // Assume the values are monotonic initially
232 if (arc.contains(timing_arc_name) && arc[timing_arc_name].contains("values")) {
233 std::vector<std::vector<double>> value_matrix;
234 // Generate a matrix of values from the JSON array
235 for (const auto &row_val : arc[timing_arc_name]["values"]) {
236 std::vector<double> row_data;
237 // Check if the value is an array
238 if (!row_val.is_array()) {
239 logger_->error("Invalid format: '{}' values should be an array of arrays in cell '{}', pin "
240 "'{}', related_pin '{}'",
241 timing_arc_name, cell["cell_name"].get<std::string>(),
242 pin["pin_name"].get<std::string>(), arc["related_pin"].get<std::string>());
243 return false;
244 }
245 // Check for non-numeric values
246 for (const auto &val : row_val) {
247 if (val.is_number()) {
248 row_data.push_back(val.get<double>());
249 } else {
250 logger_->warn("Non-numeric value found in '{}'.values, skipping value: {} in cell '{}', "
251 "pin '{}', related_pin '{}'",
252 timing_arc_name, val.dump(), cell["cell_name"].get<std::string>(),
253 pin["pin_name"].get<std::string>(), arc["related_pin"].get<std::string>());
254 }
255 }
256 value_matrix.push_back(row_data);
257 }
258 // Check the matrix for monotonicity
259 if (!value_matrix.empty() && !value_matrix[0].empty()) {
260 // Check the monotonic incrementality of rows (by output load capacitance, index 2)
261 for (size_t i = 0; i < value_matrix.size(); ++i) {
262 for (size_t j = 1; j < value_matrix[i].size(); ++j) {
263 if ((value_matrix[i][j] < value_matrix[i][j - 1] && pin["pin_name"] != arc["related_pin"]) ||
264 (value_matrix[i][j] == 0 && value_matrix[i][j - 1] == 0)) {
265 // If contains "when" key, log the when value
266 if (arc.contains("when") && !arc["when"].get<std::string>().empty()) {
267 logger_->warn("Non-monotonic (by load) '{}' values: ({}, {}) {} < ({}, {}) {} "
268 "for Cell: {} Pin: {}->{} when: \"{}\"",
269 timing_arc_name, i, j, value_matrix[i][j], i, j - 1, value_matrix[i][j - 1],
270 cell["cell_name"].get<std::string>(), arc["related_pin"].get<std::string>(),
271 pin["pin_name"].get<std::string>(), arc["when"].get<std::string>());
272 } else {
273 logger_->warn("Non-monotonic (by load) '{}' values: ({}, {}) {} < ({}, {}) {} "
274 "for Cell: {} Pin: {}->{}",
275 timing_arc_name, i, j, value_matrix[i][j], i, j - 1, value_matrix[i][j - 1],
276 cell["cell_name"].get<std::string>(), arc["related_pin"].get<std::string>(),
277 pin["pin_name"].get<std::string>());
278 }
279 is_monotonic = false;
280 }
281 }
282 }
283 if (is_slew) {
284 // Check the monotonic incrementality of columns (by input slew, index 1)
285 for (size_t j = 0; j < value_matrix[0].size(); ++j) {
286 for (size_t i = 1; i < value_matrix.size(); ++i) {
287 if ((value_matrix[i][j] < value_matrix[i - 1][j] && pin["pin_name"] != arc["related_pin"]) ||
288 (value_matrix[i][j] == 0 && value_matrix[i - 1][j] == 0)) {
289 // If contains "when" key, log the when value
290 if (arc.contains("when") && !arc["when"].get<std::string>().empty()) {
291 logger_->warn("Non-monotonic (by slew) '{}' values: ({}, {}) {} < ({}, {}) {} "
292 "for Cell: {} Pin: {}->{} when: \"{}\"",
293 timing_arc_name, i, j, value_matrix[i][j], i - 1, j,
294 value_matrix[i - 1][j], cell["cell_name"].get<std::string>(),
295 arc["related_pin"].get<std::string>(), pin["pin_name"].get<std::string>(),
296 arc["when"].get<std::string>());
297 } else {
298 logger_->warn("Non-monotonic (by slew) '{}' values: ({}, {}) {} < ({}, {}) {} "
299 "for Cell: {} Pin: {}->{}",
300 timing_arc_name, i, j, value_matrix[i][j], i - 1, j,
301 value_matrix[i - 1][j], cell["cell_name"].get<std::string>(),
302 arc["related_pin"].get<std::string>(), pin["pin_name"].get<std::string>());
303 }
304 is_monotonic = false;
305 }
306 }
307 }
308 }
309 } else {
310 logger_->warn(
311 "Empty or invalid 'values' array found for cell: '{}', pin: '{}', related_pin: '{}', "
312 "timing_arc: '{}'",
313 cell["cell_name"].get<std::string>(), pin["pin_name"].get<std::string>(),
314 arc["related_pin"].get<std::string>(), timing_arc_name);
315 }
316 }
317 return is_monotonic;
318}

340void LibFile::mono(const bool is_slew) {
341 /*
342 * Use this command to check the following data in the current library to ensure
343 * the tables are monotonically increasing with respect to output load:
344 * cell_rise retaining_rise
345 * cell_fall retaining_fall
346 * rise_transition retain_rise_slew
347 * fall_transition retain_fall_slew
348 * mpw
349 */
350 logger_->info("Monotonicity check of '{}' starting ...", filename_);
351
352 if (!std::filesystem::exists(jsonname_)) {
353 logger_->info("JSON file not found. Parsing Liberty file first.");
354 this->parse();
355 this->writeJsonToFile();
356 } else {
357 // Read the JSON file into json object
358 std::ifstream in(jsonname_);
359 if (!in.is_open()) {
360 logger_->error("Could not open file '{}' for reading", jsonname_);
361 return;
362 }
363 try {
364 lib_json_ = json::parse(in);
365 } catch (const json::parse_error &e) {
366 logger_->error("JSON parsing error in file '{}': {}", jsonname_, e.what());
367 in.close();
368 return;
369 }
370 in.close();
371 }
372
373 std::map<std::string, bool> cell_monotonicity_status; // Track pass/fail status for each cell
374 std::vector<std::string> failed_cells; // List of failed cell names
375 int total_cells = 0;
376 int passed_cells = 0;
377
378 // Check the monotonicity of delay values
379 // The index 1 (time values) must be monotonically increasing and >= 0
380 for (const auto &cell : lib_json_["cells"]) {
381 total_cells++;
382 bool cell_is_monotonic = true; // Assume cell is monotonic initially
383 std::string cell_name = cell["cell_name"].get<std::string>();
384 cell_monotonicity_status[cell_name] =
385 true; // Initialize to pass, will be set to false if any check fails
386
387 if (cell.contains("output_pins")) {
388 for (const auto &pin : cell["output_pins"]) {
389 if (pin.contains("timing_arcs")) {
390 for (const auto &arc : pin["timing_arcs"]) {
391 // Check 4 timing arc names and accumulate monotonicity status
392 for (const auto &name : {"cell_rise", "cell_fall", "rise_transition", "fall_transition"}) {
393 if (!checkTimingArcMonotonicity(cell, pin, arc, name, is_slew)) {
394 cell_is_monotonic = false;
395 }
396 }
397 }
398 }
399 }
400 }
401 if (cell.contains("input_pins")) {
402 for (const auto &pin : cell["input_pins"]) {
403 if (!pin.contains("clock")) {
404 continue; // Skip non-clock pins
405 }
406 // logger_->debug("Clock pin: {}", pin["pin_name"].get<std::string>());
407 if (pin.contains("timing_arcs")) {
408 for (const auto &arc : pin["timing_arcs"]) {
409 if (arc.contains("timing_type")) {
410 // logger_->debug("Timing type: {}", arc["timing_type"].get<std::string>());
411 if (arc["timing_type"].get<std::string>() == "min_pulse_width") {
412 for (const auto &name : {"rise_constraint", "fall_constraint"}) {
413 if (!checkTimingArcMonotonicity(cell, pin, arc, name, is_slew)) {
414 cell_is_monotonic = false;
415 }
416 }
417 }
418 }
419 }
420 }
421 }
422 }
423 // Update cell status based on all checks
424 cell_monotonicity_status[cell_name] = cell_is_monotonic;
425 if (cell_is_monotonic) {
426 passed_cells++;
427 } else {
428 failed_cells.push_back(cell_name);
429 }
430 }
431 logger_->info("Monotonicity check of '{}' completed.", filename_);
432
433 // Output summary statistics
434 logger_->info("validate_monotonicity : {} out of {} cells passed", passed_cells, total_cells);
435 logger_->info("validate_monotonicity : {} out of {} cells failed", total_cells - passed_cells,
436 total_cells);
437 if (!failed_cells.empty()) {
438 std::stringstream failed_cells_ss;
439 for (size_t i = 0; i < failed_cells.size(); ++i) {
440 failed_cells_ss << failed_cells[i];
441 if (i < failed_cells.size() - 1) {
442 failed_cells_ss << ", "; // Add comma if not the last element
443 }
444 }
445 logger_->info("Failed cell list : {}", failed_cells_ss.str());
446 }
447}

467void LibFile::supercell(const int chain_length, const std::vector<std::string> &cell_names) {
468 /*
469 * Supercells are named as follows:
470 * <cellname>__X<chain_length>__<input_pin>__<output_pin>
471 */
472 logger_->info("Creating supercells for '{}'", filename_);
473
474 if (!std::filesystem::exists(jsonname_)) {
475 logger_->info("JSON file not found. Parsing Liberty file first.");
476 this->parse();
477 this->writeJsonToFile();
478 } else {
479 // Read the JSON file into json object
480 std::ifstream in(jsonname_);
481 if (!in.is_open()) {
482 logger_->error("Could not open file '{}' for reading", jsonname_);
483 return;
484 }
485 try {
486 lib_json_ = json::parse(in);
487 } catch (const json::parse_error &e) {
488 logger_->error("JSON parsing error in file '{}': {}", jsonname_, e.what());
489 in.close();
490 return;
491 }
492 in.close();
493 }
494
495 // write to .map file
496 std::ofstream out(basename_ + ".map");
497 if (!out.is_open()) {
498 logger_->error("Could not open file '{}' for writing", basename_ + ".map");
499 return;
500 }
501 // if celll_names is empty, create supercells for all cells
502 // else create supercells for only the specified cells
503
504 // Check if we're processing specific cells or all cells
505 bool process_all_cells = cell_names.empty();
506
507 if (process_all_cells) {
508 logger_->info("Creating supercells for ALL cells in '{}'", filename_);
509 } else {
510 logger_->info("Creating supercells for {} specified cells in '{}'", cell_names.size(), filename_);
511 }
512
513 // Create a set for faster lookups if we have specific cell names
514 std::unordered_set<std::string> cell_set;
515 std::unordered_set<std::string> found_cells;
516 if (!process_all_cells) {
517 cell_set.insert(cell_names.begin(), cell_names.end());
518 }
519
520 for (const auto &cell : lib_json_["cells"]) {
521 bool is_sequential = false;
522 std::string cell_name = cell["cell_name"].get<std::string>();
523
524 // Skip cells not in the specified list
525 if (!process_all_cells && cell_set.find(cell_name) == cell_set.end()) {
526 continue;
527 }
528
529 // Mark this cell as found
530 if (!process_all_cells) {
531 found_cells.insert(cell_name);
532 }
533
534 logger_->debug("Creating supercells for cell: '{}'", cell_name);
535
536 std::vector<std::string> output_pins;
537 std::vector<std::string> input_pins;
538 // Extract input and output pins
539 if (cell.contains("output_pins")) {
540 for (const auto &pin : cell["output_pins"]) {
541 output_pins.push_back(pin["pin_name"].get<std::string>());
542 }
543 }
544 if (cell.contains("input_pins")) {
545 for (const auto &pin : cell["input_pins"]) {
546 if (pin.contains("clock")) {
547 is_sequential = true;
548 // clock pin not use for supercell
549 continue;
550 }
551 input_pins.push_back(pin["pin_name"].get<std::string>());
552 }
553 }
554
555 // create supercells for all combinations of input and output pins
556 for (const auto &output_pin : output_pins) {
557 for (const auto &input_pin : input_pins) {
558 if (is_sequential) {
559 const int chain_len = 1;
560 std::string supercell_name =
561 cell_name + "__X" + std::to_string(chain_len) + "__" + input_pin + "__" + output_pin;
562 out << cell_name << " " << supercell_name << std::endl;
563 } else {
564 std::string supercell_name =
565 cell_name + "__X" + std::to_string(chain_length) + "__" + input_pin + "__" + output_pin;
566 out << cell_name << " " << supercell_name << std::endl;
567 }
568 }
569 }
570 }
571
572 // Check for cells that were specified but not found
573 if (!process_all_cells) {
574 std::vector<std::string> not_found_cells;
575 for (const auto &requested_cell : cell_names) {
576 if (found_cells.find(requested_cell) == found_cells.end()) {
577 not_found_cells.push_back(requested_cell);
578 }
579 }
580
581 // Output warning for cells that weren't found
582 if (!not_found_cells.empty()) {
583 std::string missing_cells = not_found_cells[0];
584 for (size_t i = 1; i < not_found_cells.size(); ++i) {
585 missing_cells += ", " + not_found_cells[i];
586 }
587 logger_->warn("{} specified cell{} not found in the library: {}", not_found_cells.size(),
588 not_found_cells.size() > 1 ? "s were" : " was", missing_cells);
589 }
590 }
591
592 out.close();
593 logger_->info("Supercell creation complete in '{}'", basename_ + ".map");
594}

596void LibFile::verilog(const int chain_length, const std::vector<std::string> &cell_names) {
597 logger_->info("Creating Verilog for '{}'", filename_);
598
599 this->supercell(chain_length, cell_names);
600
601 // write to .v file
602 std::ofstream out(basename_ + ".v");
603 if (!out.is_open()) {
604 logger_->error("Could not open file '{}' for writing", basename_ + ".v");
605 return;
606 }
607
608 // Read the .map file into a vector to preserve all entries and their order
609 std::vector<std::pair<std::string, std::string>> supercell_entries;
610 std::ifstream in(basename_ + ".map");
611 if (!in.is_open()) {
612 logger_->error("Could not open file '{}' for reading", basename_ + ".map");
613 return;
614 }
615
616 std::string line;
617 while (std::getline(in, line)) {
618 std::istringstream iss(line);
619 std::string cell_name, supercell_name;
620 if (!(iss >> cell_name >> supercell_name)) {
621 logger_->warn("Invalid line format in map file: '{}'", line);
622 continue;
623 }
624 supercell_entries.push_back({cell_name, supercell_name});
625 }
626 in.close();
627 logger_->info("Read {} supercells from '{}'", supercell_entries.size(), basename_ + ".map");
628
629 // Generate verilog module for each supercell
630 for (const auto &supercell_entry : supercell_entries) {
631 // Check if the cell is sequential
632 bool is_sequential = false;
633 // Count the number of instances will be created in verilog
634 int instance_count = 0;
635 // Get original cell name from pair
636 const std::string &cell_name = supercell_entry.first;
637 // Get supercell name(as well as module name) from pair
638 const std::string &module_name = supercell_entry.second;
639
640 logger_->info("Creating Module: '{}' from Cell '{}", module_name, cell_name);
641
642 // Get the cell JSON object
643 json cell_json;
644 for (const auto &cell : lib_json_["cells"]) {
645 if (cell["cell_name"].get<std::string>() == cell_name) {
646 cell_json = cell;
647 break;
648 }
649 }
650
651 // Get input/output pins from the cell JSON object
652 std::vector<std::string> input_pins;
653 std::vector<std::string> output_pins;
654 std::stringstream input_pins_ss;
655 std::stringstream output_pins_ss;
656 if (cell_json.contains("input_pins")) {
657 for (const auto &pin : cell_json["input_pins"]) {
658 if (pin.contains("clock")) {
659 is_sequential = true;
660 }
661 input_pins.push_back(pin["pin_name"].get<std::string>());
662 input_pins_ss << pin["pin_name"].get<std::string>() << ", ";
663 }
664 }
665 if (cell_json.contains("output_pins")) {
666 for (const auto &pin : cell_json["output_pins"]) {
667 output_pins.push_back(pin["pin_name"].get<std::string>());
668 output_pins_ss << pin["pin_name"].get<std::string>() << ", ";
669 }
670 }
671 logger_->debug("Input pins set: {}", input_pins_ss.str());
672 logger_->debug("Output pins set: {}", output_pins_ss.str());
673
674 // Sequential cells will have only one instance
675 // Combinational cells will have instances equal to chain length
676 if (is_sequential) {
677 instance_count = 1;
678 } else {
679 instance_count = chain_length;
680 }
681
682 // Create ANSI port list
683 std::string port_list_text = "(";
684 // Add input ports
685 bool isFirstPort = true;
686 for (const auto &input_pin : input_pins) {
687 if (!isFirstPort) {
688 port_list_text += ", ";
689 }
690 port_list_text += "input " + input_pin;
691 isFirstPort = false;
692 }
693 // Add output ports
694 for (const auto &output_pin : output_pins) {
695 if (!isFirstPort) {
696 port_list_text += ", ";
697 }
698 port_list_text += "output " + output_pin;
699 isFirstPort = false;
700 }
701 port_list_text += ")";
702 logger_->debug("ANSI Port list: {}", port_list_text);
703
704 // Creat full module header text
705 std::string fullModuleText = "module " + module_name + port_list_text + ";\nendmodule";
706 logger_->debug("Full module text: {}", fullModuleText);
707
708 // Generate the syntax tree for the module using slang library
709 auto tree = slang::syntax::SyntaxTree::fromText(fullModuleText);
710 if (tree) {
711 // Add ports to the syntax tree
712 ModuleRewriter rewriter(input_pins, output_pins, supercell_entry, instance_count, logger_);
713
714 tree = rewriter.transform(tree);
715 // Revisit the syntax tree to check architecture
716 // rewriter.visit(tree->root());
717
718 // Output the transformed syntax tree
719 out << "`timescale 1ns/1ps" << std::endl;
720 out << slang::syntax::SyntaxPrinter::printFile(*tree) << std::endl << std::endl;
721 } else {
722 logger_->error("Failed to create syntax tree for module '{}'", module_name);
723 continue;
724 }
725 }
