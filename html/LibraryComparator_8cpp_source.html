98void LibraryComparator::compareLut(const std::string &cell_name, const std::string &pin_name,
99 const std::string &timing_type, const std::string &related_pin,
100 const std::string &arc_name, const json &ref_lut,
101 const json &comp_lut, Table &table) {
102 spdlog::info("Comparing LUT of '{}' ...", arc_name);
103 std::vector<double> ref_index_1 = ref_lut["index_1"].get<std::vector<double>>();
104 std::vector<double> ref_index_2 = ref_lut["index_2"].get<std::vector<double>>();
105 std::vector<double> comp_index_1 = comp_lut["index_1"].get<std::vector<double>>();
106 std::vector<double> comp_index_2 = comp_lut["index_2"].get<std::vector<double>>();
107 if (ref_index_1 != comp_index_1 || ref_index_2 != comp_index_2) {
108 spdlog::warn("Mismatch in LUT indices for '{}' in cell: '{}', pin: '{}', related_pin: '{}', "
109 "timing_type: '{}'",
110 arc_name, cell_name, pin_name, related_pin, timing_type);
111 return;
112 } else {
113 spdlog::debug("LUT indices match for '{}'", arc_name);
114
115 std::vector<std::vector<double>> comp_value_matrix;
116 // Generate a matrix of values from the JSON array for comparison library
117 for (const auto &row_val : comp_lut["values"]) {
118 std::vector<double> row_data;
119 // Check if the row value is an array
120 if (!row_val.is_array()) {
121 spdlog::error(
122 "Invalid LUT format: '{}' values should be an array of arrays in comp_lib '{}', "
123 "cell '{}', pin '{}', "
124 "related_pin '{}', timing_type '{}'",
125 arc_name, this->comp_lib_path_.string(), cell_name, pin_name, related_pin, timing_type);
126 return;
127 }
128 // Check if non-numeric values
129 for (const auto &val : row_val) {
130 if (val.is_number()) {
131 row_data.push_back(val.get<double>());
132 } else {
133 spdlog::error("Non-numeric value found in '{}'.values, skipping value: {} in comp_lib "
134 "'{}', cell '{}', "
135 "pin '{}', related_pin '{}', timing_type '{}'",
136 arc_name, val.dump(), this->comp_lib_path_.string(), cell_name, pin_name,
137 related_pin, timing_type);
138 return;
139 }
140 }
141 comp_value_matrix.push_back(row_data);
142 }
143 // Generate a matrix of values from the JSON array for reference library
144 std::vector<std::vector<double>> ref_value_matrix;
145 for (const auto &row_val : ref_lut["values"]) {
146 std::vector<double> row_data;
147 // Check if the row value is an array
148 if (!row_val.is_array()) {
149 spdlog::error(
150 "Invalid LUT format: '{}' values should be an array of arrays in ref_lib '{}', "
151 "cell '{}', pin '{}', "
152 "related_pin '{}', timing_type '{}'",
153 arc_name, this->ref_lib_path_.string(), cell_name, pin_name, related_pin, timing_type);
154 return;
155 }
156 // Check if non-numeric values
157 for (const auto &val : row_val) {
158 if (val.is_number()) {
159 row_data.push_back(val.get<double>());
160 } else {
161 spdlog::error("Non-numeric value found in '{}'.values, skipping value: {} in ref_lib "
162 "'{}', cell '{}', "
163 "pin '{}', related_pin '{}', timing_type '{}'",
164 arc_name, val.dump(), this->ref_lib_path_.string(), cell_name, pin_name,
165 related_pin, timing_type);
166 return;
167 }
168 }
169 ref_value_matrix.push_back(row_data);
170 }
171 // Compare the matrix for relative tolerance
172 if (!comp_value_matrix.empty() && !comp_value_matrix[0].empty()) {
173 for (size_t i = 0; i < comp_value_matrix.size(); ++i) {
174 for (size_t j = 0; j < comp_value_matrix[i].size(); ++j) {
175 double ref_val = ref_value_matrix[i][j];
176 double comp_val = comp_value_matrix[i][j];
177 if (std::abs(ref_val - comp_val) > reltol_ * std::abs(ref_val) &&
178 std::abs(ref_val - comp_val) > abstol_) {
179 spdlog::debug("LUT value mismatch for '{}', index_1: {}, index_2: {}", arc_name,
180 ref_index_1[i], ref_index_2[j]);
181 table.add_row({related_pin + "->" + pin_name, std::to_string(ref_val),
182 std::to_string(comp_val), std::to_string(comp_val - ref_val),
183 std::to_string((comp_val - ref_val) / ref_val * 100), timing_type,
184 arc_name, std::to_string(i + 1), std::to_string(ref_index_1[i]),
185 std::to_string(j + 1), std::to_string(ref_index_2[j]), "<"});
186 spdlog::debug("table size: {}", table.size());
187 } else {
188 spdlog::debug("LUT value match for '{}', index_1: {}, index_2: {}", arc_name,
189 ref_index_1[i], ref_index_2[j]);
190 }
191 }
192 }
193 } else {
194 spdlog::error(
195 "Empty or invalid 'values' array found for cell: '{}', pin: '{}', related_pin: '{}', "
196 "timing_type: '{}', arc: '{}'",
197 cell_name, pin_name, related_pin, timing_type, arc_name);
198 }
199 }

261void LibraryComparator::comparePin(const std::string &cell_name, const std::string &pin_name,
262 const json &ref_pin, const json &comp_pin, Table &table) {
263 spdlog::info("Comparing pin '{}' ...", pin_name);
264
265 if (comp_pin.contains("timing_arcs")) {
266 for (const auto &comp_arc : comp_pin["timing_arcs"]) {
267 std::string timing_type = comp_arc["timing_type"].get<std::string>();
268 spdlog::debug("Comparing timing type: '{}'", timing_type);
269 // Look for the timing type in the reference JSON
270 auto ref_arc_it = std::find_if(
271 ref_pin["timing_arcs"].begin(), ref_pin["timing_arcs"].end(),
272 [&timing_type](const json &ref_arc) { return ref_arc["timing_type"] == timing_type; });
273 if (ref_arc_it != ref_pin["timing_arcs"].end()) {
274 // Found this timing type
275 spdlog::debug("Found timing type: '{}'", timing_type);
276 compareTimingArc(cell_name, pin_name, timing_type, *ref_arc_it, comp_arc, table);
277 } else {
278 // timing arc not found in reference JSON
279 spdlog::warn("Timing arc: '{}' not found in reference JSON", timing_type);
280 }
281 }
282 } else {
283 spdlog::info("No timing arcs found for pin: '{}'", comp_pin["pin_name"].get<std::string>());

343void LibraryComparator::generateReport(const std::string &output_file) {
344 std::ofstream outfile(output_file);
345 outfile << "# LIBRARY comparison\n" << std::endl;
346 outfile << "**Reference library: " << ref_lib_path_ << "**\n" << std::endl;
347 outfile << "**Comparison library: " << comp_lib_path_ << "**\n" << std::endl;
348 outfile << "**Absolute tolerance: " << abstol_ << "**\n" << std::endl;
349 outfile << "**Relative tolerance: " << reltol_ << "**\n" << std::endl;
350 outfile << "**Performed by " << APP_NAME << " v" << APP_VERSION << " from " << APP_AUTHOR;
351 auto now = std::chrono::system_clock::now();
352 std::time_t now_time_t = std::chrono::system_clock::to_time_t(now);
353 std::tm *now_tm = std::localtime(&now_time_t);
354 outfile << ". on: " << std::put_time(now_tm, "%c") << "**\n" << std::endl;
355 outfile << "> Legend: < outlier, * scaled, ! indices switched, ^ slews extrapolated, ~ loads "
356 "extrapolated,\n";
357 outfile << "> \n";
358 outfile << "> Legend: + padding added, /0 divide by zero, / slews interpolated, # loads "
359 "interpolated\n";
360 outfile << "> \n";
361 outfile << "> Legend: << value is less but unknown, >> value is more but unknown\n\n";
362
363 spdlog::info("Starting comparison report generation ...");
364 for (const auto &comp_cell : comp_json_["cells"]) {
365 std::string cell_name = comp_cell["cell_name"].get<std::string>();
366 spdlog::debug("Comparing cell: '{}'", cell_name);
367 // Look for the cell in the reference JSON
368 auto ref_cell_it = std::find_if(
369 ref_json_["cells"].begin(), ref_json_["cells"].end(),
370 [&cell_name](const json &ref_cell) { return ref_cell["cell_name"] == cell_name; });
371 if (ref_cell_it != ref_json_["cells"].end()) {
372 // Found this cell
373 spdlog::debug("Found cell: '{}'", cell_name);
374 Table table;
375 table.add_row({"Pin Name", "Reference", "Comparison", "Diff", "Diff %", "Type", "Arc Name",
376 "Row #", "Index_1", "Col #", "Index_2", "Note"});
377 // Header formatting
378 for (size_t i = 0; i < table[0].size(); ++i) {
379 table[0][i].format().font_color(Color::yellow).font_style({FontStyle::bold});
380 }
381 compareCell(cell_name, *ref_cell_it, comp_cell, table);
382
383 if (table.size() > 1) {
384 // Data starts from row 1, row 0 is header
385 size_t failed_count = table.size() - 1;
386 double sum_diff = 0.0;
387 double sum_diff_percent = 0.0;
388 size_t outlier_count = 0;
389 double max_diff = 0;
390 double max_diff_percent = 0;
391 std::vector<double> diff_values; // Store diff values for std deviation calculation
392
393 // Index of columns
394 const int diff_index = 3;
395 const int diff_percent_index = 4;
396 const int note_index = 11;
397
398 for (size_t i = 1; i < table.size(); ++i) {
399 try {
400 double diff = std::stod(table[i][diff_index].get_text());
401 double diff_percent = std::stod(table[i][diff_percent_index].get_text());
402
403 sum_diff += diff;
404 sum_diff_percent += diff_percent;
405 diff_values.push_back(diff_percent);
406
407 if (table[i][note_index].get_text() == "<") {
408 outlier_count++;
409 }
410 // Change to absolute value for max diff
411 if (std::abs(diff) > std::abs(max_diff)) {
412 max_diff = diff;
413 max_diff_percent = diff_percent;
414 }
415
416 } catch (const std::invalid_argument &e) {
417 spdlog::error("Could not convert value to double: {}", e.what());
418 continue; // Skip this row if there's an error
419 }
420 }
421
422 double avg_diff = sum_diff / failed_count;
423 double avg_diff_percent = sum_diff_percent / failed_count;
424
425 outfile << "## " << cell_name << "\n\n";
426 outfile << "Delay Comparison in ns\n\n";
427 if (output_file.substr(output_file.size() - 3) == ".md") {
428 MarkdownExporter exporter;
429 auto markdown = exporter.dump(table);
430 outfile << markdown << std::endl;
431 } else {
432 outfile << table << std::endl;
433 }
434 outfile << std::endl;
435 std::cout << table << std::endl;
436 outfile << cell_name << " delay SUMMARY (abstol: " << abstol_ << ", reltol: " << reltol_
437 << ")\n";
438
439 // Create summary table
440 Table summary_table;
441 summary_table.add_row({"Cell Name", "Data Type", "Failed Count", "Avg Diff", "Avg Diff%",
442 "Max Diff", "Max Diff%", "Outliers"});
443 summary_table.add_row({cell_name, "delay(ns)", std::to_string(failed_count),
444 std::to_string(avg_diff), std::to_string(avg_diff_percent) + "%",
445 std::to_string(max_diff), std::to_string(max_diff_percent) + "%",
446 std::to_string(outlier_count)});
447
448 // Output summary table
449 // Use std::filesystem to reliably get the extension
450 std::filesystem::path file_path(output_file);
451 if (file_path.has_extension() && file_path.extension() == ".md") {
452 MarkdownExporter exporter;
453 auto markdown = exporter.dump(summary_table);
454 outfile << markdown << std::endl;
455 } else {
456 outfile << summary_table << std::endl;
457 }
458 outfile << "Worst delay outlier: Max Abs: " << max_diff << ", Max Rel: " << max_diff_percent
459 << "%, Outliers: " << outlier_count << "\n\n";
460 }
461 } else {
462 // cell not found in reference JSON
463 spdlog::warn("Cell: '{}' not found in reference JSON", cell_name);
464 }
465 }
466
467 outfile.close();
