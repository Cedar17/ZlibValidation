77// Handle hierarchical instantiation (module instance)
78void VerilogVisitor::handle(const slang::syntax::HierarchyInstantiationSyntax &hierarchyInst) {
79 if (!inTargetModule_) {
80 return;
81 }
82
83 std::string indent(depth_ * 2, ' ');
84
85 if (hierarchyInst.type.valid()) {
86 std::string_view instType = hierarchyInst.type.valueText();
87 spdlog::info("{}Hierarchy Instance Type: {}", indent, instType);
88
89 // Safely print parameter values (if any)
90 try {
91 if (hierarchyInst.parameters) {
92 spdlog::info("{}Parameters:", indent);
93 for (const auto &param : hierarchyInst.parameters->parameters) {
94 if (!param)
95 continue; // Null pointer check
96
97 if (auto *orderedParam = param->as_if<slang::syntax::OrderedParamAssignmentSyntax>()) {
98 if (orderedParam->expr) {
99 spdlog::info("{} Parameter: {}", indent, orderedParam->expr->toString());
100 }
101 } else if (auto *namedParam = param->as_if<slang::syntax::NamedParamAssignmentSyntax>()) {
102 if (namedParam->name.valid() && namedParam->expr) {
103 spdlog::info("{} Parameter {}: {}", indent, namedParam->name.valueText(),
104 namedParam->expr->toString());
105 }
106 }
107 }
108 }
109 } catch (const std::exception &e) {
110 spdlog::warn("{}Exception while processing parameters: {}", indent, e.what());
111 }
112
113 // Safely handle instances
114 try {
115 for (const auto &instance : hierarchyInst.instances) {
116 if (!instance) {
117 spdlog::warn("{}Invalid instance", indent);
118 continue;
119 } else if (!instance->decl) {
120 spdlog::warn("{}Invalid instance declaration", indent);
121 // continue;
122 } else {
123 try {
124 if (instance->decl->name.valid()) {
125 std::string_view instName = instance->decl->name.valueText();
126 spdlog::info("{}Instance Name: {}", indent, instName);
127 }
128 } catch (...) {
129 spdlog::debug("{}Error printing name", indent);
130 }
131
132 // Safely print dimensions
133 try {
134 if (!instance->decl->dimensions.empty()) {
135 spdlog::info("{} Dimensions: {}", indent, instance->decl->dimensions.toString());
136 }
137 } catch (...) {
138 spdlog::debug("{}Error printing dimensions", indent);
139 }
140 }
141
142 // Safely print port connections
143 spdlog::info("{} Port connections:", indent);
144 for (const auto &conn : instance->connections) {
145 if (!conn) {
146 spdlog::warn("{} Null connection", indent);
147 continue; // Null pointer check
148 }
149
150 try {
151 // Use as_if method for safe type conversion
152 if (auto *ordered = conn->as_if<slang::syntax::OrderedPortConnectionSyntax>()) {
153 if (ordered->expr) {
154 spdlog::info("{} Ordered connection: {}", indent, ordered->expr->toString());
155 } else {
156 spdlog::info("{} Ordered connection: <empty>", indent);
157 }
158 } else if (auto *named = conn->as_if<slang::syntax::NamedPortConnectionSyntax>()) {
159 if (named->name.valid()) {
160 std::string exprStr = named->expr ? named->expr->toString() : "<empty>";
161 spdlog::info("{} .{}({})", indent, named->name.valueText(), exprStr);
162 }
163 } else if (conn->as_if<slang::syntax::EmptyPortConnectionSyntax>()) {
164 spdlog::info("{} <empty connection>", indent);
165 } else if (conn->as_if<slang::syntax::WildcardPortConnectionSyntax>()) {
166 spdlog::info("{} .* (wildcard connection)", indent);
167 } else {
168 spdlog::info("{} Unknown connection type: {}", indent,
169 slang::syntax::toString(conn->kind));
170 }
171 } catch (const std::exception &e) {
172 spdlog::warn("{}Exception processing connection: {}", indent, e.what());
173 } catch (...) {
174 spdlog::warn("{}Unknown exception processing connection", indent);
175 }
176 }
177 }
178 } catch (const std::exception &e) {
179 spdlog::warn("{}Exception while processing instances: {}", indent, e.what());
180 }
181
182 // Continue processing child nodes
183 depth_++;
184 visitDefault(hierarchyInst);
185 depth_--;
186 }

257// Handle specify block
258void VerilogVisitor::handle(const slang::syntax::SpecifyBlockSyntax &specifyBlock) {
259 if (!inTargetModule_) {
260 return;
261 }
262
263 std::string indent(depth_ * 2, ' ');
264 spdlog::info("{}Specify Block:", indent);
265
266 // Iterate through all path declarations
267 for (const auto &item : specifyBlock.items) {
268 if (auto *pathDecl = item->as_if<slang::syntax::PathDeclarationSyntax>()) {
269 if (pathDecl->desc) {
270 std::string pathSrc;
271 std::string pathDst;
272
273 // Get path source
274 if (!pathDecl->desc->inputs.empty()) {
275 // Get the first input
276 if (auto *identifier =
277 pathDecl->desc->inputs[0]->as_if<slang::syntax::IdentifierNameSyntax>()) {
278 if (identifier->identifier.valid()) {
279 pathSrc = identifier->identifier.valueText();
280 }
281 }
282 }
283
284 // Get path destination
285 if (pathDecl->desc->suffix) {
286 if (auto *simpleSuffix =
287 pathDecl->desc->suffix->as_if<slang::syntax::SimplePathSuffixSyntax>()) {
288 if (!simpleSuffix->outputs.empty()) {
289 if (auto *identifier =
290 simpleSuffix->outputs[0]->as_if<slang::syntax::IdentifierNameSyntax>()) {
291 if (identifier->identifier.valid()) {
292 pathDst = identifier->identifier.valueText();
293 }
294 }
295 } else if (auto *edgeSuffix =
296 pathDecl->desc->suffix
297 ->as_if<slang::syntax::EdgeSensitivePathSuffixSyntax>()) {
298 if (!edgeSuffix->outputs.empty()) {
299 if (auto *identifier =
300 edgeSuffix->outputs[0]->as_if<slang::syntax::IdentifierNameSyntax>()) {
301 if (identifier->identifier.valid()) {
302 pathDst = identifier->identifier.valueText();
303 }
304 }
305 }
306 }
307 }
308
309 // Construct path string
310 std::string pathStr = pathSrc + " => " + pathDst;
311
312 // Get path delay
313 std::string delayStr = "(";
314 for (size_t i = 0; i < pathDecl->delays.size(); ++i) {
315 if (i > 0)
316 delayStr += ", ";
317 delayStr += pathDecl->delays[i]->toString();
318 }
319 delayStr += ")";
320
321 spdlog::info("{} Path: {} = {}", indent, pathStr, delayStr);
322 }
323 }
324 // Can add handling for ConditionalPathDeclarationSyntax and IfNonePathDeclarationSyntax
325 else if (auto *condPath = item->as_if<slang::syntax::ConditionalPathDeclarationSyntax>()) {
326 if (condPath->path && condPath->predicate) {
327 spdlog::info("{} Conditional Path (if {})", indent, condPath->predicate->toString());
328 // Recursively handle internal path
329 handle(*condPath->path);
330 }
331 } else if (auto *ifNonePath = item->as_if<slang::syntax::IfNonePathDeclarationSyntax>()) {
332 if (ifNonePath->path) {
333 spdlog::info("{} If-None Path", indent);
334 // Recursively handle internal path
335 handle(*ifNonePath->path);
336 }
337 }
338 }
339
340 // Continue processing child nodes
341 depth_++;
342 visitDefault(specifyBlock);
343 depth_--;
344 }

406void ModuleRewriter::handle(const slang::syntax::ModuleDeclarationSyntax &module) {
407 logger_->debug("Processing module: {}", module.header->name.valueText());
408
409 // Create intermediate wires for connections between instances
410 for (int i = 0; i < instance_count_ - 1; i++) {
411 auto &newNetNode = parse("\n wire OP_" + std::to_string(i) + ";");
412 insertAtBack(module.members, newNetNode);
413 logger_->debug("Added intermediate wire: {}", newNetNode.toString());
414 }
415
416 // Get critical input port & output port from
417 // the module name pattern: CELLNAME__X#__CRITICALPORT__OUTPUTPORT
418 std::string criticalInputPort = "";
419 std::string criticalOutputPort = "";
420 if (!this->moduleName_.empty()) {
421 size_t firstSep = this->moduleName_.find("__");
422 if (firstSep != std::string::npos) {
423 size_t secondSep = this->moduleName_.find("__", firstSep + 2);
424 if (secondSep != std::string::npos) {
425 size_t thirdSep = this->moduleName_.find("__", secondSep + 2);
426 if (thirdSep != std::string::npos) {
427 criticalInputPort = this->moduleName_.substr(secondSep + 2, thirdSep - (secondSep + 2));
428 criticalOutputPort = this->moduleName_.substr(thirdSep + 2);
429 logger_->debug("Extracted critical input port: {}", criticalInputPort);
430 logger_->debug("Extracted output port: {}", criticalOutputPort);
431 } else {
432 logger_->warn("Can't find thirdSep. Invalid module name pattern: {}", this->moduleName_);
433 return;
434 }
435 } else {
436 logger_->warn("Can't find secondSep. Invalid module name pattern: {}", this->moduleName_);
437 return;
438 }
439 } else {
440 logger_->warn("Can't find firstSep. Invalid module name pattern: {}", this->moduleName_);
441 return;
442 }
443 } else {
444 logger_->warn("Module name is empty. Can't extract critical input port.");
445 return;
446 }
447
448 // Add additional wires for intermediate outputs that aren't part of the chain
449 if (this->outputPins_.size() > 1) { // Only add if there are multiple output pins
450 for (int i = 0; i < instance_count_ - 1; i++) { // Skip the first and last instances
451 for (const auto &outputPin : this->outputPins_) {
452 if (outputPin != criticalOutputPort) { // Found an intermediate output pin
453 auto &newNetNode = parse("\n wire P_" + std::to_string(i) + "__" + outputPin + ";");
454 insertAtBack(module.members, newNetNode);
455 logger_->debug("Added intermediate wire: {}", newNetNode.toString());
456 }
457 }
458 }
459 }
460
461 std::string portList_str = module.header->ports->toString();
462 logger_->debug("Port list: {}", portList_str);
463
464 std::vector<std::string> allPorts;
465 auto ansiPortList = module.header->ports->as_if<slang::syntax::AnsiPortListSyntax>();
466 if (!ansiPortList) {
467 logger_->warn("Port list is not ANSI style.");
468 return;
469 }
470 portInfoMap_.clear(); // Clear the port info map
471 for (const auto portMember : ansiPortList->ports) {
472 if (portMember->kind == slang::syntax::SyntaxKind::ImplicitAnsiPort) {
473 const auto implicitPort = portMember->as_if<slang::syntax::ImplicitAnsiPortSyntax>();
474 const auto &directionToken =
475 implicitPort->header->as_if<slang::syntax::VariablePortHeaderSyntax>()->direction;
476 const auto &nameToken = implicitPort->declarator->name;
477
478 std::string_view portName = nameToken.valueText();
479 std::string_view direction = directionToken.valueText();
480 portInfoMap_[std::string(portName)] = std::string(direction); // Store port info in the map
481 logger_->debug("Port Name: {}, Direction: {}", portName, direction);
482 } else {
483 logger_->warn("Port member is not ImplicitAnsiPort.");
484 }
485 }
486
487 for (int i = 0; i < instance_count_; i++) {
488 std::string instanceName = "I_" + this->cellName_ + "__X" + std::to_string(i) + "__" +
489 criticalInputPort + "__" + criticalOutputPort;
490 std::string instanceCode = "\n " + this->cellName_ + " " + instanceName + " (";
491 std::vector<std::string> portConnections;
492
493 for (const auto &pair : portInfoMap_) {
494 std::string portName = pair.first;
495 std::string direction = pair.second;
496 std::string connectionName;
497
498 if (portName == criticalInputPort) {
499 if (i == 0) {
500 connectionName = portName; // First instance: connect to module input port
501 } else {
502 connectionName =
503 "OP_" + std::to_string(i - 1); // Intermediate instances: connect to previous OP wire
504 }
505 } else if (portName == criticalOutputPort) {
506 if (i == instance_count_ - 1) {
507 connectionName = portName; // Last instance: connect to module output port
508 } else {
509 connectionName = "OP_" + std::to_string(i); // Intermediate instances: connect to OP wire
510 }
511 } else if (direction == "output") { // Handle other output ports
512 if (i < instance_count_ - 1) {
513 connectionName = "P_" + std::to_string(i) + "__" +
514 portName; // Connect to intermediate P_i__portName wire
515 } else {
516 connectionName = portName; // Last instance: connect to module output port
517 }
518 } else { // For input ports (and inout?), connect directly to module port
519 connectionName = portName;
520 }
521 portConnections.push_back("." + portName + "(" + connectionName + ")");
522 }
523
524 // Connect all ports with comma and space
525 if (!portConnections.empty()) {
526 instanceCode += portConnections[0];
527 for (size_t i = 1; i < portConnections.size(); ++i) {
528 instanceCode += ", " + portConnections[i];
529 }
530 }
531 instanceCode += ");";
532
533 // Blank line before the first instance
534 if (i == 0) {
535 instanceCode = "\n" + instanceCode;
536 }
537
538 // Insert the instance code
539 auto &instanceNode = parse(instanceCode);
540 insertAtBack(module.members, instanceNode);
541 }
