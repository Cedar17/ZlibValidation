58bool isIdentifier(const std::string &token) {
59 if (token.empty())
60 return false;
61 // Check if the token is a valid identifier (upper alpha+ numeric + underscore)
62 bool has_alpha = false;
63 for (char c : token) {
64 if (std::isalpha(c)) {
65 has_alpha = true;
66 } else if (!std::isalnum(c) && c != '_') {
67 return false; // Contains invalid char
68 }
69 }
70 // Must start with an uppercase alphabetic character
71 return has_alpha && std::isupper(token[0]);
72}

88std::string LogicComparator::preprocessExpression(const std::string &input_expr) {
89 std::string processed = input_expr;
90 spdlog::debug("Preprocessing raw expression: {}", processed);
91
92 // 0. Trim leading/trailing whitespace first
93 processed = std::regex_replace(processed, std::regex("^\\s+|\\s+$"), "");
94
95 // 1. Make replacements more robust by adding spaces first, then replacing
96 // Add spaces around operators and parentheses that need them
97 processed = std::regex_replace(processed, std::regex("\\("), " ( ");
98 processed = std::regex_replace(processed, std::regex("\\)"), " ) ");
99 processed = std::regex_replace(processed, std::regex("\\+"), " + ");
100 processed = std::regex_replace(processed, std::regex("\\^"), " ^ ");
101 processed = std::regex_replace(processed, std::regex("\\*"), " * ");
102 // Handle '!' carefully, might be part of !=
103 // Let's assume '!' is only used for NOT for now
104 processed =
105 std::regex_replace(processed, std::regex("!([^=])"), " not $1"); // \! followed by non-=
106 processed = std::regex_replace(processed, std::regex("^!"), " not "); // \! at the beginning
107
108 // 2. Replace logical operators with their equivalents(with spaces around them)
109 processed = std::regex_replace(processed, std::regex(" \\+ "), " or ");
110 processed = std::regex_replace(processed, std::regex(" \\^ "), " xor ");
111 processed = std::regex_replace(processed, std::regex(" \\* "), " and ");
112
113 spdlog::debug("Processed expression after replacements: {}", processed);
114
115 // 3. Tokenize based on spaces
116 std::stringstream ss(processed);
117 std::istream_iterator<std::string> begin(ss);
118 std::istream_iterator<std::string> end;
119 std::vector<std::string> tokens(begin, end);
120
121 if (tokens.empty()) {
122 spdlog::debug("No tokens found after preprocessing.");
123 return "";
124 }
125
126 // 4. Insert implied 'and'
127 std::vector<std::string> processed_tokens;
128 processed_tokens.push_back(tokens[0]); // 先添加第一个 token
129
130 for (size_t i = 0; i < tokens.size() - 1; ++i) {
131 const std::string &current_token = tokens[i];
132 const std::string &next_token = tokens[i + 1];
133
134 // 定义可以构成隐式 AND 的结尾和开头 token 类型
135 bool current_ends_operand = isIdentifier(current_token) || current_token == ")";
136 bool next_starts_operand = isIdentifier(next_token) || next_token == "(";
137
138 // 定义不能插入 AND 的情况：当前或下一个是显式操作符，或括号组合不当
139 bool current_is_invalid_before_and = isOperator(current_token) || current_token == "(";
140 bool next_is_invalid_after_and = isOperator(next_token) || next_token == ")";
141
142 // 检查是否应该插入 AND
143 if (current_ends_operand && next_starts_operand && !current_is_invalid_before_and &&
144 !next_is_invalid_after_and) {
145 processed_tokens.push_back("and");
146 }
147
148 processed_tokens.push_back(next_token); // 添加下一个 token
149 }
150
151 // 5. Reconstruct the string
152 std::string final_expr;
153 for (size_t i = 0; i < processed_tokens.size(); ++i) {
154 final_expr += processed_tokens[i] + (i == processed_tokens.size() - 1 ? "" : " ");
155 }
156
157 // 6. Final cleanup of spaces, especially around parentheses
158 final_expr = std::regex_replace(final_expr, std::regex("\\s+"), " ");
159 final_expr = std::regex_replace(final_expr, std::regex(" \\( "), "(");
160 final_expr = std::regex_replace(final_expr, std::regex("\\( "), "(");
161 final_expr = std::regex_replace(final_expr, std::regex(" \\("), "(");
162 final_expr = std::regex_replace(final_expr, std::regex(" \\) "), ")");
163 final_expr = std::regex_replace(final_expr, std::regex("\\) "), ")");
164 final_expr = std::regex_replace(final_expr, std::regex(" \\)"), ")");
165 final_expr = std::regex_replace(final_expr, std::regex("^\\s+|\\s+$"), "");
166
167 spdlog::debug("Preprocessed expression result: {}", final_expr);
168 return final_expr;
169}
