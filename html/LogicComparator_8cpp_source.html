74bool isIdentifier(const std::string &token) {
75 if (token.empty())
76 return false;
77 // Check if the token is a valid identifier (upper alpha+ numeric + underscore)
78 bool has_alpha = false;
79 for (char c : token) {
80 if (std::isalpha(c)) {
81 has_alpha = true;
82 } else if (!std::isalnum(c) && c != '_') {
83 return false; // Contains invalid char
84 }
85 }
86 // Must start with an uppercase alphabetic character
87 return has_alpha && std::isupper(token[0]);
88}

104std::string LogicComparator::preprocessExpression(const std::string &input_expr) {
105 std::string processed = input_expr;
106 spdlog::debug("Preprocessing raw expression: {}", processed);
107
108 // 0. Trim leading/trailing whitespace first
109 processed = std::regex_replace(processed, std::regex("^\\s+|\\s+$"), "");
110 if (processed.empty()) {
111 spdlog::debug("Input expression is empty.");
112 return ""; // Handle empty input early
113 }
114
115 // --- Helper function: build log string ---
116 auto build_log_string = [](const auto &container) {
117 std::ostringstream oss;
118 for (auto it = container.begin(); it != container.end(); ++it) {
119 oss << *it;
120 if (std::next(it) != container.end()) {
121 oss << ", ";
122 }
123 }
124 return oss.str();
125 };
126
127 // 1. Handle '!' for NOT carefully BEFORE adding general spaces.
128 // Replace logical NOT '!' with " not ", ensuring not to replace '!='.
129 std::string temp_processed;
130 temp_processed.reserve(processed.length() * 1.2); // Pre-allocate a bit more space
131 for (size_t i = 0; i < processed.length(); ++i) {
132 if (processed[i] == '!') {
133 if (i + 1 < processed.length() && processed[i + 1] == '=') {
134 temp_processed += "!="; // Keep inequality operator
135 i++; // Skip the '='
136 } else {
137 temp_processed += " not "; // Replace logical NOT with spaced operator
138 }
139 } else {
140 temp_processed += processed[i];
141 }
142 }
143 processed = temp_processed;
144 spdlog::trace("After '!' to 'not' conversion: {}", processed);
145
146 // 2. Add spaces around other operators and parentheses that need them
147 processed = std::regex_replace(processed, std::regex("\\("), " ( ");
148 processed = std::regex_replace(processed, std::regex("\\)"), " ) ");
149 processed = std::regex_replace(processed, std::regex("\\+"), " + "); // OR
150 processed = std::regex_replace(processed, std::regex("\\^"), " ^ "); // XOR
151 processed = std::regex_replace(processed, std::regex("\\*"), " * "); // AND
152 // Consolidate multiple spaces into one and trim again
153 processed = std::regex_replace(processed, std::regex("\\s+"), " ");
154 processed = std::regex_replace(processed, std::regex("^\\s+|\\s+$"), "");
155 spdlog::trace("After adding spaces: {}", processed);
156
157 // 3. Replace symbolic operators with their keyword equivalents
158 // Note: 'not' is already handled.
159 processed = std::regex_replace(processed, std::regex(" \\+ "), " or ");
160 processed = std::regex_replace(processed, std::regex(" \\^ "), " xor ");
161 processed = std::regex_replace(processed, std::regex(" \\* "), " and ");
162 spdlog::trace("After operator keyword replacement: {}", processed);
163
164 // 4. Tokenize based on spaces
165 std::stringstream ss(processed);
166 // Read tokens skipping whitespace
167 std::vector<std::string> tokens{std::istream_iterator<std::string>(ss),
168 std::istream_iterator<std::string>()};
169
170 if (tokens.empty()) {
171 spdlog::debug("No tokens found after tokenization.");
172 return "";
173 }
174 spdlog::trace("Tokens after initial processing: [{}]", build_log_string(tokens));
175
176 // 5. Insert implied 'and'
177 std::vector<std::string> tokens_with_implied_and;
178 if (!tokens.empty()) {
179 tokens_with_implied_and.push_back(tokens[0]);
180 for (size_t i = 0; i < tokens.size() - 1; ++i) {
181 const std::string &current_token = tokens[i];
182 const std::string &next_token = tokens[i + 1];
183
184 // An operand ends if it's an identifier or a closing parenthesis.
185 bool current_ends_operand = isIdentifier(current_token) || current_token == ")";
186 // An operand starts if it's an identifier, an opening parenthesis, or 'not'.
187 bool next_starts_operand =
188 isIdentifier(next_token) || next_token == "(" || next_token == "not";
189
190 // We should insert 'and' if the current token ends an operand AND the next token starts one,
191 // UNLESS the current token is already an operator/opening bracket OR the next token is an
192 // operator/closing bracket. Explicit check: Don't insert 'and' if an operator already exists
193 // between them.
194 bool current_is_op_or_open =
195 isOperator(current_token) || current_token == "not" || current_token == "(";
196 bool next_is_op_or_close = isOperator(next_token) || next_token == "not" || next_token == ")";
197
198 if (current_ends_operand && next_starts_operand && !current_is_op_or_open &&
199 !next_is_op_or_close) {
200 spdlog::trace("Inserting implied 'and' between '{}' and '{}'", current_token, next_token);
201 tokens_with_implied_and.push_back("and");
202 }
203
204 tokens_with_implied_and.push_back(next_token); // Add the next token regardless
205 }
206 } else {
207 // Handle case where initial tokenization yielded no tokens
208 return "";
209 }
210
211 spdlog::trace("Tokens after implied 'and': [{}]", build_log_string(tokens_with_implied_and));
212
213 // 6. Handle 'not Identifier' ONLY during reconstruction (NEW LOGIC)
214 std::vector<std::string> final_tokens;
215 for (size_t i = 0; i < tokens_with_implied_and.size(); ++i) {
216 const std::string &current_token = tokens_with_implied_and[i];
217
218 if (current_token == "not") {
219 // Check if the *next* token is an Identifier
220 if (i + 1 < tokens_with_implied_and.size() && isIdentifier(tokens_with_implied_and[i + 1])) {
221 // Found 'not Identifier', transform to 'not(Identifier)'
222 final_tokens.push_back("not");
223 final_tokens.push_back("(");
224 final_tokens.push_back(tokens_with_implied_and[i + 1]); // The identifier
225 final_tokens.push_back(")");
226 i++; // Increment i to skip the identifier we just processed
227 } else {
228 // 'not' is followed by '(', another operator, or is at the end.
229 // Add 'not' as is, let ExprTk parse 'not(...)' or handle errors.
230 final_tokens.push_back("not");
231 }
232 } else {
233 // Token is not 'not', just add it to the final list
234 final_tokens.push_back(current_token);
235 }
236 }
237 spdlog::trace("Tokens after 'not' parenthesis handling: [{}]", build_log_string(final_tokens));
238
239 // 7. Reconstruct the final expression string from tokens
240 std::string final_expr;
241 if (!final_tokens.empty()) {
242 final_expr = final_tokens[0];
243 for (size_t i = 1; i < final_tokens.size(); ++i) {
244 const std::string &prev = final_tokens[i - 1];
245 const std::string &curr = final_tokens[i];
246
247 // Add space smartly: No space after '(', no space before ')',
248 // no space after 'not' if followed by '(', no space before ',' (in function args, if
249 // applicable)
250 bool add_space = true;
251 if (prev == "(" || curr == ")" || curr == ",") {
252 add_space = false;
253 }
254 if (prev == "not" && curr == "(") {
255 add_space = false; // Already handled by 'not(...)' structure
256 }
257 // Potentially add more rules if needed for other operators/functions
258
259 if (add_space) {
260 final_expr += " ";
261 }
262 final_expr += curr;
263 }
264 }
265
266 // 8. Final cleanup (consolidate any remaining multiple spaces and trim)
267 final_expr = std::regex_replace(final_expr, std::regex("\\s+"), " ");
268 final_expr = std::regex_replace(final_expr, std::regex("^\\s+|\\s+$"), "");
269
270 spdlog::debug("Preprocessed expression result: {}", final_expr);
271 return final_expr;
272}

529void LogicComparator::compareSingleExpressionPair(const std::string &ref_expression_processed,
530 const std::string &comp_expression_processed,
531 const std::vector<std::string> &sorted_vars,
532 PinComparisonResult &result) {
533 // Store processed expressions in the result struct
534 result.ref_expr_processed = ref_expression_processed;
535 result.comp_expr_processed = comp_expression_processed;
536 result.comparison_possible = true; // Assume comparison is possible initially
537
538 spdlog::debug("Comparing processed expressions for pin '{}':", result.pin_name);
539 spdlog::debug(" Ref : {}", ref_expression_processed);
540 spdlog::debug(" Comp: {}", comp_expression_processed);
541
542 // --- ExprTk Setup ---
543 typedef exprtk::symbol_table<double> symbol_table_t;
544 typedef exprtk::expression<double> expression_t;
545 typedef exprtk::parser<double> parser_t;
546
547 // Setup for Reference Expression
548 symbol_table_t ref_symbol_table;
549 expression_t ref_expression;
550 parser_t ref_parser;
551
552 for (const auto &var_name : sorted_vars) {
553 // Create variable, ExprTk manages its memory within the table
554 if (!ref_symbol_table.create_variable(var_name)) {
555 result.error_message = "Failed to create variable '" + var_name + "' in ref symbol table.";
556 spdlog::error(result.error_message);
557 result.comparison_possible = false;
558 return; // Cannot proceed
559 }
560 }
561 ref_expression.register_symbol_table(ref_symbol_table);
562
563 // Compile Reference Expression
564 result.ref_compiles = ref_parser.compile(ref_expression_processed, ref_expression);
565 if (!result.ref_compiles) {
566 result.error_message = "Reference expression compilation failed: " + ref_parser.error();
567 spdlog::warn("Pin '{}': {}", result.pin_name, result.error_message);
568 result.comparison_possible = false;
569 // Don't return yet, maybe the comparison one also fails
570 } else {
571 spdlog::debug("Pin '{}': Reference expression compiled successfully.", result.pin_name);
572 }
573
574 // Setup for Comparison Expression
575 symbol_table_t comp_symbol_table;
576 expression_t comp_expression;
577 parser_t comp_parser;
578
579 for (const auto &var_name : sorted_vars) {
580 if (!comp_symbol_table.create_variable(var_name)) {
581 // Append error if ref also failed, otherwise set it
582 std::string comp_err = "Failed to create variable '" + var_name + "' in comp symbol table.";
583 result.error_message += (result.error_message.empty() ? "" : "\n") + comp_err;
584 spdlog::error(comp_err);
585 result.comparison_possible = false; // Mark as impossible now
586 return; // Cannot proceed if variable creation fails
587 }
588 }
589 comp_expression.register_symbol_table(comp_symbol_table);
590
591 // Compile Comparison Expression
592 result.comp_compiles = comp_parser.compile(comp_expression_processed, comp_expression);
593 if (!result.comp_compiles) {
594 std::string comp_err = "Comparison expression compilation failed: " + comp_parser.error();
595 result.error_message += (result.error_message.empty() ? "" : "\n") + comp_err;
596 spdlog::warn("Pin '{}': {}", result.pin_name, comp_err);
597 result.comparison_possible = false; // Mark as impossible if not already
598 } else {
599 spdlog::debug("Pin '{}': Comparison expression compiled successfully.", result.pin_name);
600 }
601
602 // If either failed to compile, comparison is not possible
603 if (!result.ref_compiles || !result.comp_compiles) {
604 result.comparison_possible = false;
605 return;
606 }
607
608 // --- Truth Table Generation and Comparison ---
609 size_t num_vars = sorted_vars.size();
610 // Use unsigned long long for potentially large number of combinations
611 unsigned long long num_combinations = 1ULL << num_vars;
612 // Prevent excessively large tables (e.g., > 20 variables is 1M+ rows)
613 const unsigned long long MAX_COMBINATIONS = 1ULL << 20; // Limit to 2^20 combinations
614 if (num_vars > 20) { // Check against var count for clarity
615 result.error_message = "Too many variables (" + std::to_string(num_vars) +
616 "). Max supported for truth table is 20.";
617 spdlog::error("Pin '{}': {}", result.pin_name, result.error_message);
618 result.comparison_possible = false;
619 return;
620 }
621 if (num_combinations == 0 && num_vars > 0) { // Overflow check
622 result.error_message = "Number of combinations calculation overflowed for " +
623 std::to_string(num_vars) + " variables.";
624 spdlog::error("Pin '{}': {}", result.pin_name, result.error_message);
625 result.comparison_possible = false;
626 return;
627 }
628
629 std::vector<bool> ref_results;
630 std::vector<bool> comp_results;
631 ref_results.reserve(static_cast<size_t>(num_combinations)); // Avoid reallocations
632 comp_results.reserve(static_cast<size_t>(num_combinations));
633
634 // --- Prepare Tabulate Tables ---
635 Table ref_table;
636 Table comp_table;
637 Table::Row_t header_row;
638 header_row.push_back("#"); // Row number column
639 for (const auto &var_name : sorted_vars) {
640 header_row.push_back(var_name);
641 }
642 header_row.push_back(ref_expression_processed); // Reference expression as last column header
643 ref_table.add_row(header_row);
644
645 // Adjust header for comparison table
646 header_row.back() = comp_expression_processed; // Change last header to comparison expression
647 comp_table.add_row(header_row);
648
649 // Format header rows
650 for (size_t i = 0; i < ref_table[0].size(); ++i) {
651 ref_table[0][i].format().font_color(Color::yellow).font_style({FontStyle::bold});
652 comp_table[0][i].format().font_color(Color::yellow).font_style({FontStyle::bold});
653 }
654
655 spdlog::debug("Pin '{}': Generating truth table with {} variables ({} combinations)...",
656 result.pin_name, num_vars, num_combinations);
657
658 bool evaluation_error = false;
659 for (unsigned long long i = 0; i < num_combinations; ++i) {
660 Table::Row_t ref_data_row;
661 Table::Row_t comp_data_row;
662 ref_data_row.push_back(std::to_string(i));
663 comp_data_row.push_back(std::to_string(i));
664
665 // Set variable values for this combination
666 for (size_t j = 0; j < num_vars; ++j) {
667 // The j-th bit of i determines the value of the j-th variable
668 bool bit_value = ((i >> j) & 1ULL);
669 double var_value = bit_value ? double(1.0) : double(0.0); // ExprTk uses floating point
670
671 // Get variable reference and assign value
672 // Need error checking here in theory, but create_variable should have caught issues
673 ref_symbol_table.get_variable(sorted_vars[j])->ref() = var_value;
674 comp_symbol_table.get_variable(sorted_vars[j])->ref() = var_value;
675
676 // Add input value to table rows (as string "0" or "1")
677 std::string bit_str = bit_value ? "1" : "0";
678 ref_data_row.push_back(bit_str);
679 comp_data_row.push_back(bit_str);
680 }
681
682 // Evaluate expressions
683 double ref_val, comp_val;
684 try {
685 ref_val = ref_expression.value();
686 } catch (const std::exception &e) {
687 result.error_message +=
688 "\nReference evaluation failed at combination " + std::to_string(i) + ": " + e.what();
689 spdlog::error("Pin '{}': {}", result.pin_name, result.error_message);
690 result.comparison_possible = false;
691 evaluation_error = true;
692 break; // Stop evaluation
693 }
694 try {
695 comp_val = comp_expression.value();
696 } catch (const std::exception &e) {
697 result.error_message +=
698 "\nComparison evaluation failed at combination " + std::to_string(i) + ": " + e.what();
699 spdlog::error("Pin '{}': {}", result.pin_name, result.error_message);
700 result.comparison_possible = false;
701 evaluation_error = true;
702 break; // Stop evaluation
703 }
704
705 // Store boolean results (commonly, non-zero is true in logic contexts)
706 bool ref_bool_result = (ref_val != double(0.0));
707 bool comp_bool_result = (comp_val != double(0.0));
708
709 ref_results.push_back(ref_bool_result);
710 comp_results.push_back(comp_bool_result);
711
712 // Add output results to table rows (as string "0" or "1")
713 ref_data_row.push_back(ref_bool_result ? "1" : "0");
714 comp_data_row.push_back(comp_bool_result ? "1" : "0");
715
716 // Add rows to tables
717 ref_table.add_row(ref_data_row);
718 comp_table.add_row(comp_data_row);
719
720 } // End of combination loop
721
722 if (evaluation_error) {
723 return; // Don't proceed if evaluation failed
724 }
725
726 // --- Final Comparison ---
727 if (result.comparison_possible) {
728 result.are_equivalent = (ref_results == comp_results);
729 if (result.are_equivalent) {
730 spdlog::info("Pin '{}': Expressions ARE logically equivalent.", result.pin_name);
731 } else {
732 spdlog::warn("Pin '{}': Expressions ARE NOT logically equivalent.", result.pin_name);
733 result.error_message +=
734 "\nTruth table outputs differ."; // Add specific reason for non-equivalence
735 }
736
737 // Store the generated tables in the result struct
738 result.ref_truth_table = ref_table;
739 result.comp_truth_table = comp_table;
740 }

788void LogicComparator::compareCellLogic() {
789
790 // 1. Collect all unique pin names from both maps
791 std::set<std::string> unique_pin_names;
792 for (const auto &pair : ref_outpin_map_) {
793 unique_pin_names.insert(pair.first);
794 }
795 for (const auto &pair : comp_outpin_map_) {
796 unique_pin_names.insert(pair.first);
797 }
798
799 spdlog::info("Starting logic comparison for cell '{}' with {} unique output pins...", cell_name_,
800 unique_pin_names.size());
801
802 // 2. Iterate through each unique pin name
803 for (const std::string &pin_name : unique_pin_names) {
804 PinComparisonResult pin_result;
805 pin_result.pin_name = pin_name;
806 pin_result.comparison_possible = true; // Assume possible initially
807
808 // 3. Get raw expressions
809 auto ref_it = ref_outpin_map_.find(pin_name);
810 auto comp_it = comp_outpin_map_.find(pin_name);
811
812 if (ref_it == ref_outpin_map_.end()) {
813 pin_result.error_message = "Pin not found in reference map.";
814 spdlog::warn("Cell '{}', Pin '{}': {}", cell_name_, pin_name, pin_result.error_message);
815 pin_result.comparison_possible = false;
816 // Still try to get the comparison expression for reporting
817 if (comp_it != comp_outpin_map_.end()) {
818 pin_result.comp_expr_raw = comp_it->second;
819 }
820 } else {
821 pin_result.ref_expr_raw = ref_it->second;
822 spdlog::debug("Pin -> Expression: {} -> {}", pin_name, pin_result.ref_expr_raw);
823 }
824
825 if (comp_it == comp_outpin_map_.end()) {
826 std::string comp_err = "Pin not found in comparison map.";
827 pin_result.error_message += (pin_result.error_message.empty() ? "" : "\n") + comp_err;
828 spdlog::warn("Cell '{}', Pin '{}': {}", cell_name_, pin_name, comp_err);
829 pin_result.comparison_possible = false;
830 // If ref existed, store it
831 if (ref_it != ref_outpin_map_.end()) {
832 pin_result.ref_expr_raw = ref_it->second; // Already stored if ref exists
833 }
834 } else {
835 pin_result.comp_expr_raw = comp_it->second;
836 spdlog::debug("Pin => Expression: {} => {}", pin_name, pin_result.comp_expr_raw);
837 }
838
839 // If pin missing in either, we can't compare, but store result and continue
840 if (!pin_result.comparison_possible) {
841 all_pin_results_[pin_name] = pin_result;
842 continue;
843 }
844
845 // 4. Preprocess expressions
846 std::string ref_processed = preprocessExpression(pin_result.ref_expr_raw);
847 std::string comp_processed = preprocessExpression(pin_result.comp_expr_raw);
848
849 if (ref_processed.empty()) {
850 pin_result.error_message += "\nReference expression became empty after preprocessing.";
851 spdlog::warn("Cell '{}', Pin '{}': {}", cell_name_, pin_name,
852 "Reference expression became empty after preprocessing.");
853 pin_result.comparison_possible = false;
854 }
855 if (comp_processed.empty()) {
856 std::string comp_err = "\nComparison expression became empty after preprocessing.";
857 pin_result.error_message += (pin_result.error_message.empty() ? "" : "\n") + comp_err;
858 spdlog::warn("Cell '{}', Pin '{}': {}", cell_name_, pin_name,
859 "Comparison expression became empty after preprocessing.");
860 pin_result.comparison_possible = false; // Mark as impossible if not already
861 }
862 // If preprocessing failed for either, store and continue
863 if (ref_processed.empty() || comp_processed.empty()) {
864 pin_result.ref_expr_processed = ref_processed; // Store possibly empty strings
865 pin_result.comp_expr_processed = comp_processed;
866 all_pin_results_[pin_name] = pin_result;
867 continue;
868 }
869
870 // 5. Extract and validate variables
871 std::vector<std::string> sorted_vars;
872 // Pass RAW expressions to extractVariables as it uses regex on original format
873 bool variables_match =
874 extractVariables(pin_result.ref_expr_raw, pin_result.comp_expr_raw, sorted_vars);
875
876 if (!variables_match) {
877 pin_result.error_message = "Variable sets do not match between expressions.";
878 // extractVariables already logs details
879 pin_result.comparison_possible = false;
880 pin_result.ref_expr_processed = ref_processed; // Store processed even if vars mismatch
881 pin_result.comp_expr_processed = comp_processed;
882 all_pin_results_[pin_name] = pin_result;
883 continue;
884 }
885 // --- Helper function: build log string ---
886 auto build_log_string = [](const auto &container) {
887 std::ostringstream oss;
888 for (auto it = container.begin(); it != container.end(); ++it) {
889 oss << *it;
890 if (std::next(it) != container.end()) {
891 oss << ", ";
892 }
893 }
894 return oss.str();
895 };
896 spdlog::info("Pin '{}': Variable sets match. Found unique variables: {}", pin_name,
897 build_log_string(sorted_vars));
898
899 // 6. Compare the single pair using the processed expressions
900 // Pass pin_result by reference - it will be populated by the function
901 compareSingleExpressionPair(ref_processed, comp_processed, sorted_vars, pin_result);
902
903 // 7. Store the detailed result for this pin
904 all_pin_results_[pin_name] = pin_result;
905
906 } // End of pin loop
907
908 spdlog::info("Logic comparison finished for cell '{}'.", cell_name_);
909}
