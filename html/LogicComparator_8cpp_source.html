74bool isIdentifier(const std::string &token) {
75 if (token.empty())
76 return false;
77 // Check if the token is a valid identifier (upper alpha+ numeric + underscore)
78 bool has_alpha = false;
79 for (char c : token) {
80 if (std::isalpha(c)) {
81 has_alpha = true;
82 } else if (!std::isalnum(c) && c != '_') {
83 return false; // Contains invalid char
84 }
85 }
86 // Must start with an uppercase alphabetic character
87 return has_alpha && std::isupper(token[0]);
88}

128std::string LogicComparator::preprocessExpression(const std::string &input_expr) {
129 std::string processed = input_expr;
130 spdlog::debug("Preprocessing raw expression: {}", processed);
131
132 // 0. Trim leading/trailing whitespace first
133 processed = std::regex_replace(processed, std::regex("^\\s+|\\s+$"), "");
134 if (processed.empty()) {
135 spdlog::debug("Input expression is empty.");
136 return ""; // Handle empty input early
137 }
138
139 // --- Helper function: build log string ---
140 auto build_log_string = [](const auto &container) {
141 std::ostringstream oss;
142 for (auto it = container.begin(); it != container.end(); ++it) {
143 oss << *it;
144 if (std::next(it) != container.end()) {
145 oss << ", ";
146 }
147 }
148 return oss.str();
149 };
150
151 // 1. Handle '!' for NOT carefully BEFORE adding general spaces.
152 // Replace logical NOT '!' with " not ", ensuring not to replace '!='.
153 std::string temp_processed;
154 temp_processed.reserve(processed.length() * 1.2); // Pre-allocate a bit more space
155 for (size_t i = 0; i < processed.length(); ++i) {
156 if (processed[i] == '!') {
157 if (i + 1 < processed.length() && processed[i + 1] == '=') {
158 temp_processed += "!="; // Keep inequality operator
159 i++; // Skip the '='
160 } else {
161 temp_processed += " not "; // Replace logical NOT with spaced operator
162 }
163 } else {
164 temp_processed += processed[i];
165 }
166 }
167 processed = temp_processed;
168 spdlog::trace("After '!' to 'not' conversion: {}", processed);
169
170 // 2. Add spaces around other operators and parentheses that need them
171 processed = std::regex_replace(processed, std::regex("\\("), " ( ");
172 processed = std::regex_replace(processed, std::regex("\\)"), " ) ");
173 processed = std::regex_replace(processed, std::regex("\\+"), " + "); // OR
174 processed = std::regex_replace(processed, std::regex("\\^"), " ^ "); // XOR
175 processed = std::regex_replace(processed, std::regex("\\*"), " * "); // AND
176 // Consolidate multiple spaces into one and trim again
177 processed = std::regex_replace(processed, std::regex("\\s+"), " ");
178 processed = std::regex_replace(processed, std::regex("^\\s+|\\s+$"), "");
179 spdlog::trace("After adding spaces: {}", processed);
180
181 // 3. Replace symbolic operators with their keyword equivalents
182 // Note: 'not' is already handled.
183 processed = std::regex_replace(processed, std::regex(" \\+ "), " or ");
184 processed = std::regex_replace(processed, std::regex(" \\^ "), " xor ");
185 processed = std::regex_replace(processed, std::regex(" \\* "), " and ");
186 spdlog::trace("After operator keyword replacement: {}", processed);
187
188 // 4. Tokenize based on spaces
189 std::stringstream ss(processed);
190 // Read tokens skipping whitespace
191 std::vector<std::string> tokens{std::istream_iterator<std::string>(ss),
192 std::istream_iterator<std::string>()};
193
194 if (tokens.empty()) {
195 spdlog::debug("No tokens found after tokenization.");
196 return "";
197 }
198 spdlog::trace("Tokens after initial processing: [{}]", build_log_string(tokens));
199
200 // 5. Insert implied 'and'
201 std::vector<std::string> tokens_with_implied_and;
202 if (!tokens.empty()) {
203 tokens_with_implied_and.push_back(tokens[0]);
204 for (size_t i = 0; i < tokens.size() - 1; ++i) {
205 const std::string &current_token = tokens[i];
206 const std::string &next_token = tokens[i + 1];
207
208 // An operand ends if it's an identifier or a closing parenthesis.
209 bool current_ends_operand = isIdentifier(current_token) || current_token == ")";
210 // An operand starts if it's an identifier, an opening parenthesis, or 'not'.
211 bool next_starts_operand =
212 isIdentifier(next_token) || next_token == "(" || next_token == "not";
213
214 // We should insert 'and' if the current token ends an operand AND the next token starts one,
215 // UNLESS the current token is already an operator/opening bracket OR the next token is an
216 // operator/closing bracket. Explicit check: Don't insert 'and' if an operator already exists
217 // between them.
218 bool current_is_op_or_open =
219 isOperator(current_token) || current_token == "not" || current_token == "(";
220 bool next_is_op_or_close = isOperator(next_token) || next_token == "not" || next_token == ")";
221
222 if (current_ends_operand && next_starts_operand && !current_is_op_or_open &&
223 !next_is_op_or_close) {
224 spdlog::trace("Inserting implied 'and' between '{}' and '{}'", current_token, next_token);
225 tokens_with_implied_and.push_back("and");
226 }
227
228 tokens_with_implied_and.push_back(next_token); // Add the next token regardless
229 }
230 } else {
231 // Handle case where initial tokenization yielded no tokens
232 return "";
233 }
234
235 spdlog::trace("Tokens after implied 'and': [{}]", build_log_string(tokens_with_implied_and));
236
237 // 6. Handle 'not Identifier' ONLY during reconstruction (NEW LOGIC)
238 std::vector<std::string> final_tokens;
239 for (size_t i = 0; i < tokens_with_implied_and.size(); ++i) {
240 const std::string &current_token = tokens_with_implied_and[i];
241
242 if (current_token == "not") {
243 // Check if the *next* token is an Identifier
244 if (i + 1 < tokens_with_implied_and.size() && isIdentifier(tokens_with_implied_and[i + 1])) {
245 // Found 'not Identifier', transform to 'not(Identifier)'
246 final_tokens.push_back("not");
247 final_tokens.push_back("(");
248 final_tokens.push_back(tokens_with_implied_and[i + 1]); // The identifier
249 final_tokens.push_back(")");
250 i++; // Increment i to skip the identifier we just processed
251 } else {
252 // 'not' is followed by '(', another operator, or is at the end.
253 // Add 'not' as is, let ExprTk parse 'not(...)' or handle errors.
254 final_tokens.push_back("not");
255 }
256 } else {
257 // Token is not 'not', just add it to the final list
258 final_tokens.push_back(current_token);
259 }
260 }
261 spdlog::trace("Tokens after 'not' parenthesis handling: [{}]", build_log_string(final_tokens));
262
263 // 7. Reconstruct the final expression string from tokens
264 std::string final_expr;
265 if (!final_tokens.empty()) {
266 final_expr = final_tokens[0];
267 for (size_t i = 1; i < final_tokens.size(); ++i) {
268 const std::string &prev = final_tokens[i - 1];
269 const std::string &curr = final_tokens[i];
270
271 // Add space smartly: No space after '(', no space before ')',
272 // no space after 'not' if followed by '(', no space before ',' (in function args, if
273 // applicable)
274 bool add_space = true;
275 if (prev == "(" || curr == ")" || curr == ",") {
276 add_space = false;
277 }
278 if (prev == "not" && curr == "(") {
279 add_space = false; // Already handled by 'not(...)' structure
280 }
281 // Potentially add more rules if needed for other operators/functions
282
283 if (add_space) {
284 final_expr += " ";
285 }
286 final_expr += curr;
287 }
288 }
289
290 // 8. Final cleanup (consolidate any remaining multiple spaces and trim)
291 final_expr = std::regex_replace(final_expr, std::regex("\\s+"), " ");
292 final_expr = std::regex_replace(final_expr, std::regex("^\\s+|\\s+$"), "");
293
294 spdlog::debug("Preprocessed expression result: {}", final_expr);
295 return final_expr;
296}

553void LogicComparator::compareSingleExpressionPair(const std::string &ref_expression_processed,
554 const std::string &comp_expression_processed,
555 const std::vector<std::string> &sorted_vars,
556 PinComparisonResult &result) {
557 // Store processed expressions in the result struct
558 result.ref_expr_processed = ref_expression_processed;
559 result.comp_expr_processed = comp_expression_processed;
560 result.comparison_possible = true; // Assume comparison is possible initially
561
562 spdlog::debug("Comparing processed expressions for pin '{}':", result.pin_name);
563 spdlog::debug(" Ref : {}", ref_expression_processed);
564 spdlog::debug(" Comp: {}", comp_expression_processed);
565
566 // --- ExprTk Setup ---
567 typedef exprtk::symbol_table<double> symbol_table_t;
568 typedef exprtk::expression<double> expression_t;
569 typedef exprtk::parser<double> parser_t;
570
571 // Setup for Reference Expression
572 symbol_table_t ref_symbol_table;
573 expression_t ref_expression;
574 parser_t ref_parser;
575
576 for (const auto &var_name : sorted_vars) {
577 // Create variable, ExprTk manages its memory within the table
578 if (!ref_symbol_table.create_variable(var_name)) {
579 result.error_message = "Failed to create variable '" + var_name + "' in ref symbol table.";
580 spdlog::error(result.error_message);
581 result.comparison_possible = false;
582 return; // Cannot proceed
583 }
584 }
585 ref_expression.register_symbol_table(ref_symbol_table);
586
587 // Compile Reference Expression
588 result.ref_compiles = ref_parser.compile(ref_expression_processed, ref_expression);
589 if (!result.ref_compiles) {
590 result.error_message = "Reference expression compilation failed: " + ref_parser.error();
591 spdlog::warn("Pin '{}': {}", result.pin_name, result.error_message);
592 result.comparison_possible = false;
593 // Don't return yet, maybe the comparison one also fails
594 } else {
595 spdlog::debug("Pin '{}': Reference expression compiled successfully.", result.pin_name);
596 }
597
598 // Setup for Comparison Expression
599 symbol_table_t comp_symbol_table;
600 expression_t comp_expression;
601 parser_t comp_parser;
602
603 for (const auto &var_name : sorted_vars) {
604 if (!comp_symbol_table.create_variable(var_name)) {
605 // Append error if ref also failed, otherwise set it
606 std::string comp_err = "Failed to create variable '" + var_name + "' in comp symbol table.";
607 result.error_message += (result.error_message.empty() ? "" : "\n") + comp_err;
608 spdlog::error(comp_err);
609 result.comparison_possible = false; // Mark as impossible now
610 return; // Cannot proceed if variable creation fails
611 }
612 }
613 comp_expression.register_symbol_table(comp_symbol_table);
614
615 // Compile Comparison Expression
616 result.comp_compiles = comp_parser.compile(comp_expression_processed, comp_expression);
617 if (!result.comp_compiles) {
618 std::string comp_err = "Comparison expression compilation failed: " + comp_parser.error();
619 result.error_message += (result.error_message.empty() ? "" : "\n") + comp_err;
620 spdlog::warn("Pin '{}': {}", result.pin_name, comp_err);
621 result.comparison_possible = false; // Mark as impossible if not already
622 } else {
623 spdlog::debug("Pin '{}': Comparison expression compiled successfully.", result.pin_name);
624 }
625
626 // If either failed to compile, comparison is not possible
627 if (!result.ref_compiles || !result.comp_compiles) {
628 result.comparison_possible = false;
629 return;
630 }
631
632 // --- Truth Table Generation and Comparison ---
633 size_t num_vars = sorted_vars.size();
634 // Use unsigned long long for potentially large number of combinations
635 unsigned long long num_combinations = 1ULL << num_vars;
636 // Prevent excessively large tables (e.g., > 20 variables is 1M+ rows)
637 const unsigned long long MAX_COMBINATIONS = 1ULL << 20; // Limit to 2^20 combinations
638 if (num_vars > 20) { // Check against var count for clarity
639 result.error_message = "Too many variables (" + std::to_string(num_vars) +
640 "). Max supported for truth table is 20.";
641 spdlog::error("Pin '{}': {}", result.pin_name, result.error_message);
642 result.comparison_possible = false;
643 return;
644 }
645 if (num_combinations == 0 && num_vars > 0) { // Overflow check
646 result.error_message = "Number of combinations calculation overflowed for " +
647 std::to_string(num_vars) + " variables.";
648 spdlog::error("Pin '{}': {}", result.pin_name, result.error_message);
649 result.comparison_possible = false;
650 return;
651 }
652
653 std::vector<bool> ref_results;
654 std::vector<bool> comp_results;
655 ref_results.reserve(static_cast<size_t>(num_combinations)); // Avoid reallocations
656 comp_results.reserve(static_cast<size_t>(num_combinations));
657
658 // --- Prepare Tabulate Tables ---
659 Table ref_table;
660 Table comp_table;
661 Table::Row_t header_row;
662 header_row.push_back("#"); // Row number column
663 for (const auto &var_name : sorted_vars) {
664 header_row.push_back(var_name);
665 }
666 header_row.push_back(ref_expression_processed); // Reference expression as last column header
667 ref_table.add_row(header_row);
668
669 // Adjust header for comparison table
670 header_row.back() = comp_expression_processed; // Change last header to comparison expression
671 comp_table.add_row(header_row);
672
673 // Format header rows
674 for (size_t i = 0; i < ref_table[0].size(); ++i) {
675 ref_table[0][i].format().font_color(Color::yellow).font_style({FontStyle::bold});
676 comp_table[0][i].format().font_color(Color::yellow).font_style({FontStyle::bold});
677 }
678
679 spdlog::debug("Pin '{}': Generating truth table with {} variables ({} combinations)...",
680 result.pin_name, num_vars, num_combinations);
681
682 bool evaluation_error = false;
683 for (unsigned long long i = 0; i < num_combinations; ++i) {
684 Table::Row_t ref_data_row;
685 Table::Row_t comp_data_row;
686 ref_data_row.push_back(std::to_string(i));
687 comp_data_row.push_back(std::to_string(i));
688
689 // Set variable values for this combination
690 for (size_t j = 0; j < num_vars; ++j) {
691 // The j-th bit of i determines the value of the j-th variable
692 bool bit_value = ((i >> j) & 1ULL);
693 double var_value = bit_value ? double(1.0) : double(0.0); // ExprTk uses floating point
694
695 // Get variable reference and assign value
696 // Need error checking here in theory, but create_variable should have caught issues
697 ref_symbol_table.get_variable(sorted_vars[j])->ref() = var_value;
698 comp_symbol_table.get_variable(sorted_vars[j])->ref() = var_value;
699
700 // Add input value to table rows (as string "0" or "1")
701 std::string bit_str = bit_value ? "1" : "0";
702 ref_data_row.push_back(bit_str);
703 comp_data_row.push_back(bit_str);
704 }
705
706 // Evaluate expressions
707 double ref_val, comp_val;
708 try {
709 ref_val = ref_expression.value();
710 } catch (const std::exception &e) {
711 result.error_message +=
712 "\nReference evaluation failed at combination " + std::to_string(i) + ": " + e.what();
713 spdlog::error("Pin '{}': {}", result.pin_name, result.error_message);
714 result.comparison_possible = false;
715 evaluation_error = true;
716 break; // Stop evaluation
717 }
718 try {
719 comp_val = comp_expression.value();
720 } catch (const std::exception &e) {
721 result.error_message +=
722 "\nComparison evaluation failed at combination " + std::to_string(i) + ": " + e.what();
723 spdlog::error("Pin '{}': {}", result.pin_name, result.error_message);
724 result.comparison_possible = false;
725 evaluation_error = true;
726 break; // Stop evaluation
727 }
728
729 // Store boolean results (commonly, non-zero is true in logic contexts)
730 bool ref_bool_result = (ref_val != double(0.0));
731 bool comp_bool_result = (comp_val != double(0.0));
732
733 ref_results.push_back(ref_bool_result);
734 comp_results.push_back(comp_bool_result);
735
736 // Add output results to table rows (as string "0" or "1")
737 ref_data_row.push_back(ref_bool_result ? "1" : "0");
738 comp_data_row.push_back(comp_bool_result ? "1" : "0");
739
740 // Add rows to tables
741 ref_table.add_row(ref_data_row);
742 comp_table.add_row(comp_data_row);
743
744 } // End of combination loop
745
746 if (evaluation_error) {
747 return; // Don't proceed if evaluation failed
748 }
749
750 // --- Final Comparison ---
751 if (result.comparison_possible) {
752 result.are_equivalent = (ref_results == comp_results);
753 if (result.are_equivalent) {
754 spdlog::info("Pin '{}': Expressions ARE logically equivalent.", result.pin_name);
755 } else {
756 spdlog::warn("Pin '{}': Expressions ARE NOT logically equivalent.", result.pin_name);
757 result.error_message +=
758 "\nTruth table outputs differ."; // Add specific reason for non-equivalence
759 }
760
761 // Store the generated tables in the result struct
762 result.ref_truth_table = ref_table;
763 result.comp_truth_table = comp_table;
764 }

812void LogicComparator::compareCellLogic() {
813
814 // 1. Collect all unique pin names from both maps
815 std::set<std::string> unique_pin_names;
816 for (const auto &pair : ref_outpin_map_) {
817 unique_pin_names.insert(pair.first);
818 }
819 for (const auto &pair : comp_outpin_map_) {
820 unique_pin_names.insert(pair.first);
821 }
822
823 spdlog::info("Starting logic comparison for cell '{}' with {} unique output pins...", cell_name_,
824 unique_pin_names.size());
825
826 // 2. Iterate through each unique pin name
827 for (const std::string &pin_name : unique_pin_names) {
828 PinComparisonResult pin_result;
829 pin_result.pin_name = pin_name;
830 pin_result.comparison_possible = true; // Assume possible initially
831
832 // 3. Get raw expressions
833 auto ref_it = ref_outpin_map_.find(pin_name);
834 auto comp_it = comp_outpin_map_.find(pin_name);
835
836 if (ref_it == ref_outpin_map_.end()) {
837 pin_result.error_message = "Pin not found in reference map.";
838 spdlog::warn("Cell '{}', Pin '{}': {}", cell_name_, pin_name, pin_result.error_message);
839 pin_result.comparison_possible = false;
840 // Still try to get the comparison expression for reporting
841 if (comp_it != comp_outpin_map_.end()) {
842 pin_result.comp_expr_raw = comp_it->second;
843 }
844 } else {
845 pin_result.ref_expr_raw = ref_it->second;
846 spdlog::debug("Pin -> Expression: {} -> {}", pin_name, pin_result.ref_expr_raw);
847 }
848
849 if (comp_it == comp_outpin_map_.end()) {
850 std::string comp_err = "Pin not found in comparison map.";
851 pin_result.error_message += (pin_result.error_message.empty() ? "" : "\n") + comp_err;
852 spdlog::warn("Cell '{}', Pin '{}': {}", cell_name_, pin_name, comp_err);
853 pin_result.comparison_possible = false;
854 // If ref existed, store it
855 if (ref_it != ref_outpin_map_.end()) {
856 pin_result.ref_expr_raw = ref_it->second; // Already stored if ref exists
857 }
858 } else {
859 pin_result.comp_expr_raw = comp_it->second;
860 spdlog::debug("Pin => Expression: {} => {}", pin_name, pin_result.comp_expr_raw);
861 }
862
863 // If pin missing in either, we can't compare, but store result and continue
864 if (!pin_result.comparison_possible) {
865 all_pin_results_[pin_name] = pin_result;
866 continue;
867 }
868
869 // 4. Preprocess expressions
870 std::string ref_processed = preprocessExpression(pin_result.ref_expr_raw);
871 std::string comp_processed = preprocessExpression(pin_result.comp_expr_raw);
872
873 if (ref_processed.empty()) {
874 pin_result.error_message += "\nReference expression became empty after preprocessing.";
875 spdlog::warn("Cell '{}', Pin '{}': {}", cell_name_, pin_name,
876 "Reference expression became empty after preprocessing.");
877 pin_result.comparison_possible = false;
878 }
879 if (comp_processed.empty()) {
880 std::string comp_err = "\nComparison expression became empty after preprocessing.";
881 pin_result.error_message += (pin_result.error_message.empty() ? "" : "\n") + comp_err;
882 spdlog::warn("Cell '{}', Pin '{}': {}", cell_name_, pin_name,
883 "Comparison expression became empty after preprocessing.");
884 pin_result.comparison_possible = false; // Mark as impossible if not already
885 }
886 // If preprocessing failed for either, store and continue
887 if (ref_processed.empty() || comp_processed.empty()) {
888 pin_result.ref_expr_processed = ref_processed; // Store possibly empty strings
889 pin_result.comp_expr_processed = comp_processed;
890 all_pin_results_[pin_name] = pin_result;
891 continue;
892 }
893
894 // 5. Extract and validate variables
895 std::vector<std::string> sorted_vars;
896 // Pass RAW expressions to extractVariables as it uses regex on original format
897 bool variables_match =
898 extractVariables(pin_result.ref_expr_raw, pin_result.comp_expr_raw, sorted_vars);
899
900 if (!variables_match) {
901 pin_result.error_message = "Variable sets do not match between expressions.";
902 // extractVariables already logs details
903 pin_result.comparison_possible = false;
904 pin_result.ref_expr_processed = ref_processed; // Store processed even if vars mismatch
905 pin_result.comp_expr_processed = comp_processed;
906 all_pin_results_[pin_name] = pin_result;
907 continue;
908 }
909 // --- Helper function: build log string ---
910 auto build_log_string = [](const auto &container) {
911 std::ostringstream oss;
912 for (auto it = container.begin(); it != container.end(); ++it) {
913 oss << *it;
914 if (std::next(it) != container.end()) {
915 oss << ", ";
916 }
917 }
918 return oss.str();
919 };
920 spdlog::info("Pin '{}': Variable sets match. Found unique variables: {}", pin_name,
921 build_log_string(sorted_vars));
922
923 // 6. Compare the single pair using the processed expressions
924 // Pass pin_result by reference - it will be populated by the function
925 compareSingleExpressionPair(ref_processed, comp_processed, sorted_vars, pin_result);
926
927 // 7. Store the detailed result for this pin
928 all_pin_results_[pin_name] = pin_result;
929
930 } // End of pin loop
931
932 spdlog::info("Logic comparison finished for cell '{}'.", cell_name_);
933}
