58bool isIdentifier(const std::string &token) {
59 if (token.empty())
60 return false;
61 // Check if the token is a valid identifier (upper alpha+ numeric + underscore)
62 bool has_alpha = false;
63 for (char c : token) {
64 if (std::isalpha(c)) {
65 has_alpha = true;
66 } else if (!std::isalnum(c) && c != '_') {
67 return false; // Contains invalid char
68 }
69 }
70 // Must start with an uppercase alphabetic character
71 return has_alpha && std::isupper(token[0]);
72}

88std::string LogicComparator::preprocessExpression(const std::string &input_expr) {
89 std::string processed = input_expr;
90 spdlog::debug("Preprocessing raw expression: {}", processed);
91
92 // 0. Trim leading/trailing whitespace first
93 processed = std::regex_replace(processed, std::regex("^\\s+|\\s+$"), "");
94 if (processed.empty()) {
95 spdlog::debug("Input expression is empty.");
96 return ""; // Handle empty input early
97 }
98
99 // --- Helper function: build log string ---
100 auto build_log_string = [](const auto &container) {
101 std::ostringstream oss;
102 for (auto it = container.begin(); it != container.end(); ++it) {
103 oss << *it;
104 if (std::next(it) != container.end()) {
105 oss << ", ";
106 }
107 }
108 return oss.str();
109 };
110
111 // 1. Handle '!' for NOT carefully BEFORE adding general spaces.
112 // Replace logical NOT '!' with " not ", ensuring not to replace '!='.
113 std::string temp_processed;
114 temp_processed.reserve(processed.length() * 1.2); // Pre-allocate a bit more space
115 for (size_t i = 0; i < processed.length(); ++i) {
116 if (processed[i] == '!') {
117 if (i + 1 < processed.length() && processed[i + 1] == '=') {
118 temp_processed += "!="; // Keep inequality operator
119 i++; // Skip the '='
120 } else {
121 temp_processed += " not "; // Replace logical NOT with spaced operator
122 }
123 } else {
124 temp_processed += processed[i];
125 }
126 }
127 processed = temp_processed;
128 spdlog::trace("After '!' to 'not' conversion: {}", processed);
129
130 // 2. Add spaces around other operators and parentheses that need them
131 processed = std::regex_replace(processed, std::regex("\\("), " ( ");
132 processed = std::regex_replace(processed, std::regex("\\)"), " ) ");
133 processed = std::regex_replace(processed, std::regex("\\+"), " + "); // OR
134 processed = std::regex_replace(processed, std::regex("\\^"), " ^ "); // XOR
135 processed = std::regex_replace(processed, std::regex("\\*"), " * "); // AND
136 // Consolidate multiple spaces into one and trim again
137 processed = std::regex_replace(processed, std::regex("\\s+"), " ");
138 processed = std::regex_replace(processed, std::regex("^\\s+|\\s+$"), "");
139 spdlog::trace("After adding spaces: {}", processed);
140
141 // 3. Replace symbolic operators with their keyword equivalents
142 // Note: 'not' is already handled.
143 processed = std::regex_replace(processed, std::regex(" \\+ "), " or ");
144 processed = std::regex_replace(processed, std::regex(" \\^ "), " xor ");
145 processed = std::regex_replace(processed, std::regex(" \\* "), " and ");
146 spdlog::trace("After operator keyword replacement: {}", processed);
147
148 // 4. Tokenize based on spaces
149 std::stringstream ss(processed);
150 // Read tokens skipping whitespace
151 std::vector<std::string> tokens{std::istream_iterator<std::string>(ss),
152 std::istream_iterator<std::string>()};
153
154 if (tokens.empty()) {
155 spdlog::debug("No tokens found after tokenization.");
156 return "";
157 }
158 spdlog::trace("Tokens after initial processing: [{}]", build_log_string(tokens));
159
160 // 5. Insert implied 'and'
161 std::vector<std::string> tokens_with_implied_and;
162 if (!tokens.empty()) {
163 tokens_with_implied_and.push_back(tokens[0]);
164 for (size_t i = 0; i < tokens.size() - 1; ++i) {
165 const std::string &current_token = tokens[i];
166 const std::string &next_token = tokens[i + 1];
167
168 // An operand ends if it's an identifier or a closing parenthesis.
169 bool current_ends_operand = isIdentifier(current_token) || current_token == ")";
170 // An operand starts if it's an identifier, an opening parenthesis, or 'not'.
171 bool next_starts_operand =
172 isIdentifier(next_token) || next_token == "(" || next_token == "not";
173
174 // We should insert 'and' if the current token ends an operand AND the next token starts one,
175 // UNLESS the current token is already an operator/opening bracket OR the next token is an
176 // operator/closing bracket. Explicit check: Don't insert 'and' if an operator already exists
177 // between them.
178 bool current_is_op_or_open =
179 isOperator(current_token) || current_token == "not" || current_token == "(";
180 bool next_is_op_or_close = isOperator(next_token) || next_token == "not" || next_token == ")";
181
182 if (current_ends_operand && next_starts_operand && !current_is_op_or_open &&
183 !next_is_op_or_close) {
184 spdlog::trace("Inserting implied 'and' between '{}' and '{}'", current_token, next_token);
185 tokens_with_implied_and.push_back("and");
186 }
187
188 tokens_with_implied_and.push_back(next_token); // Add the next token regardless
189 }
190 } else {
191 // Handle case where initial tokenization yielded no tokens
192 return "";
193 }
194
195 spdlog::trace("Tokens after implied 'and': [{}]", build_log_string(tokens_with_implied_and));
196
197 // 6. Handle 'not Identifier' ONLY during reconstruction (NEW LOGIC)
198 std::vector<std::string> final_tokens;
199 for (size_t i = 0; i < tokens_with_implied_and.size(); ++i) {
200 const std::string &current_token = tokens_with_implied_and[i];
201
202 if (current_token == "not") {
203 // Check if the *next* token is an Identifier
204 if (i + 1 < tokens_with_implied_and.size() && isIdentifier(tokens_with_implied_and[i + 1])) {
205 // Found 'not Identifier', transform to 'not(Identifier)'
206 final_tokens.push_back("not");
207 final_tokens.push_back("(");
208 final_tokens.push_back(tokens_with_implied_and[i + 1]); // The identifier
209 final_tokens.push_back(")");
210 i++; // Increment i to skip the identifier we just processed
211 } else {
212 // 'not' is followed by '(', another operator, or is at the end.
213 // Add 'not' as is, let ExprTk parse 'not(...)' or handle errors.
214 final_tokens.push_back("not");
215 }
216 } else {
217 // Token is not 'not', just add it to the final list
218 final_tokens.push_back(current_token);
219 }
220 }
221 spdlog::trace("Tokens after 'not' parenthesis handling: [{}]", build_log_string(final_tokens));
222
223 // 7. Reconstruct the final expression string from tokens
224 std::string final_expr;
225 if (!final_tokens.empty()) {
226 final_expr = final_tokens[0];
227 for (size_t i = 1; i < final_tokens.size(); ++i) {
228 const std::string &prev = final_tokens[i - 1];
229 const std::string &curr = final_tokens[i];
230
231 // Add space smartly: No space after '(', no space before ')',
232 // no space after 'not' if followed by '(', no space before ',' (in function args, if
233 // applicable)
234 bool add_space = true;
235 if (prev == "(" || curr == ")" || curr == ",") {
236 add_space = false;
237 }
238 if (prev == "not" && curr == "(") {
239 add_space = false; // Already handled by 'not(...)' structure
240 }
241 // Potentially add more rules if needed for other operators/functions
242
243 if (add_space) {
244 final_expr += " ";
245 }
246 final_expr += curr;
247 }
248 }
249
250 // 8. Final cleanup (consolidate any remaining multiple spaces and trim)
251 final_expr = std::regex_replace(final_expr, std::regex("\\s+"), " ");
252 final_expr = std::regex_replace(final_expr, std::regex("^\\s+|\\s+$"), "");
253
254 spdlog::debug("Preprocessed expression result: {}", final_expr);
255 return final_expr;
256}

497template <typename T>
498void LogicComparator::compareSingleExpressionPair(const std::string &ref_expression_processed,
499 const std::string &comp_expression_processed,
500 const std::vector<std::string> &sorted_vars,
501 PinComparisonResult &result) {
502 // Store processed expressions in the result struct
503 result.ref_expr_processed = ref_expression_processed;
504 result.comp_expr_processed = comp_expression_processed;
505 result.comparison_possible = true; // Assume comparison is possible initially
506
507 spdlog::debug("Comparing processed expressions for pin '{}':", result.pin_name);
508 spdlog::debug(" Ref : {}", ref_expression_processed);
509 spdlog::debug(" Comp: {}", comp_expression_processed);
510
511 // --- ExprTk Setup ---
512 typedef exprtk::symbol_table<T> symbol_table_t;
513 typedef exprtk::expression<T> expression_t;
514 typedef exprtk::parser<T> parser_t;
515
516 // Setup for Reference Expression
517 symbol_table_t ref_symbol_table;
518 expression_t ref_expression;
519 parser_t ref_parser;
520
521 for (const auto &var_name : sorted_vars) {
522 // Create variable, ExprTk manages its memory within the table
523 if (!ref_symbol_table.create_variable(var_name)) {
524 result.error_message = "Failed to create variable '" + var_name + "' in ref symbol table.";
525 spdlog::error(result.error_message);
526 result.comparison_possible = false;
527 return; // Cannot proceed
528 }
529 }
530 ref_expression.register_symbol_table(ref_symbol_table);
531
532 // Compile Reference Expression
533 result.ref_compiles = ref_parser.compile(ref_expression_processed, ref_expression);
534 if (!result.ref_compiles) {
535 result.error_message = "Reference expression compilation failed: " + ref_parser.error();
536 spdlog::warn("Pin '{}': {}", result.pin_name, result.error_message);
537 result.comparison_possible = false;
538 // Don't return yet, maybe the comparison one also fails
539 } else {
540 spdlog::debug("Pin '{}': Reference expression compiled successfully.", result.pin_name);
541 }
542
543 // Setup for Comparison Expression
544 symbol_table_t comp_symbol_table;
545 expression_t comp_expression;
546 parser_t comp_parser;
547
548 for (const auto &var_name : sorted_vars) {
549 if (!comp_symbol_table.create_variable(var_name)) {
550 // Append error if ref also failed, otherwise set it
551 std::string comp_err = "Failed to create variable '" + var_name + "' in comp symbol table.";
552 result.error_message += (result.error_message.empty() ? "" : "\n") + comp_err;
553 spdlog::error(comp_err);
554 result.comparison_possible = false; // Mark as impossible now
555 return; // Cannot proceed if variable creation fails
556 }
557 }
558 comp_expression.register_symbol_table(comp_symbol_table);
559
560 // Compile Comparison Expression
561 result.comp_compiles = comp_parser.compile(comp_expression_processed, comp_expression);
562 if (!result.comp_compiles) {
563 std::string comp_err = "Comparison expression compilation failed: " + comp_parser.error();
564 result.error_message += (result.error_message.empty() ? "" : "\n") + comp_err;
565 spdlog::warn("Pin '{}': {}", result.pin_name, comp_err);
566 result.comparison_possible = false; // Mark as impossible if not already
567 } else {
568 spdlog::debug("Pin '{}': Comparison expression compiled successfully.", result.pin_name);
569 }
570
571 // If either failed to compile, comparison is not possible
572 if (!result.ref_compiles || !result.comp_compiles) {
573 result.comparison_possible = false;
574 return;
575 }
576
577 // --- Truth Table Generation and Comparison ---
578 size_t num_vars = sorted_vars.size();
579 // Use unsigned long long for potentially large number of combinations
580 unsigned long long num_combinations = 1ULL << num_vars;
581 // Prevent excessively large tables (e.g., > 20 variables is 1M+ rows)
582 const unsigned long long MAX_COMBINATIONS = 1ULL << 20; // Limit to 2^20 combinations
583 if (num_vars > 20) { // Check against var count for clarity
584 result.error_message = "Too many variables (" + std::to_string(num_vars) +
585 "). Max supported for truth table is 20.";
586 spdlog::error("Pin '{}': {}", result.pin_name, result.error_message);
587 result.comparison_possible = false;
588 return;
589 }
590 if (num_combinations == 0 && num_vars > 0) { // Overflow check
591 result.error_message = "Number of combinations calculation overflowed for " +
592 std::to_string(num_vars) + " variables.";
593 spdlog::error("Pin '{}': {}", result.pin_name, result.error_message);
594 result.comparison_possible = false;
595 return;
596 }
597
598 std::vector<bool> ref_results;
599 std::vector<bool> comp_results;
600 ref_results.reserve(static_cast<size_t>(num_combinations)); // Avoid reallocations
601 comp_results.reserve(static_cast<size_t>(num_combinations));
602
603 // --- Prepare Tabulate Tables ---
604 Table ref_table;
605 Table comp_table;
606 Table::Row_t header_row;
607 header_row.push_back("#"); // Row number column
608 for (const auto &var_name : sorted_vars) {
609 header_row.push_back(var_name);
610 }
611 header_row.push_back(ref_expression_processed); // Reference expression as last column header
612 ref_table.add_row(header_row);
613
614 // Adjust header for comparison table
615 header_row.back() = comp_expression_processed; // Change last header to comparison expression
616 comp_table.add_row(header_row);
617
618 // Format header rows
619 for (size_t i = 0; i < ref_table[0].size(); ++i) {
620 ref_table[0][i].format().font_color(Color::yellow).font_style({FontStyle::bold});
621 comp_table[0][i].format().font_color(Color::yellow).font_style({FontStyle::bold});
622 }
623
624 spdlog::debug("Pin '{}': Generating truth table with {} variables ({} combinations)...",
625 result.pin_name, num_vars, num_combinations);
626
627 bool evaluation_error = false;
628 for (unsigned long long i = 0; i < num_combinations; ++i) {
629 Table::Row_t ref_data_row;
630 Table::Row_t comp_data_row;
631 ref_data_row.push_back(std::to_string(i));
632 comp_data_row.push_back(std::to_string(i));
633
634 // Set variable values for this combination
635 for (size_t j = 0; j < num_vars; ++j) {
636 // The j-th bit of i determines the value of the j-th variable
637 bool bit_value = ((i >> j) & 1ULL);
638 T var_value = bit_value ? T(1.0) : T(0.0); // ExprTk uses floating point
639
640 // Get variable reference and assign value
641 // Need error checking here in theory, but create_variable should have caught issues
642 ref_symbol_table.get_variable(sorted_vars[j])->ref() = var_value;
643 comp_symbol_table.get_variable(sorted_vars[j])->ref() = var_value;
644
645 // Add input value to table rows (as string "0" or "1")
646 std::string bit_str = bit_value ? "1" : "0";
647 ref_data_row.push_back(bit_str);
648 comp_data_row.push_back(bit_str);
649 }
650
651 // Evaluate expressions
652 T ref_val, comp_val;
653 try {
654 ref_val = ref_expression.value();
655 } catch (const std::exception &e) {
656 result.error_message +=
657 "\nReference evaluation failed at combination " + std::to_string(i) + ": " + e.what();
658 spdlog::error("Pin '{}': {}", result.pin_name, result.error_message);
659 result.comparison_possible = false;
660 evaluation_error = true;
661 break; // Stop evaluation
662 }
663 try {
664 comp_val = comp_expression.value();
665 } catch (const std::exception &e) {
666 result.error_message +=
667 "\nComparison evaluation failed at combination " + std::to_string(i) + ": " + e.what();
668 spdlog::error("Pin '{}': {}", result.pin_name, result.error_message);
669 result.comparison_possible = false;
670 evaluation_error = true;
671 break; // Stop evaluation
672 }
673
674 // Store boolean results (commonly, non-zero is true in logic contexts)
675 bool ref_bool_result = (ref_val != T(0.0));
676 bool comp_bool_result = (comp_val != T(0.0));
677
678 ref_results.push_back(ref_bool_result);

704 }
705
706 // Store the generated tables in the result struct
707 result.ref_truth_table = ref_table;
708 result.comp_truth_table = comp_table;
709 }
