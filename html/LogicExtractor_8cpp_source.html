480// Recursive function with memoization
510std::string LogicExtractor::deriveLogicRecursive(const std::string &signalName) {
511 // 1. Check Cache (Memoization)
512 if (logicCache_.count(signalName)) {
513 return logicCache_.at(signalName);
514 }
515
516 // 2. Base Case: Is it a primary input?
517 if (primaryInputs_.count(signalName)) {
518 // Already cached during port handling, but double-check
519 if (!logicCache_.count(signalName)) {
520 logicCache_[signalName] = signalName;
521 }
522 return signalName;
523 }
524
525 // 3. Recursive Step: Is it driven by a gate?
526 if (gateOutputDrivers_.count(signalName)) {
527 const GateInfo &driverGate = gateOutputDrivers_.at(signalName);
528 std::vector<std::string> inputExpressions;
529 spdlog::debug(" Tracing signal '{}', driven by {} gate", signalName,
530 driverGate.gateTypeName);
531
532 // Recursively find expressions for all inputs of this gate
533 for (const std::string &inputSig : driverGate.inputSignals) {
534 if (inputSig.empty()) {
535 throw std::runtime_error("Empty input signal name encountered for gate driving " +
536 signalName);
537 }
538 spdlog::debug(" Recursing for input: {}", inputSig);
539 inputExpressions.push_back(deriveLogicRecursive(inputSig));
540 }
541
542 // Format the expression based on gate type and input expressions
543 std::string currentExpr = formatExpression(driverGate, inputExpressions);
544
545 // Cache the result
546 logicCache_[signalName] = currentExpr;
547 return currentExpr;
548 }
549
550 // 4. Handle Assign statements (if implemented)
551 // if (assignDrivers_.count(signalName)) { ... }
552
553 // 5. Error Case: Signal not found or not driven by known element
554 // Check if it's just an internal wire that wasn't driven?
555 if (internalWires_.count(signalName)) {
556 throw std::runtime_error(
557 "Signal '" + signalName +
558 "' is an internal wire but has no identified driver (gate or assign).");
559 } else {
560 throw std::runtime_error(
561 "Signal '" + signalName +
562 "' is not a primary input, known wire, or driven by a recognized gate/assignment.");
563 }

566// Helper to format the expression string based on gate type
584std::string LogicExtractor::formatExpression(const GateInfo &gateInfo,
585 const std::vector<std::string> &inputExprs) {
586 if (inputExprs.empty() && gateInfo.kind != slang::parsing::TokenKind::NotKeyword &&
587 gateInfo.kind != slang::parsing::TokenKind::BufKeyword) {
588 // Gates like AND/OR/XOR need inputs
589 spdlog::warn("Gate type {} requires inputs, but none were provided/derived for output {}",
590 gateInfo.gateTypeName, gateInfo.outputSignal);
591 return "/*<Error: Missing Inputs for " + gateInfo.gateTypeName + ">*/";
592 }
593
594 std::string result = "";
595
596 // Use gateInfo.type (enum) for reliable checking
597 switch (gateInfo.kind) {
598 case slang::parsing::TokenKind::AndKeyword:
599 result = "(" + inputExprs[0];
600 for (size_t i = 1; i < inputExprs.size(); ++i)
601 result += " * " + inputExprs[i]; // Use * for AND
602 result += ")";
603 break;
604 case slang::parsing::TokenKind::NandKeyword:
605 result = "!(" + inputExprs[0];
606 for (size_t i = 1; i < inputExprs.size(); ++i)
607 result += " * " + inputExprs[i];
608 result += ")";
609 break;
610 case slang::parsing::TokenKind::OrKeyword:
611 result = "(" + inputExprs[0];
612 for (size_t i = 1; i < inputExprs.size(); ++i)
613 result += " + " + inputExprs[i]; // Use + for OR
614 result += ")";
615 break;
616 case slang::parsing::TokenKind::NorKeyword:
617 result = "!(" + inputExprs[0];
618 for (size_t i = 1; i < inputExprs.size(); ++i)
619 result += " + " + inputExprs[i];
620 result += ")";
621 break;
622 case slang::parsing::TokenKind::XorKeyword:
623 result = "(" + inputExprs[0];
624 for (size_t i = 1; i < inputExprs.size(); ++i)
625 result += " ^ " + inputExprs[i]; // Use ^ for XOR
626 result += ")";
627 break;
628 case slang::parsing::TokenKind::XnorKeyword:
629 result = "!(" + inputExprs[0];
630 for (size_t i = 1; i < inputExprs.size(); ++i)
631 result += " ^ " + inputExprs[i];
632 result += ")";
633 break;
634 case slang::parsing::TokenKind::NotKeyword: // NOT gate
635 if (inputExprs.size() != 1) {
636 spdlog::warn("NOT gate expects 1 input, got {}", inputExprs.size());
637 return "/*<Error: Incorrect Inputs for NOT>*/";
638 }
639 result = "!" + inputExprs[0]; // Use ! for NOT
640 break;
641 case slang::parsing::TokenKind::BufKeyword: // BUF gate
642 if (inputExprs.size() != 1) {
643 spdlog::warn("BUF gate expects 1 input, got {}", inputExprs.size());
644 return "/*<Error: Incorrect Inputs for BUF>*/";
645 }
646 result = inputExprs[0]; // Output is the same as input
647 break;
648 // Add cases for bufif0, bufif1, notif0, notif1, pullup, pulldown, cmos, nmos, pmos, tran etc. if
649 // needed
650 default:
651 spdlog::warn("Unsupported primitive gate type for logic expression generation: {}",
652 gateInfo.gateTypeName);
653 result = "/*<Unsupported Gate: " + gateInfo.gateTypeName + ">*/";
654 break;
655 }
656 return result;
657}
