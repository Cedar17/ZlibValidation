24template<typename TDerived>
25class SyntaxVisitor {
26public:
28 template<typename T>
29 void visit(T&& t) {
30 if constexpr (requires { DERIVED->handle(t); })
31 DERIVED->handle(t);
32 else
33 DERIVED->visitDefault(t);
34 }
35
38 template<typename T>
39 void visitDefault(T&& node) {
40 for (uint32_t i = 0; i < node.getChildCount(); i++) {
41 auto child = node.childNode(i);
42 if (child)
43 child->visit(*DERIVED);
44 else {
45 auto token = node.childToken(i);
46 if (token)
47 DERIVED->visitToken(token);
48 }
49 }
50 }
51
52private:
53 // This is to make things compile if the derived class doesn't provide an implementation.
54 void visitToken(parsing::Token) {}
55};

119 commits.clear();
120 tempTrees.clear();

122 tree->root().visit(*DERIVED);
